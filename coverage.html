// downloadPerm is the permission of a board saved with download.
const downloadPerm = 0o644

// execAdd appends text to the board.
func execAdd(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdin io.Reader, stdout, stderr io.Writer) int {
        text, err := readText(positionals, opts.files, stdin)
        if err != nil {
                return fail(opts.program, stderr, err)
        }
        if err := validateText(text, "the text to add"); err != nil {
                return fail(opts.program, stderr, err)
        }

        if opts.dryRun {
                current, err := client.Get(ctx, opts.board)
                if err != nil {
                        return fail(opts.program, stderr, err)
                }
                updated := boardclient.AppendText(current, text)
                return report(opts, stdout, stderr, actionReport{
                        Board:      opts.board,
                        Action:     "append",
                        Bytes:      len(text),
                        TotalBytes: len(updated),
                        DryRun:     true,
                }, "board %q: would add %d byte(s), %d byte(s) in total (dry run)\n", opts.board, len(text), len(updated))
        }

        updated, err := client.Append(ctx, opts.board, text)
        if err != nil {
                return fail(opts.program, stderr, err)
        }

        return report(opts, stdout, stderr, actionReport{
                Board:      opts.board,
                Action:     "append",
                Bytes:      len(text),
                TotalBytes: len(updated),
        }, "board %q updated: %d byte(s) added, %d byte(s) in total\n", opts.board, len(text), len(updated))
}

// execSet replaces the whole text of the board.
func execSet(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdin io.Reader, stdout, stderr io.Writer) int {
        text, err := readText(positionals, opts.files, stdin)
        if err != nil {
                return fail(opts.program, stderr, err)
        }
        if err := validateText(text, "the text to write"); err != nil {
                return fail(opts.program, stderr, err)
        }

        result := actionReport{Board: opts.board, Action: "set", Bytes: len(text), DryRun: opts.dryRun}
        if opts.dryRun {
                return report(opts, stdout, stderr, result,
                        "board %q: would be replaced with %d byte(s) (dry run)\n", opts.board, len(text))
        }

        if err := client.Set(ctx, opts.board, text); err != nil {
                return fail(opts.program, stderr, err)
        }

        return report(opts, stdout, stderr, result, "board %q replaced: %d byte(s)\n", opts.board, len(text))
}

// execRead prints the board text on stdout.
func execRead(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdout, stderr io.Writer) int {
        if len(positionals) > 0 {
                return usageError(opts.program, stderr, "read", "unexpected argument %q", positionals[0])
        }
        if opts.json && opts.stat {
                return usageError(opts.program, stderr, "read", "--json already carries the revision and the update time: drop --stat")
        }

        if opts.json || opts.stat {
                board, err := client.GetBoard(ctx, opts.board)
                if err != nil {
                        return fail(opts.program, stderr, err)
                }
                if board.Name == "" {
                        board.Name = opts.board
                }

                if opts.json {
                        if err := json.NewEncoder(stdout).Encode(board); err != nil {
                                return fail(opts.program, stderr, fmt.Errorf("writing the board state: %w", err))
                        }
                        return exitOK
                }

                writeMessage(stderr, "%s\n", statLine(board))
                if err := writeBoardText(stdout, board.Text); err != nil {
                        return fail(opts.program, stderr, err)
                }
                return exitOK
        }

        text, err := client.Get(ctx, opts.board)
        if err != nil {
                return fail(opts.program, stderr, err)
        }
        if err := writeBoardText(stdout, text); err != nil {
                return fail(opts.program, stderr, err)
        }
        return exitOK
}

// execUpload loads a board from a file, replacing its text or appending to it.
func execUpload(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdin io.Reader, stdout, stderr io.Writer) int {
        if len(positionals) != 1 {
                return usageError(opts.program, stderr, "upload", "needs exactly one file (use - to read stdin)")
        }

        source := positionals[0]
        text, err := readSource(source, stdin)
        if err != nil {
                return fail(opts.program, stderr, err)
        }
        if err := validateText(text, source); err != nil {
                return fail(opts.program, stderr, err)
        }

        if opts.appendMode {
                if opts.dryRun {
                        current, err := client.Get(ctx, opts.board)
                        if err != nil {
                                return fail(opts.program, stderr, err)
                        }
                        updated := boardclient.AppendText(current, text)
                        return report(opts, stdout, stderr, actionReport{
                                Board:      opts.board,
                                Action:     "append",
                                Source:     source,
                                Bytes:      len(text),
                                TotalBytes: len(updated),
                                DryRun:     true,
                        }, "board %q: would add %d byte(s) from %s, %d byte(s) in total (dry run)\n", opts.board, len(text), source, len(updated))
                }

                updated, err := client.Append(ctx, opts.board, text)
                if err != nil {
                        return fail(opts.program, stderr, err)
                }
                return report(opts, stdout, stderr, actionReport{
                        Board:      opts.board,
                        Action:     "append",
                        Source:     source,
                        Bytes:      len(text),
                        TotalBytes: len(updated),
                }, "board %q updated: %d byte(s) added from %s, %d byte(s) in total\n", opts.board, len(text), source, len(updated))
        }

        result := actionReport{Board: opts.board, Action: "set", Source: source, Bytes: len(text), DryRun: opts.dryRun}
        if opts.dryRun {
                return report(opts, stdout, stderr, result,
                        "board %q: would be loaded from %s with %d byte(s) (dry run)\n", opts.board, source, len(text))
        }

        if err := client.Set(ctx, opts.board, text); err != nil {
                return fail(opts.program, stderr, err)
        }

        return report(opts, stdout, stderr, result, "board %q loaded from %s: %d byte(s)\n", opts.board, source, len(text))
}

// execDownload saves the board text to a file, `<board>.txt` by default.
func execDownload(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdout, stderr io.Writer) int {
        if opts.output != "" && len(positionals) > 0 {
                return usageError(opts.program, stderr, "download", "give the output either with -o or as an argument, not both")
        }
        if len(positionals) > 1 {
                return usageError(opts.program, stderr, "download", "takes at most one output file")
        }

        target := opts.output
        if target == "" && len(positionals) == 1 {
                target = positionals[0]
        }
        if target == "" {
                target = opts.board + ".txt"
        }

        text, err := client.Get(ctx, opts.board)
        if err != nil {
                return fail(opts.program, stderr, err)
        }

        if target == "-" {
                // The board text and a JSON result would both land on stdout, so the
                // two outputs cannot be asked for together: read --json carries the text
                // and the revision in one object.
                if opts.json {
                        return usageError(opts.program, stderr, "download", "the board text and the JSON result would share stdout: drop --json or write to a file")
                }
                if err := writeBoardText(stdout, text); err != nil {
                        return fail(opts.program, stderr, err)
                }
                return exitOK
        }

        if !opts.force {
                if _, statErr := os.Stat(target); statErr == nil {
                        return fail(opts.program, stderr, fmt.Errorf("%s already exists: use --force to overwrite it", target))
                } else if !errors.Is(statErr, fs.ErrNotExist) {
                        return fail(opts.program, stderr, fmt.Errorf("checking %s: %w", target, statErr))
                }
        }

        if err := os.WriteFile(target, []byte(text), downloadPerm); err != nil {
                return fail(opts.program, stderr, fmt.Errorf("writing %s: %w", target, err))
        }

        return report(opts, stdout, stderr, actionReport{
                Board:  opts.board,
                Action: "download",
                Path:   target,
                Bytes:  len(text),
        }, "board %q saved to %s (%d byte(s))\n", opts.board, target, len(text))
}

// execFollow prints the board and then every change until the user stops it.
func execFollow(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdout, stderr io.Writer) int {
        if len(positionals) > 0 {
                return usageError(opts.program, stderr, "follow", "unexpected argument %q", positionals[0])
        }
        if opts.json {
                return usageError(opts.program, stderr, "follow", "streams the board text: --json is not supported, use read --json for one revision")
        }

        // The stream stays open until the server closes it, so Ctrl-C has to cancel
        // the request instead of killing the process mid-write.
        ctx, cancel := context.WithCancel(ctx)
        defer cancel()

        cancelled := watchSignals(cancel)
        defer cancelled()

        // The announcement waits for the first revision: a wrong URL must print the
        // error alone, without claiming the stream was opened.
        announced := false
        previous := ""

        if err := client.Follow(ctx, opts.board, func(board boardclient.Board) {
                if !announced {
                        announced = true
                        notify(opts, stderr, "following board %q on %s (press Ctrl-C to stop)\n", opts.board, opts.url)
                }

                if opts.diff {
                        writeBoardDelta(stdout, previous, board.Text, previous == "")
                        previous = board.Text
                        return
                }
                writeBoardSnapshot(stdout, board.Text)
        }); err != nil {
                return fail(opts.program, stderr, err)
        }

        return exitOK
}

// execEdit opens the board in $VISUAL/$EDITOR and writes it back when it
// changed. The editor runs attached to the terminal, so it can be interactive.
func execEdit(ctx context.Context, client *boardclient.Client, opts options, positionals []string, stdin io.Reader, stdout, stderr io.Writer) int {
        if len(positionals) > 0 {
                return usageError(opts.program, stderr, "edit", "unexpected argument %q", positionals[0])
        }

        editor := editorCommand()
        if len(editor) == 0 {
                return fail(opts.program, stderr, errors.New("no editor found: set VISUAL or EDITOR"))
        }

        current, err := client.Get(ctx, opts.board)
        if err != nil {
                return fail(opts.program, stderr, err)
        }

        path, err := writeTempText(current)
        if err != nil {
                return fail(opts.program, stderr, err)
        }
        defer removeFile(path)

        command := exec.Command(editor[0], append(editor[1:], path)...)
        command.Stdin = stdin
        command.Stdout = stdout
        command.Stderr = stderr
        if err := command.Run(); err != nil {
                return fail(opts.program, stderr, fmt.Errorf("running the editor: %w", err))
        }

        data, err := os.ReadFile(path)
        if err != nil {
                return fail(opts.program, stderr, fmt.Errorf("reading the edited text: %w", err))
        }

        updated := string(data)
        if updated == current {
                return report(opts, stdout, stderr, actionReport{
                        Board:  opts.board,
                        Action: "unchanged",
                        Bytes:  len(updated),
                }, "board %q unchanged\n", opts.board)
        }
        if err := validateText(updated, "the edited text"); err != nil {
                return fail(opts.program, stderr, err)
        }

        if err := client.Set(ctx, opts.board, updated); err != nil {
                return fail(opts.program, stderr, err)
        }

        return report(opts, stdout, stderr, actionReport{
                Board:  opts.board,
                Action: "set",
                Bytes:  len(updated),
        }, "board %q replaced: %d byte(s)\n", opts.board, len(updated))
}

// configPathReport is one configuration file the CLI looks for.
type configPathReport struct {
        Path   string `json:"path"`
        Exists bool   `json:"exists"`
}

// configReport is the machine-readable form of the config command.
type configReport struct {
        ConfigFile     string             `json:"config_file"`
        ServerURL      string             `json:"server_url"`
        Board          string             `json:"board"`
        Executable     string             `json:"executable"`
        Paths          []configPathReport `json:"paths,omitempty"`
        Environment    []string           `json:"environment,omitempty"`
        Warnings       []string           `json:"warnings,omitempty"`
        TimeoutSeconds int                `json:"timeout_secs"`
}

// execConfig prints the configuration the other commands would use. It never
// contacts the server, so a wrong URL, a configuration file that is not being
// found or a configuration that requires a flag are all visible before a
// request is made.
func execConfig(opts options, cfg boardclient.Config, candidates []string, positionals []string, stdout, stderr io.Writer) int {
        if len(positionals) > 0 {
                return usageError(opts.program, stderr, "config", "unexpected argument %q", positionals[0])
        }

        source := cfg.FilePath
        if source == "" {
                source = "none found, using the built-in defaults"
        }
        if opts.noConfig {
                source = "ignored (--no-config), using the environment and the flags"
        }

        executable := "unknown"
        if path := executablePath(); path != "" {
                executable = path
        }

        var paths []configPathReport
        if opts.paths {
                paths = make([]configPathReport, 0, len(candidates))
                for _, candidate := range candidates {
                        _, err := os.Stat(candidate)
                        paths = append(paths, configPathReport{Path: candidate, Exists: err == nil})
                }
        }

        if opts.json {
                report := configReport{
                        ConfigFile:     source,
                        ServerURL:      opts.url,
                        Board:          opts.board,
                        Executable:     executable,
                        TimeoutSeconds: int(opts.timeout.Seconds()),
                        Paths:          paths,
                        Environment:    cfg.EnvOverrides,
                        Warnings:       cfg.Warnings,
                }
                if err := json.NewEncoder(stdout).Encode(report); err != nil {
                        return fail(opts.program, stderr, fmt.Errorf("writing the configuration: %w", err))
                }
        } else {
                writeMessage(stdout, "config file:  %s\n", source)
                writeMessage(stdout, "server url:   %s\n", opts.url)
                writeMessage(stdout, "board:        %s\n", opts.board)
                writeMessage(stdout, "timeout:      %s\n", opts.timeout)
                writeMessage(stdout, "executable:   %s\n", executable)
                if len(cfg.EnvOverrides) > 0 {
                        writeMessage(stdout, "environment:  %s (overrides the file)\n", strings.Join(cfg.EnvOverrides, ", "))
                }
                for _, warning := range cfg.Warnings {
                        writeMessage(stdout, "warning:      %s\n", warning)
                }
                if opts.paths {
                        writeMessage(stdout, "\nlooked up:\n")
                        for _, path := range paths {
                                state := "missing"
                                if path.Exists {
                                        state = "found"
                                }
                                writeMessage(stdout, "  %-60s %s\n", path.Path, state)
                        }
                }
        }

        if err := (boardclient.Config{URL: opts.url, Board: opts.board, Timeout: opts.timeout}).Validate(); err != nil {
                return fail(opts.program, stderr, err)
        }

        return exitOK
}

// readText resolves the text of add and set from an argument, from -f (possibly
// repeated) or from stdin.
func readText(positionals, files []string, stdin io.Reader) (string, error) {
        if len(files) > 0 && len(positionals) > 0 {
                return "", misuse(errors.New("give the text either as an argument or with -f, not both"))
        }

        if len(files) > 0 {
                // Several -f values are joined with the same rule as add itself, so two
                // files without a trailing newline never glue their last and first line.
                var combined string
                for _, file := range files {
                        text, err := readSource(file, stdin)
                        if err != nil {
                                return "", err
                        }
                        combined = boardclient.AppendText(combined, text)
                }
                return combined, nil
        }

        if len(positionals) > 0 {
                return strings.Join(positionals, " "), nil
        }
        if isTerminal(stdin) {
                return "", misuse(errors.New("no text given: pass it as an argument, with -f or on stdin"))
        }
        return readSource("-", stdin)
}

// readSource reads a file, or stdin when name is "-".
func readSource(name string, stdin io.Reader) (string, error) {
        if name == "-" {
                data, err := io.ReadAll(stdin)
                if err != nil {
                        return "", fmt.Errorf("reading stdin: %w", err)
                }
                return string(data), nil
        }

        data, err := os.ReadFile(name)
        if err != nil {
                return "", fmt.Errorf("reading %s: %w", name, err)
        }
        return string(data), nil
}

// validateText rejects a text a board cannot hold, naming its source so the
// message points at the file to fix. The server rejects invalid UTF-8 too; the
// local check avoids a request and gives a message about the local file.
func validateText(text, source string) error {
        if utf8.ValidString(text) {
                return nil
        }
        return fmt.Errorf("%s is not valid UTF-8: boards hold text only, save it as UTF-8", source)
}

// editorCommand returns the editor to run, honouring VISUAL before EDITOR. The
// value may carry arguments, so it is split on whitespace.
func editorCommand() []string {
        for _, name := range []string{"VISUAL", "EDITOR"} {
                value := strings.TrimSpace(os.Getenv(name))
                if value == "" {
                        continue
                }
                if parts := strings.Fields(value); len(parts) > 0 {
                        return parts
                }
        }
        return nil
}

// writeTempText stores the board text in a temporary file for the editor.
func writeTempText(text string) (string, error) {
        file, err := os.CreateTemp("", "gb-cli-*.txt")
        if err != nil {
                return "", fmt.Errorf("creating the temporary file: %w", err)
        }

        path := file.Name()
        if err := file.Close(); err != nil {
                return "", fmt.Errorf("closing the temporary file: %w", err)
        }
        if err := os.WriteFile(path, []byte(text), 0o600); err != nil {
                return "", fmt.Errorf("writing the temporary file: %w", err)
        }
        return path, nil
}

// removeFile deletes a temporary file. A failure is ignored: the command has
// already produced its result and the file lives in the temporary directory.
func removeFile(path string) {
        if err := os.Remove(path); err != nil && !errors.Is(err, fs.ErrNotExist) {
                return
        }
}

// watchSignals cancels the stream on Ctrl-C. The returned function stops
// watching.
func watchSignals(cancel context.CancelFunc) func() {
        signals := make(chan os.Signal, 1)
        signal.Notify(signals, os.Interrupt, syscall.SIGTERM)
        done := make(chan struct{})

        go func() {
                select {
                case <-signals:
                        cancel()
                case <-done:
                }
        }()

        return func() {
                signal.Stop(signals)
                close(done)
        }
}

// execCompletion prints a completion script for the requested shell. It runs
// before the configuration is validated, so completing works even while the URL
// is being fixed.
func execCompletion(opts options, positionals []string, stdout, stderr io.Writer) int {
        if len(positionals) != 1 {
                return usageError(opts.program, stderr, "completion", "needs a shell: bash, zsh or fish")
        }

        var script string
        switch positionals[0] {
        case "bash":
                script = bashCompletion(opts.program)
        case "zsh":
                script = zshCompletion(opts.program)
        case "fish":
                script = fishCompletion(opts.program)
        default:
                return usageError(opts.program, stderr, "completion", "unsupported shell %q: use bash, zsh or fish", positionals[0])
        }

        writeMessage(stdout, "%s", script)
        return exitOK
}

// bashCompletion renders a bash completion script for the program.
func bashCompletion(program string) string {
        function := "complete_" + sanitizeFunctionName(program)

        return fmt.Sprintf(`# bash completion for %[1]s
%[2]s() {
    local commands="%[3]s"
    local flags="%[4]s"
    local current="${COMP_WORDS[COMP_CWORD]}"

    if [ "$COMP_CWORD" -eq 1 ]; then
        COMPREPLY=( $(compgen -W "$commands $flags" -- "$current") )
        return 0
    fi

    COMPREPLY=( $(compgen -W "$flags" -- "$current") )
}

complete -F %[2]s %[1]s
`, program, function, completionCommands, completionFlags)
}

// zshCompletion renders a zsh completion script for the program.
func zshCompletion(program string) string {
        return fmt.Sprintf(`#compdef %[1]s
# zsh completion for %[1]s

_%[2]s() {
    local commands=(%[3]s)
    local flags=(%[4]s)

    if (( CURRENT == 2 )); then
        _describe 'command' commands
        return
    fi

    _describe 'flag' flags
}

compdef _%[2]s %[1]s
`, program, sanitizeFunctionName(program), completionCommands, completionFlags)
}

// fishCompletion renders a fish completion script for the program.
func fishCompletion(program string) string {

complete -c %[1]s -f -n '__fish_use_subcommand' -a '%[2]s'
complete -c %[1]s -l board -s b -d 'board to use'

// sanitizeFunctionName turns a program name into a valid shell function name.
func sanitizeFunctionName(program string) string {
        var builder strings.Builder
        for _, char := range program {
                switch {
                case char >= 'a' && char <= 'z', char >= 'A' && char <= 'Z', char >= '0' && char <= '9':
                        builder.WriteRune(char)
                default:
                        builder.WriteRune('_')
                }
        }

        name := builder.String()
        if name == "" {
                return defaultProgramName
        }
        return name
}

// stringList collects a flag that may be repeated, such as -f.
type stringList []string

// String renders the collected values for the flag package.
func (l *stringList) String() string {
        return strings.Join(*l, ", ")
}

// Set appends one value.
func (l *stringList) Set(value string) error {
        *l = append(*l, value)
        return nil
}

// headerFlags collects repeated --header "Name: Value" values.
type headerFlags struct {
        values http.Header
}

// String renders the collected header names for the flag package.
func (h headerFlags) String() string {
        if len(h.values) == 0 {
                return ""
        }

        names := make([]string, 0, len(h.values))
        for name := range h.values {
                names = append(names, name)
        }
        return strings.Join(names, ", ")
}

// Set parses one "Name: Value" header.
func (h *headerFlags) Set(value string) error {
        name, raw, found := strings.Cut(value, ":")
        name = strings.TrimSpace(name)
        if !found || name == "" {
                return fmt.Errorf("header %q must look like \"Name: Value\"", value)
        }

        if h.values == nil {
                h.values = make(http.Header)
        }
        h.values.Add(name, strings.TrimSpace(raw))
        return nil
}

// parseFlags parses the flags and returns the positional arguments. The
// standard flag package stops at the first positional argument, which would
// silently turn a flag written after it into text, so parsing is repeated on
// the remainder until every argument is consumed.
func parseFlags(flags *flag.FlagSet, args []string) ([]string, error) {
        var positionals []string
        rest := args
        for {
                if err := flags.Parse(rest); err != nil {
                        return nil, err
                }
                remaining := flags.Args()
                if len(remaining) == 0 {
                        return positionals, nil
                }
                positionals = append(positionals, remaining[0])
                rest = remaining[1:]
        }
}

// splitCommand finds the subcommand and returns it with every other argument.
// Flags before the command are kept, so both "gb-cli -b x add text" and
// "gb-cli add -b x text" are accepted.
func splitCommand(args []string) (string, []string, error) {
        rest := make([]string, 0, len(args))

        for i := 0; i < len(args); i++ {
                arg := args[i]

                if isCommand(arg) {
                        rest = append(rest, args[i+1:]...)
                        return arg, rest, nil
                }

                if flagTakesValue(arg) {
                        rest = append(rest, arg)
                        if i+1 < len(args) {
                                rest = append(rest, args[i+1])
                                i++
                        }
                        continue
                }

                if strings.HasPrefix(arg, "-") {
                        rest = append(rest, arg)
                        continue
                }

                return "", nil, fmt.Errorf("unknown command %q", arg)
        }

        return "", nil, nil
}

// isCommand reports whether token is one of the subcommands.
func isCommand(token string) bool {
        switch token {
        case "add", "set", "read", "upload", "download", "follow", "edit", "config", "completion", "version":
                return true
        default:
                return false
        }
}

// flagTakesValue reports whether arg is a flag whose value is the next
// argument, so the value is not mistaken for the subcommand.
func flagTakesValue(arg string) bool {
        switch arg {
        case "-b", "--board", "-u", "--url", "--timeout", "--config", "--retry", "--header", "-f", "-o":
                return true
        default:
                return false
        }
}

// prescanFlag reads a string flag before the command flags are parsed, so the
// configuration file is known before its values become flag defaults. It stops
// at the `--` terminator and keeps the last occurrence, exactly like the flag
// package, so a text that starts with "-" is never read as a flag.
func prescanFlag(args []string, name string) (string, error) {
        long := "--" + name
        short := "-" + name

        value := ""
        for i := 0; i < len(args); i++ {
                arg := args[i]
                if arg == "--" {
                        return value, nil
                }

                switch {
                case arg == long || arg == short:
                        if i+1 >= len(args) {
                                return "", fmt.Errorf("flag %s needs a value", arg)
                        }
                        value = args[i+1]
                        i++
                case strings.HasPrefix(arg, long+"="):
                        value = strings.TrimPrefix(arg, long+"=")
                case strings.HasPrefix(arg, short+"="):
                        value = strings.TrimPrefix(arg, short+"=")
                }
        }
        return value, nil
}

// prescanBool reports whether a boolean flag is present, so a flag that changes
// how the configuration is loaded can be honoured before parsing. It stops at
// the `--` terminator, where every following argument is text.
func prescanBool(args []string, name string) bool {
        long := "--" + name
        short := "-" + name

        for _, arg := range args {
                if arg == "--" {
                        break
                }
                if arg == long || arg == short {
                        return true
                }
        }
        return false
}

// isHelp reports whether the invocation asks for the usage text.
func isHelp(args []string) bool {
        switch args[0] {
        case "-h", "--help", "-help", "help":
                return true
        default:
                return false
        }
}

// isVersionArg reports whether the invocation is the version flag.
func isVersionArg(arg string) bool {
        switch arg {
        case "-version", "--version", "-V":
                return true
        default:
                return false
        }
}

// boards (ghostboards) served by a running GoSpin instance.

// local server, a container or a remote host without touching any file of the

const (
        // Exit codes of the CLI. The first three follow the usual convention; the
        // last two let a script tell a wrong URL and a wrong board name from a
        // generic failure.
        exitOK        = 0
        exitError     = 1
        exitUsage     = 2
        exitEndpoint  = 3
        exitBoardName = 4
)

// defaultProgramName is the binary name used when the invoked name cannot be
// determined. It is also the name the build targets produce.
const defaultProgramName = "gb-cli"

// programPlaceholder is replaced with the invoked program name in the help
// text, so the help always matches the name of the executable.
const programPlaceholder = "{program}"

func main() {
        os.Exit(run(programName(), os.Args[1:], os.Stdin, os.Stdout, os.Stderr))
}

// programName is the name the binary was invoked as, so the help and the error
// messages follow a renamed executable.
func programName() string {
        if len(os.Args) == 0 || os.Args[0] == "" {
                return defaultProgramName
        }

        name := filepath.Base(os.Args[0])
        if name == "" || name == "." || name == string(filepath.Separator) {
                return defaultProgramName
        }
        return strings.TrimSuffix(name, ".exe")
}

// run executes one CLI invocation and returns its exit code. It is separated
// from main so the whole surface can be exercised by tests; program is the name
// the executable was invoked as.
func run(program string, args []string, stdin io.Reader, stdout, stderr io.Writer) int {
        if program == "" {
                program = defaultProgramName
        }

        if len(args) == 0 {
                writeMessage(stderr, "%s: no command given\n\n", program)
                printUsage(stderr, program)
                return exitUsage
        }
        if isHelp(args) {
                printUsage(stdout, program)
                return exitOK
        }
        if isVersionArg(args[0]) {
                return execVersion(options{program: program}, nil, stdout, stderr)
        }

        command, rest, err := splitCommand(args)
        if err != nil {
                writeMessage(stderr, "%s: %v\n\n", program, err)
                printUsage(stderr, program)
                return exitUsage
        }
        if command == "" {
                writeMessage(stderr, "%s: no command given\n\n", program)
                printUsage(stderr, program)
                return exitUsage
        }

        // The configuration file is resolved before the flags are parsed, because
        // the flag defaults come from it; --config and --no-config are therefore
        // pre-scanned.
        noConfig := prescanBool(args, "no-config")
        configPath := ""
        if !noConfig {
                if configPath, err = prescanFlag(args, "config"); err != nil {
                        writeMessage(stderr, "%s: %v\n", program, err)
                        return exitUsage
                }
        }

        loadOptions := boardclient.LoadOptions{
                HomeDir:        homeDir(),
                ExecutablePath: executablePath(),
                XDGConfigHome:  os.Getenv("XDG_CONFIG_HOME"),
                Getenv:         os.Getenv,
                ConfigPath:     configPath,
                SkipFile:       noConfig,
        }

        cfg, err := boardclient.Load(loadOptions)
        if err != nil {
                return fail(program, stderr, err)
        }

        opts := options{program: program, board: cfg.Board, url: cfg.URL, timeout: cfg.Timeout, configPath: configPath}

        flags := flag.NewFlagSet(command, flag.ContinueOnError)
        flags.SetOutput(io.Discard)
        registerSharedFlags(flags, &opts)
        registerCommandFlags(flags, &opts, command)

        positionals, err := parseFlags(flags, rest)
        if err != nil {
                if errors.Is(err, flag.ErrHelp) {
                        printUsage(stdout, program)
                        return exitOK
                }
                writeMessage(stderr, "%s %s: %v\n", program, command, err)
                return exitUsage
        }

        // A value given on the command line is a misuse of the command line, not a
        // failure of the server or of the configuration file, so it exits with the
        // code scripts use to tell the two apart.
        if opts.retry < 0 {
                return usageError(program, stderr, command, "--retry must not be negative")
        }
        if opts.timeout <= 0 && flagWasSet(flags, "timeout") {
                return usageError(program, stderr, command, "--timeout must be positive")
        }

        // The commands that only report things run before the configuration is
        // validated: they must keep working while a value is being fixed.
        switch command {
        case "version":
                return execVersion(opts, positionals, stdout, stderr)
        case "config":
                return execConfig(opts, cfg, boardclient.ConfigPaths(loadOptions), positionals, stdout, stderr)
        case "completion":
                return execCompletion(opts, positionals, stdout, stderr)
        }

        // The effective configuration is validated only now, after the flags: a
        // value from the configuration file is a default that a flag may correct.
        if err := (boardclient.Config{URL: opts.url, Board: opts.board, Timeout: opts.timeout}).Validate(); err != nil {
                return fail(program, stderr, err)
        }

        client, err := boardclient.NewWithOptions(opts.url, boardclient.Options{
                Timeout: opts.timeout,
                Retries: opts.retry,
                Headers: requestHeaders(opts),
        })
        if err != nil {
                return fail(program, stderr, err)
        }

        ctx := context.Background()
        switch command {
        case "add":
                return execAdd(ctx, client, opts, positionals, stdin, stdout, stderr)
        case "set":
                return execSet(ctx, client, opts, positionals, stdin, stdout, stderr)
        case "read":
                return execRead(ctx, client, opts, positionals, stdout, stderr)
        case "upload":
                return execUpload(ctx, client, opts, positionals, stdin, stdout, stderr)
        case "download":
                return execDownload(ctx, client, opts, positionals, stdout, stderr)
        case "follow":
                return execFollow(ctx, client, opts, positionals, stdout, stderr)
        case "edit":
                return execEdit(ctx, client, opts, positionals, stdin, stdout, stderr)
        default:
                // splitCommand only returns the commands listed above.
                return usageError(program, stderr, command, "unknown command")
        }
}

// requestHeaders merges --header with the token environment variable, so a
// board behind an authenticating proxy works without repeating the flag.
func requestHeaders(opts options) http.Header {
        headers := opts.headers.values
        if headers == nil {
                headers = make(http.Header)
        }
        if headers.Get("Authorization") != "" {
                return headers
        }
        if token := strings.TrimSpace(os.Getenv("GO_SPIN_BOARD_TOKEN")); token != "" {
                headers.Set("Authorization", "Bearer "+token)
        }
        return headers
}

// versionReport is the machine-readable form of the version command.
type versionReport struct {
        Program   string `json:"program"`
        Version   string `json:"version"`
        BuildTime string `json:"build_time"`
        GitCommit string `json:"git_commit"`
        GoVersion string `json:"go_version"`
}

// execVersion prints the build metadata the CLI was compiled with.
func execVersion(opts options, positionals []string, stdout, stderr io.Writer) int {
        if len(positionals) > 0 {
                return usageError(opts.program, stderr, "version", "unexpected argument %q", positionals[0])
        }

        if opts.json {
                report := versionReport{
                        Program:   opts.program,
                        Version:   version.Version,
                        BuildTime: version.BuildTime,
                        GitCommit: version.GitCommit,
                        GoVersion: version.GoVersion,
                }
                if err := json.NewEncoder(stdout).Encode(report); err != nil {
                        return fail(opts.program, stderr, fmt.Errorf("writing the build metadata: %w", err))
                }
                return exitOK
        }

        writeMessage(stdout, "%s %s\n", opts.program, version.BuildInfo())
        return exitOK
}

// misuseError marks a failure caused by how the command line was written, so
// it exits with the misuse code instead of the generic error code.
type misuseError struct {
        err error
}

// Error renders the underlying failure.
func (e *misuseError) Error() string {
        return e.err.Error()
}

// Unwrap exposes the underlying failure to errors.Is and errors.As.
func (e *misuseError) Unwrap() error {
        return e.err
}

// misuse marks err as a misuse of the command line.
func misuse(err error) error {
        if err == nil {
                return nil
        }
        return &misuseError{err: err}
}

// fail reports a runtime failure and maps the typed failures to their exit
// code so scripts can tell them apart.
func fail(program string, stderr io.Writer, err error) int {
        writeMessage(stderr, "%s: %v\n", program, err)

        var endpointErr *boardclient.EndpointError
        if errors.As(err, &endpointErr) {
                return exitEndpoint
        }
        var boardNameErr *boardclient.BoardNameError
        if errors.As(err, &boardNameErr) {
                return exitBoardName
        }
        var misuseErr *misuseError
        if errors.As(err, &misuseErr) {
                return exitUsage
        }
        return exitError
}

// flagWasSet reports whether a flag was given on the command line, so a value
// that is invalid only fails as a misuse when the user actually wrote it: the
// same value coming from the configuration file is a configuration error.
func flagWasSet(flags *flag.FlagSet, name string) bool {
        found := false
        flags.Visit(func(f *flag.Flag) {
                if f.Name == name {
                        found = true
                }
        })
        return found
}

// usageError reports a misuse of a command.
func usageError(program string, stderr io.Writer, command, format string, args ...any) int {
        writeMessage(stderr, "%s %s: %s\n", program, command, fmt.Sprintf(format, args...))
        return exitUsage
}

// homeDir returns the user's home directory, or an empty string when it cannot
// be determined.
func homeDir() string {
        dir, err := os.UserHomeDir()
        if err != nil {
                return ""
        }
        return dir
}

// executablePath returns the path of the running binary, or an empty string
// when it cannot be determined.
func executablePath() string {
        path, err := os.Executable()
        if err != nil {
                return ""
        }
        return path
}

// writeMessage writes a human-readable message to w. A failure is deliberately
// ignored: w is the diagnostic stream, so there is nowhere left to report it.
func writeMessage(w io.Writer, format string, args ...any) {
        if _, err := fmt.Fprintf(w, format, args...); err != nil {
                return
        }
}

// notify writes a confirmation unless the command runs quietly.
func notify(opts options, w io.Writer, format string, args ...any) {
        if opts.quiet {
                return
        }
        writeMessage(w, format, args...)
}

// actionReport is the machine-readable result of a mutation.
type actionReport struct {
        Board      string `json:"board"`
        Action     string `json:"action"`
        Source     string `json:"source,omitempty"`
        Path       string `json:"path,omitempty"`
        Bytes      int    `json:"bytes"`
        TotalBytes int    `json:"total_bytes,omitempty"`
        DryRun     bool   `json:"dry_run,omitempty"`
}

// report prints a mutation result: JSON on stdout with --json, otherwise the
// human confirmation on stderr (suppressed by --quiet). It returns the exit
// code of the command.
func report(opts options, stdout, stderr io.Writer, result actionReport, human string, args ...any) int {
        if !opts.json {
                notify(opts, stderr, human, args...)
                return exitOK
        }

        if err := json.NewEncoder(stdout).Encode(result); err != nil {
                writeMessage(stderr, "%s: writing the result: %v\n", opts.program, err)
                return exitError
        }
        return exitOK
}

// writeBoardText writes the board text verbatim. The missing final newline is
// added only on a terminal, so an interactive read does not glue the shell
// prompt to the text while pipes and files still get the exact bytes.
func writeBoardText(stdout io.Writer, text string) error {
        if text == "" {
                return nil
        }

        out := text
        if isTerminal(stdout) && !strings.HasSuffix(text, "\n") {
                out = text + "\n"
        }

        if _, err := io.WriteString(stdout, out); err != nil {
                return fmt.Errorf("writing board text: %w", err)
        }
        return nil
}

// writeBoardSnapshot prints one revision of a board in follow mode. Each event
// carries the whole text, so a revision is always printed in full; the missing
// final newline is added so consecutive revisions never glue together.
func writeBoardSnapshot(stdout io.Writer, text string) {
        writeBoardChunk(stdout, text)
}

// writeBoardDelta prints only what a revision added to the previous one. The
// first revision is printed in full; a revision that is not an extension of the
// previous one (an edit or a replacement) is printed in full too, because there
// is no meaningful delta to show.
func writeBoardDelta(stdout io.Writer, previous, current string, first bool) {
        // An empty board has nothing to show, and a blank line would only be noise
        // at the start of the stream.
        if first && current == "" {
                return
        }
        if first || !strings.HasPrefix(current, previous) {
                writeBoardSnapshot(stdout, current)
                return
        }

        // The separator add inserts belongs to the existing text, not to the new
        // lines, so it is not reprinted as an empty line.
        delta := strings.TrimPrefix(current[len(previous):], "\n")
        if delta == "" {
                return
        }
        writeBoardChunk(stdout, delta)
}

// writeBoardChunk writes a text followed by a newline when it lacks one.
func writeBoardChunk(stdout io.Writer, text string) {
        out := text
        if !strings.HasSuffix(out, "\n") {
                out += "\n"
        }

        // A failure is deliberately ignored: stdout is the stream being watched.
        if _, err := io.WriteString(stdout, out); err != nil {
                return
        }
}

// isTerminal reports whether a stream is an interactive terminal. It accepts
// readers and writers because both stdin and stdout are probed.
func isTerminal(stream any) bool {
        file, ok := stream.(*os.File)
        if !ok {
                return false
        }

        info, err := file.Stat()
        if err != nil {
                return false
        }
        return info.Mode()&os.ModeCharDevice != 0
}

// statLine formats the revision summary printed by read --stat. A board that
// was never written has no update time, and inventing one would be a lie.
func statLine(board boardclient.Board) string {
        if board.UpdatedAt == nil {
                return fmt.Sprintf("board %q: revision %d", board.Name, board.Revision)
        }
        return fmt.Sprintf("board %q: revision %d, updated %s", board.Name, board.Revision, board.UpdatedAt.UTC().Format(time.RFC3339))
}

func main() {
        cfg, err := config.LoadConfig()
        if err != nil {
                logger.WithComponent("main").Fatalf("configuration error: %v", err)
        }

        // Set log level from configuration
        logLevel, err := logrus.ParseLevel(cfg.Misc.LogLevel)
        if err != nil {
                logger.WithComponent("main").Warnf("invalid log level '%s', using 'info': %v", cfg.Misc.LogLevel, err)
                logLevel = logrus.InfoLevel
        }
        logger.Logger.SetLevel(logLevel)
        logger.WithComponent("main").Debugf("log level set to: %s", logLevel.String())
        // Emit build identity eagerly so it shows up in container logs /
        // `docker logs` output from the first line — complement to the
        // homepage footer summary rendered via the /homepage JSON response.
        logger.WithComponent("main").Infof("starting gospin %s", version.BuildInfo())
        logger.WithComponent("main").Infof("Waiting server will run on port: %d", cfg.Server.WaitingServerPort)
        logger.WithComponent("main").Infof("App will run on port: %d", cfg.Server.Port)

        repo, err := repository.NewJSONRepository(cfg.Data.FilePath)
        if err != nil {
                logger.WithComponent("main").Fatalf("cannot init repository: %v", err)
        }

        jsonDoc, err := repo.Load(context.Background())
        if err != nil {
                logger.WithComponent("main").Fatalf("cannot load data file: %v", err)
        }

        cacheStore := cache.NewStore(*jsonDoc)
        rt, err := runtime.NewRuntimeFromConfig(cfg.Misc.RuntimeType, jsonDoc, cacheStore)
        if err != nil {
                logger.WithComponent("main").Fatalf("cannot init runtime: %v", err)
        }

        app, err := appctx.New(cfg, repo, cacheStore, rt)
        if err != nil {
                logger.WithComponent("main").Fatalf("cannot init app: %v", err)
        }
        defer app.Shutdown()

        app.StartWatchers()

        gin.SetMode(cfg.Misc.GinMode)
        gin.DefaultWriter = logger.Logger.Writer()
        gin.DefaultErrorWriter = logger.Logger.Writer()

        // Setup and start the secondary waiting server
        waitingSrv := createWaitingServer(app, logger.Logger)
        go func() {
                if err := waitingSrv.ListenAndServe(fmt.Sprintf(":%d", cfg.Server.WaitingServerPort)); err != nil && !errors.Is(err, http.ErrServerClosed) {
                        logger.WithComponent("main").Errorf("Waiting server error: %v", err)
                }
        }()

        //setup main server routes and start it!
        r := route.SetupRoutes(app, logger.Logger)
        mainSrv := createGraceHttpServer(app.BaseCtx, "main-server", app.Config.Server, r)

        if err := mainSrv.ListenAndServe(fmt.Sprintf(":%d", cfg.Server.Port)); err != nil && !errors.Is(err, http.ErrServerClosed) {
                logger.WithComponent("main").Fatal(err)
        }
}

// createWaitingServer creates a secondary HTTP server dedicated to serving only the waiting page.
// It exposes a single route GET /:name that triggers RuntimeController.WaitingPage.
func createWaitingServer(app *appctx.App, logger *logrus.Logger) *httpgrace.Server {
        r := gin.New()
        r.Use(middleware.HoneybadgerMiddleware(logger))
        r.Use(gin.Recovery())

        // Create RuntimeController for the waiting page
        rc := controller.NewRuntimeController(app)
        cc := controller.NewContainerController(app.BaseCtx, app.Cache, app.Runtime, app.Config.Misc.CosmosBaseUrl, app.Config.Misc.CosmosToken)

        r.GET("/container/:name/ready", cc.Ready)
        r.GET("/:name", rc.WaitingPage)

        return createGraceHttpServer(app.BaseCtx, "waiting-server", app.Config.Server, r)
}

func createGraceHttpServer(ctx context.Context, name string, serverConfig config.ServerConfig, r *gin.Engine) *httpgrace.Server {
        slogLogger := slog.New(slog.NewTextHandler(logger.Logger.Writer(), nil))

        srv := httpgrace.NewServer(r,
                httpgrace.WithTimeout(serverConfig.ShutDownTimeout),
                httpgrace.WithSignals(syscall.SIGTERM, syscall.SIGINT),
                httpgrace.WithLogger(slogLogger),
                httpgrace.WithBeforeShutdown(func() {
                        logger.WithComponent("http").Infof("Shutting down %s server....", name)
                }),
                httpgrace.WithServerOptions(
                        httpgrace.WithReadTimeout(serverConfig.ReadTimeout),
                        httpgrace.WithWriteTimeout(serverConfig.WriteTimeout),
                        httpgrace.WithIdleTimeout(serverConfig.IdleTimeout),
                        func(srv *http.Server) {
                                srv.BaseContext = func(_ net.Listener) context.Context {
                                        return ctx
                                }
                        },
                        func(srv *http.Server) {
                                srv.ErrorLog = log.New(logger.Logger.Writer(), fmt.Sprintf("[%s] ", name), log.LstdFlags)
                        },
                ),
        )
        return srv
}

// DefaultActionsConfigPath is the default path for the actions.yml configuration file.
const DefaultActionsConfigPath = "./config/actions.yml"

// ActionController handles action execution endpoints. It acts as a server-side
// HTTP proxy: the browser sends an action name, and this controller looks up
// the technical details (method, URL, headers, body) from actions.yml and
// executes the request server-side.
type ActionController struct {
        actionsCache   *config.ActionsFile
        actionsCfgPath string
        actionsMode    string
        timeout        time.Duration

        // mu guards actionsCache and loadFailureLogged. Requests may recover the
        // configuration at runtime (see actionsConfig), so the cached pointer is
        // replaced while other goroutines read it. Declared after the fields above
        // because govet's fieldalignment wants the pointer-free data last.
        mu                sync.RWMutex
        loadFailureLogged bool
}

// NewActionController creates a new ActionController.
func NewActionController(actionsCfgPath, actionsMode string, timeout time.Duration) *ActionController {
        if actionsMode == "" {
                actionsMode = "async"
        }
        if timeout <= 0 {
                timeout = 30 * time.Second
        }

        ac := &ActionController{
                actionsCfgPath: actionsCfgPath,
                actionsMode:    actionsMode,
                timeout:        timeout,
        }

        // Load the configuration once at startup and cache it. A missing file is
        // not an error: LoadActionsConfig returns an empty action set, so any error
        // reported here means the file exists but cannot be used (invalid YAML,
        // missing method/url, unsupported URL scheme).
        if actionsCfg, err := config.LoadActionsConfig(actionsCfgPath); err != nil {
                logger.WithComponent("actions_controller").Warnf("failed to load actions config from %s: %v", actionsCfgPath, err)
                ac.loadFailureLogged = true
        } else {
                ac.actionsCache = actionsCfg
                if len(actionsCfg.Actions) == 0 {
                        logger.WithComponent("actions_controller").Infof("no actions defined in %s (create it from actions.yml.example to enable actions)", actionsCfgPath)
                }
        }

        return ac
}

// actionsConfig returns the action definitions, loading them on demand when the
// startup load failed (for example because actions.yml had a syntax error and
// was fixed afterwards). The mutex keeps concurrent first requests from writing
// the cached pointer at the same time and from re-parsing the file in a loop.
func (ac *ActionController) actionsConfig() (*config.ActionsFile, error) {
        ac.mu.RLock()
        cached := ac.actionsCache
        ac.mu.RUnlock()
        if cached != nil {
                return cached, nil
        }

        ac.mu.Lock()
        defer ac.mu.Unlock()
        if ac.actionsCache != nil { // loaded by a concurrent request while we waited
                return ac.actionsCache, nil
        }

        cfg, err := config.LoadActionsConfig(ac.actionsCfgPath)
        if err != nil {
                // Log the first failure only: while the file stays broken every request
                // would otherwise repeat the same error line.
                if !ac.loadFailureLogged {
                        logger.WithComponent("actions_controller").Errorf("failed to load actions config from %s: %v", ac.actionsCfgPath, err)
                        ac.loadFailureLogged = true
                }
                return nil, err
        }
        ac.actionsCache = cfg
        ac.loadFailureLogged = false
        return cfg, nil
}

// cachedActions returns the currently loaded definitions without triggering a
// load, for callers that must not parse the file (startup checks).
func (ac *ActionController) cachedActions() *config.ActionsFile {
        ac.mu.RLock()
        defer ac.mu.RUnlock()
        return ac.actionsCache
}

// Timeout returns the configured per-action HTTP timeout.
func (ac *ActionController) Timeout() time.Duration {
        return ac.timeout
}

// actionRequest is the JSON body sent by the frontend.
type actionRequest struct {
        ActionName string `json:"actionName" binding:"required"`
}

// LogDanglingReferences warns about action names referenced in homepage.yaml
// (via the visual actions section) that are not defined in actions.yml. This is
// a non-fatal startup check: a dangling reference would otherwise only fail at
// click time with a generic 404. It returns the list of dangling names found.
func (ac *ActionController) LogDanglingReferences(referenced []string) []string {
        var dangling []string
        if len(referenced) == 0 {
                return dangling
        }

        actionsCfg := ac.cachedActions()
        if actionsCfg == nil {
                logger.WithComponent("actions_controller").Warnf("skipping dangling-reference check, actions config not loaded")
                return dangling
        }
        if len(actionsCfg.Actions) == 0 {
                // actions.yml is optional: with no defined action every reference would
                // look dangling, so a fresh checkout would log one warning per homepage
                // entry. Nothing can be resolved, so the check is skipped.
                logger.WithComponent("actions_controller").Debugf("skipping dangling-reference check, no actions defined in %s", ac.actionsCfgPath)
                return dangling
        }

        for _, name := range referenced {
                if name == "" {
                        // A homepage item without an action reference has nothing to resolve.
                        continue
                }
                if _, ok := actionsCfg.Actions[name]; !ok {
                        dangling = append(dangling, name)
                        logger.WithComponent("actions_controller").Warnf("action %q is referenced in homepage.yaml but not defined in %s", name, ac.actionsCfgPath)
                }
        }
        return dangling
}

// actionResponse is the JSON response returned to the frontend.
// In sync mode, StatusCode and Body are both populated.
// In async mode, only StatusCode is populated.
// Mode is always returned so the frontend can render the correct layout
// instead of inferring it from the presence of a body.
type actionResponse struct {
        Body       string `json:"body,omitempty"`
        Mode       string `json:"mode"`
        StatusCode int    `json:"statusCode"`
}

// errorResponse writes a JSON error body that also carries the configured
// action mode, so the frontend can render the result modal consistently in
// every branch (success and error) instead of inferring mode from the body.
func (ac *ActionController) errorResponse(c *gin.Context, status int, format string, args ...interface{}) {
        c.JSON(status, gin.H{
                "error": fmt.Sprintf(format, args...),
                "mode":  ac.actionsMode,
        })
}

// ExecuteAction handles POST /homepage/action. It looks up the action by name,
// executes the configured HTTP request server-side, and returns the result.
func (ac *ActionController) ExecuteAction(c *gin.Context) {
        var req actionRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                ac.errorResponse(c, http.StatusBadRequest, "actionName is required")
                return
        }

        // Use the cached configuration; it is loaded on demand (and retried) only
        // when the startup load failed.
        actionsCfg, err := ac.actionsConfig()
        if err != nil {
                // Already logged once by actionsConfig; do not repeat the same failure.
                ac.errorResponse(c, http.StatusInternalServerError, "actions configuration unavailable")
                return
        }

        actionCfg, ok := actionsCfg.Actions[req.ActionName]
        if !ok {
                logger.WithComponent("actions_controller").Warnf("action %q not found in actions config", req.ActionName)
                ac.errorResponse(c, http.StatusNotFound, "action %q not found", req.ActionName)
                return
        }

        // Build the HTTP request
        httpReq, err := http.NewRequest(actionCfg.Method, actionCfg.URL, bytes.NewBufferString(actionCfg.Body))
        if err != nil {
                logger.WithComponent("actions_controller").Errorf("failed to build HTTP request for action %q: %v", req.ActionName, err)
                ac.errorResponse(c, http.StatusInternalServerError, "failed to build HTTP request")
                return
        }

        // Honor the request context so client disconnects and the gin request
        // timeout (configured per-route from actions_timeout_seconds) actually
        // abort the outbound call instead of letting it run until its own timer.
        httpReq = httpReq.WithContext(c.Request.Context())

        for key, value := range actionCfg.Headers {
                httpReq.Header.Set(key, value)
        }

        // Execute the request
        client := &http.Client{Timeout: ac.timeout}
        logger.WithComponent("actions_controller").Infof("executing action %q: %s %s (mode=%s)", req.ActionName, actionCfg.Method, actionCfg.URL, ac.actionsMode)

        resp, err := client.Do(httpReq)
        if err != nil {
                logger.WithComponent("actions_controller").Errorf("action %q HTTP request failed: %v", req.ActionName, err)
                ac.errorResponse(c, http.StatusBadGateway, "action %q failed: %v", req.ActionName, err)
                return
        }
        defer func() {
                if cerr := resp.Body.Close(); cerr != nil {
                        logger.WithComponent("actions_controller").Debugf("failed to close response body for action %q: %v", req.ActionName, cerr)
                }
        }()

        // Build the response
        result := actionResponse{
                StatusCode: resp.StatusCode,
                Mode:       ac.actionsMode,
        }

        // In sync mode, also return the response body
        if ac.actionsMode == "sync" {
                bodyBytes, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20)) // 1 MB limit
                if err != nil {
                        logger.WithComponent("actions_controller").Warnf("action %q: failed to read response body: %v", req.ActionName, err)
                } else {
                        result.Body = string(bodyBytes)
                }
        }

        logger.WithComponent("actions_controller").Infof("action %q completed: status=%d (mode=%s)", req.ActionName, resp.StatusCode, ac.actionsMode)
        c.JSON(http.StatusOK, result)
}

// removeOrphansRequest carries the names the client wants to delete. The client
// is expected to send back exactly the list returned by GET /containers/orphans.
type removeOrphansRequest struct {
        Names []string `json:"names"`
}

// removeOrphansResponse reports what the server actually deleted after
// re-verifying each name against the runtime.
type removeOrphansResponse struct {
        Removed []string `json:"removed"`
        Skipped []string `json:"skipped"`
}

// ListOrphanContainers handles GET /containers/orphans.
// It compares the registry (cache) against the containers known to the runtime
// (docker ps -a for the Docker runtime) and returns the registry entries that no
// longer exist. It only reads state: nothing is deleted.
func (cc *ContainerController) ListOrphanContainers(c *gin.Context) {
        logger.WithComponent("container-controller").Debugf("GET /containers/orphans handler called")

        orphans, err := cc.orphanContainers(c.Request.Context())
        if err != nil {
                logger.WithComponent("container-controller").Errorf("list orphan containers: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "unable to compare the registry with the container runtime"})
                return
        }

        logger.WithComponent("container-controller").Infof("orphan container check found %d missing containers", len(orphans))
        c.JSON(http.StatusOK, orphans)
}

// RemoveOrphanContainers handles POST /containers/orphans/remove.
// Every requested name is re-verified against the runtime before deletion, so a
// container that reappeared between preview and confirmation is skipped instead
// of being removed. Deleting a container also removes the schedules targeting
// it and its references inside groups.
func (cc *ContainerController) RemoveOrphanContainers(c *gin.Context) {
        logger.WithComponent("container-controller").Debugf("POST /containers/orphans/remove handler called")

        var req removeOrphansRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                c.JSON(http.StatusBadRequest, gin.H{"error": "invalid payload"})
                return
        }

        names := dedupeNonEmpty(req.Names)
        if len(names) == 0 {
                c.JSON(http.StatusBadRequest, gin.H{"error": "no container names provided"})
                return
        }

        orphans, err := cc.orphanContainers(c.Request.Context())
        if err != nil {
                logger.WithComponent("container-controller").Errorf("remove orphan containers: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "unable to compare the registry with the container runtime"})
                return
        }

        candidates := make(map[string]struct{}, len(orphans))
        for _, container := range orphans {
                candidates[container.Name] = struct{}{}
        }

        response := removeOrphansResponse{Removed: []string{}, Skipped: []string{}}
        for _, name := range names {
                if _, ok := candidates[name]; !ok {
                        logger.WithComponent("container-controller").Warnf("skipping orphan removal for %s: container exists in the runtime or is not registered", name)
                        response.Skipped = append(response.Skipped, name)
                        continue
                }

                if _, err := cc.store.RemoveContainer(name); err != nil {
                        logger.WithComponent("container-controller").Errorf("failed to remove orphan container %s: %v", name, err)
                        c.JSON(http.StatusInternalServerError, gin.H{
                                "error":   fmt.Sprintf("failed to remove container %q", name),
                                "removed": response.Removed,
                                "skipped": response.Skipped,
                        })
                        return
                }

                logger.WithComponent("container-controller").Infof("removed orphan container %s", name)
                response.Removed = append(response.Removed, name)
        }

        logger.WithComponent("container-controller").Infof("orphan cleanup removed %d containers and skipped %d", len(response.Removed), len(response.Skipped))
        c.JSON(http.StatusOK, response)
}

// orphanContainers returns the registry entries whose name is absent from the
// runtime. The registry order is preserved so the UI shows a stable list.
func (cc *ContainerController) orphanContainers(ctx context.Context) ([]repository.Container, error) {
        doc, err := cc.store.Snapshot()
        if err != nil {
                return nil, fmt.Errorf("snapshot registry: %w", err)
        }

        names, err := cc.runtime.ListContainers(ctx)
        if err != nil {
                return nil, fmt.Errorf("list runtime containers: %w", err)
        }

        present := make(map[string]struct{}, len(names))
        for _, name := range names {
                present[name] = struct{}{}
        }

        orphans := make([]repository.Container, 0)
        for _, container := range doc.Containers {
                if _, ok := present[container.Name]; ok {
                        continue
                }
                orphans = append(orphans, container)
        }

        return orphans, nil
}

// dedupeNonEmpty trims the input names, drops empty entries and removes duplicates
// while preserving the original order.
func dedupeNonEmpty(names []string) []string {
        seen := make(map[string]struct{}, len(names))
        out := make([]string, 0, len(names))
        for _, name := range names {
                trimmed := strings.TrimSpace(name)
                if trimmed == "" {
                        continue
                }
                if _, ok := seen[trimmed]; ok {
                        continue
                }
                seen[trimmed] = struct{}{}
                out = append(out, trimmed)
        }
        return out
}

// Ready checks whether the container identified by name is reachable and responding 200.
// Route: GET /container/:name/ready
func (cc *ContainerController) Ready(c *gin.Context) {
        name := c.Param("name")
        logger.WithComponent("container-controller").Debugf("GET /container/%s/ready handler called", name)
        if name == "" {
                c.JSON(http.StatusBadRequest, gin.H{"ready": false})
                return
        }

        // Assert underlying service type to access Store and Runtime
        svc, ok := cc.crud.Service.(*ContainerCrudService)
        if !ok {
                logger.WithComponent("container-controller").Errorf("ready: unexpected service type")
                c.JSON(http.StatusInternalServerError, gin.H{"ready": false})
                return
        }

        // Find container in store snapshot
        doc, err := svc.Store.Snapshot()
        if err != nil {
                logger.WithComponent("container-controller").Errorf("ready: failed to snapshot store: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"ready": false})
                return
        }

        var container *repository.Container
        for i := range doc.Containers {
                if doc.Containers[i].Name == name {
                        container = &doc.Containers[i]
                        break
                }
        }
        if container == nil {
                logger.WithComponent("container-controller").Warnf("ready: container not found: %s", name)
                c.JSON(http.StatusNotFound, gin.H{"ready": false})
                return
        }

        // Check runtime
        running, err := svc.Runtime.IsRunning(svc.Ctx, container.Name)
        if err != nil {
                logger.WithComponent("container-controller").Warnf("ready: runtime check failed for %s: %v", container.Name, err)
                c.JSON(http.StatusOK, gin.H{"ready": false})
                return
        }
        if !running {
                c.JSON(http.StatusOK, gin.H{"ready": false})
                return
        }

        if container.URL == "" {
                logger.WithComponent("container-controller").Warnf("ready: container URL is empty: %s", name)
                c.JSON(http.StatusInternalServerError, gin.H{"ready": false})
                return
        }

        containerURL := container.URL
        if !strings.HasPrefix(containerURL, "http://") && !strings.HasPrefix(containerURL, "https://") {
                containerURL = "https://" + containerURL
        }
        if !strings.HasSuffix(containerURL, "/") {
                containerURL = containerURL + "/"
        }

        // Perform GET with timeout
        reqCtx, cancel := context.WithTimeout(c.Request.Context(), 1*time.Second)
        defer cancel()

        req, err := http.NewRequestWithContext(reqCtx, http.MethodGet, containerURL, nil)
        if err != nil {
                logger.WithComponent("container-controller").Warnf("ready: failed to create request for %s and url %s: %v", container.Name, containerURL, err)
                c.JSON(http.StatusOK, gin.H{"ready": false})
                return
        }
        client := &http.Client{}
        resp, err := client.Do(req)
        if err != nil {
                logger.WithComponent("container-controller").Warnf("ready: request failed for %s and url %s: %v", container.Name, containerURL, err)
                c.JSON(http.StatusOK, gin.H{"ready": false})
                return
        } else {
                logger.WithComponent("container-controller").Debugf("ready: request succeeded for %s and url %s with status %d", container.Name, containerURL, resp.StatusCode)
        }

        defer func() {
                if cerr := resp.Body.Close(); cerr != nil {
                        logger.WithComponent("container-controller").Debugf("failed to close response body for %s: %v", container.Name, cerr)
                }
        }()

        isContainerUrlReady := resp.StatusCode == http.StatusOK || resp.StatusCode == http.StatusPermanentRedirect || resp.StatusCode == http.StatusTemporaryRedirect
        logger.WithComponent("container-controller").Debugf("GET /container/%s/ready handled with status: %v", name, isContainerUrlReady)
        c.JSON(http.StatusOK, gin.H{"ready": isContainerUrlReady})
}

// CrudService defines the minimal interface required for CRUD operations.
type CrudService[T any] interface {
        All() ([]T, error)
        Add(item T) ([]T, error)
        Remove(name string) ([]T, error)
}

// CrudValidator defines the interface for validating a resource.
type CrudValidator[T any] interface {
        Validate(item T) error
}

// CrudController provides generic CRUD handlers for resources.
type CrudController[T any] struct {
        Service   CrudService[T]
        Validator CrudValidator[T]
}

// RegisterCrudRoutes registers CRUD endpoints for a resource on the given router group.
func (cc *CrudController[T]) RegisterCrudRoutes(rg *gin.RouterGroup, resource string) {
        rg.GET("/"+resource+"s", cc.GetAll)
        rg.POST("/"+resource, cc.CreateOrUpdate)
        rg.DELETE("/"+resource+"/:name", cc.Delete)
}

// GetAll handles GET requests to list all resources.
func (cc *CrudController[T]) GetAll(c *gin.Context) {
        items, err := cc.Service.All()
        if err != nil {
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to read resource list"})
                return
        }
        c.JSON(http.StatusOK, items)
}

// CreateOrUpdate handles POST requests to create or update a resource.
func (cc *CrudController[T]) CreateOrUpdate(c *gin.Context) {
        var item T
        if err := c.ShouldBindJSON(&item); err != nil {
                c.JSON(http.StatusBadRequest, gin.H{"error": "invalid payload"})
                return
        }
        if cc.Validator != nil {
                if err := cc.Validator.Validate(item); err != nil {
                        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
                        return
                }
        }
        items, err := cc.Service.Add(item)
        if err != nil {
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to update resource"})
                return
        }
        c.JSON(http.StatusOK, items)
}

// Delete handles DELETE requests to remove a resource by name.
func (cc *CrudController[T]) Delete(c *gin.Context) {
        name := c.Param("name")
        if name == "" {
                c.JSON(http.StatusBadRequest, gin.H{"error": "missing resource name"})
                return
        }
        items, err := cc.Service.Remove(name)
        if err != nil {
                // Check for specific "not found" errors
                if errors.Is(err, cache.ErrContainerNotFound) ||
                        errors.Is(err, cache.ErrGroupNotFound) ||
                        errors.Is(err, cache.ErrScheduleNotFound) {
                        c.JSON(http.StatusNotFound, gin.H{"error": "resource not found"})
                        return
                }
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to delete resource"})
                return
        }
        c.JSON(http.StatusOK, items)
}

const (
        // ghostboardClientHeader carries the per-tab identifier used to suppress
        // the echo of a client's own writes in its event stream.
        ghostboardClientHeader = "X-Ghostboard-Client"
        ghostboardClientIDMax  = 64

        // ghostboardBodyOverhead leaves room for the envelope around the text (JSON
        // keys, form field name) when enforcing the configured board size on the
        // request body.
        ghostboardBodyOverhead = 1024

        // ghostboardJSONEscapeFactor bounds how much a single text byte can grow when
        // encoded inside a JSON string: encoding/json writes control characters and
        // HTML-sensitive bytes as \u00XX, i.e. six bytes per input byte. Sizing the
        // body limit on the raw text would reject payloads that are within
        // max_text_bytes once decoded.
        ghostboardJSONEscapeFactor = 6

        // ghostboardFormEscapeFactor bounds the same growth for a form field, where
        // percent-encoding expands one byte to three ("%0A").
        ghostboardFormEscapeFactor = 3
)

// errGhostboardInvalidBody marks a payload that could not be decoded.
var errGhostboardInvalidBody = errors.New("invalid request body")

// errGhostboardInvalidText marks a text body that is not valid UTF-8, which
// would be mangled by the JSON persistence of the store.
var errGhostboardInvalidText = errors.New("request body is not valid UTF-8")

// GhostboardController exposes the shared text boards over HTTP.
type GhostboardController struct {
        store        *ghostboard.Store
        maxTextBytes int
        heartbeat    time.Duration
}

// NewGhostboardController creates a controller for the given store.
func NewGhostboardController(store *ghostboard.Store, maxTextBytes int, heartbeat time.Duration) *GhostboardController {
        return &GhostboardController{
                store:        store,
                maxTextBytes: maxTextBytes,
                heartbeat:    heartbeat,
        }
}

// Get returns the current text of a board.
func (gc *GhostboardController) Get(c *gin.Context) {
        board, err := gc.store.Get(ghostboardBoardName(c))
        if err != nil {
                writeGhostboardError(c, err)
                return
        }
        c.JSON(http.StatusOK, board)
}

// Update stores a new text for a board. The body can be a JSON object with a
// "text" field, a form field named "text", or the raw text itself.
func (gc *GhostboardController) Update(c *gin.Context) {
        name := ghostboardBoardName(c)
        if err := ghostboard.ValidateBoardName(name); err != nil {
                writeGhostboardError(c, err)
                return
        }

        text, err := gc.readText(c)
        if err != nil {
                writeGhostboardError(c, err)
                return
        }

        board, err := gc.store.Update(name, text, ghostboardClientID(c))
        if err != nil {
                writeGhostboardError(c, err)
                return
        }

        c.JSON(http.StatusOK, board)
}

// GetText returns a board's text as plain text for the CLI API. A board that
// was never written is served as an empty body with status 200, so a script
// can treat a missing board exactly like a cleared one.
func (gc *GhostboardController) GetText(c *gin.Context) {
        board, err := gc.store.Get(ghostboardBoardName(c))
        if err != nil {
                writeGhostboardError(c, err)
                return
        }

        c.Header("Cache-Control", "no-store")
        c.Data(http.StatusOK, "text/plain; charset=utf-8", []byte(board.Text))
}

// SetText replaces a board's text with the raw request body. The bytes are
// stored exactly as received: no JSON, form or newline handling. A successful
// write answers 204 No Content.
func (gc *GhostboardController) SetText(c *gin.Context) {
        name := ghostboardBoardName(c)
        if err := ghostboard.ValidateBoardName(name); err != nil {
                writeGhostboardError(c, err)
                return
        }

        text, err := gc.readRawText(c)
        if err != nil {
                writeGhostboardError(c, err)
                return
        }

        if _, err := gc.store.Update(name, text, ghostboardClientID(c)); err != nil {
                writeGhostboardError(c, err)
                return
        }

        c.Status(http.StatusNoContent)
}

// Events streams board changes as Server-Sent Events. The first event carries
// the current snapshot, and a keep-alive comment is sent periodically so
// proxies do not close the idle connection.
func (gc *GhostboardController) Events(c *gin.Context) {
        name := ghostboardBoardName(c)

        sub, snapshot, err := gc.store.Subscribe(name)
        if err != nil {
                writeGhostboardError(c, err)
                return
        }
        defer sub.Close()

        // The server sets an absolute write deadline per request; long-lived
        // streams must clear it or the connection would be closed mid-stream.
        controller := http.NewResponseController(c.Writer)
        if err := controller.SetWriteDeadline(time.Time{}); err != nil {
                logger.WithComponent("ghostboard").Warnf("cannot clear write deadline for board %q stream: %v", name, err)
        }

        c.Header("Content-Type", "text/event-stream")
        c.Header("Cache-Control", "no-cache")
        c.Header("Connection", "keep-alive")
        c.Header("X-Accel-Buffering", "no")
        c.Status(http.StatusOK)

        if !writeGhostboardEvent(controller, c, ghostboard.Event{Board: snapshot}) {
                return
        }

        ticker := time.NewTicker(gc.heartbeat)
        defer ticker.Stop()

        for {
                select {
                case <-c.Request.Context().Done():
                        return
                case event, ok := <-sub.Events():
                        if !ok {
                                return
                        }
                        if !writeGhostboardEvent(controller, c, event) {
                                return
                        }
                case <-ticker.C:
                        if _, err := fmt.Fprint(c.Writer, ": keep-alive\n\n"); err != nil {
                                return
                        }
                        if err := controller.Flush(); err != nil {
                                return
                        }
                }
        }
}

// bodyLimit returns the largest body accepted for a content type, derived from
// the configured board size so escaped payloads are not rejected while the text
// they decode to is still within max_text_bytes.
func (gc *GhostboardController) bodyLimit(contentType string) int64 {
        factor := int64(1)
        switch contentType {
        case "application/json":
                factor = ghostboardJSONEscapeFactor
        case "application/x-www-form-urlencoded":
                factor = ghostboardFormEscapeFactor
        }
        return int64(gc.maxTextBytes)*factor + ghostboardBodyOverhead
}

func (gc *GhostboardController) readText(c *gin.Context) (string, error) {
        c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, gc.bodyLimit(c.ContentType()))

        var text string
        switch c.ContentType() {
        case "application/json":
                var payload struct {
                        Text string `json:"text"`
                }
                if err := c.ShouldBindJSON(&payload); err != nil {
                        return "", ghostboardBodyError(err)
                }
                text = payload.Text
        case "application/x-www-form-urlencoded":
                // A body that is not a text= form field is stored as-is, matching the
                // upstream Ghostboard behaviour for plain curl -d payloads.
                body, err := io.ReadAll(c.Request.Body)
                if err != nil {
                        return "", ghostboardBodyError(err)
                }
                text = string(body)
                if values, err := url.ParseQuery(text); err == nil {
                        if texts, ok := values["text"]; ok && len(texts) > 0 {
                                text = texts[0]
                        }
                }
        default:
                body, err := io.ReadAll(c.Request.Body)
                if err != nil {
                        return "", ghostboardBodyError(err)
                }
                text = string(body)
        }

        // The limit is enforced on the decoded text, so a body padded with JSON
        // escapes cannot bypass it and an escaped body is not rejected for its size.
        if len(text) > gc.maxTextBytes {
                return "", ghostboard.ErrTextTooLarge
        }

        // Boards hold text: invalid UTF-8 would be replaced by U+FFFD when the
        // store marshals the board to its JSON file, silently rewriting what the
        // client sent. JSON decoding already yields valid strings, so this rejects
        // raw and percent-decoded bodies.
        if !utf8.ValidString(text) {
                return "", errGhostboardInvalidText
        }

        return text, nil
}

// readRawText reads the whole request body verbatim, enforcing the configured
// board size on the bytes received. Boards hold text, so a body that is not
// valid UTF-8 is rejected instead of being stored and later mangled by the
// JSON persistence.
func (gc *GhostboardController) readRawText(c *gin.Context) (string, error) {
        c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, int64(gc.maxTextBytes))

        body, err := io.ReadAll(c.Request.Body)
        if err != nil {
                return "", ghostboardBodyError(err)
        }
        if !utf8.Valid(body) {
                return "", errGhostboardInvalidText
        }
        return string(body), nil
}

func writeGhostboardEvent(controller *http.ResponseController, c *gin.Context, event ghostboard.Event) bool {
        payload, err := json.Marshal(event)
        if err != nil {
                logger.WithComponent("ghostboard").Errorf("cannot encode event for board %q: %v", event.Name, err)
                return false
        }

        if _, err := fmt.Fprintf(c.Writer, "event: board\ndata: %s\n\n", payload); err != nil {
                return false
        }
        if err := controller.Flush(); err != nil {
                logger.WithComponent("ghostboard").Debugf("cannot flush event for board %q: %v", event.Name, err)
                return false
        }
        return true
}

func ghostboardBoardName(c *gin.Context) string {
        name := strings.TrimSpace(c.Param("board"))
        if name == "" {
                return ghostboard.DefaultBoard
        }
        return name
}

func ghostboardClientID(c *gin.Context) string {
        clientID := strings.TrimSpace(c.GetHeader(ghostboardClientHeader))
        if len(clientID) > ghostboardClientIDMax {
                return clientID[:ghostboardClientIDMax]
        }
        return clientID
}

func ghostboardBodyError(err error) error {
        var maxBytesErr *http.MaxBytesError
        if errors.As(err, &maxBytesErr) {
                return ghostboard.ErrTextTooLarge
        }
        return fmt.Errorf("%w: %v", errGhostboardInvalidBody, err)
}

func writeGhostboardError(c *gin.Context, err error) {
        switch {
        case errors.Is(err, ghostboard.ErrInvalidBoardName):
                c.JSON(http.StatusBadRequest, gin.H{"error": "invalid board name: use 1-64 letters, digits, dots, dashes or underscores"})
        case errors.Is(err, ghostboard.ErrTextTooLarge):
                c.JSON(http.StatusRequestEntityTooLarge, gin.H{"error": err.Error()})
        case errors.Is(err, errGhostboardInvalidText):
                c.JSON(http.StatusBadRequest, gin.H{"error": "request body is not valid UTF-8: boards hold text only"})
        case errors.Is(err, errGhostboardInvalidBody):
                c.JSON(http.StatusBadRequest, gin.H{"error": "invalid request body"})
        default:
                logger.WithComponent("ghostboard").Errorf("ghostboard request failed: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "internal error"})
        }
}

// StartGroup handles POST /group/:name/start - starts all containers in a group.
func (gc *GroupController) StartGroup(c *gin.Context) {
        name := c.Param("name")
        logger.WithComponent("group-controller").Debugf("POST /group/%s/start handler called", name)
        if name == "" {
                logger.WithComponent("group-controller").Debugf("start group: missing name parameter")
                c.JSON(http.StatusBadRequest, gin.H{"error": "missing group name"})
                return
        }

        doc, err := gc.store.Snapshot()
        if err != nil {
                logger.WithComponent("group-controller").Errorf("start group %s: failed to read snapshot: %v", name, err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to read group data"})
                return
        }

        // Find the group
        var group *repository.Group
        for i := range doc.Groups {
                if doc.Groups[i].Name == name {
                        group = &doc.Groups[i]
                        break
                }
        }
        if group == nil {
                logger.WithComponent("group-controller").Debugf("start group %s: not found", name)
                c.JSON(http.StatusNotFound, gin.H{"error": "group not found"})
                return
        }

        if group.Active == nil || !*group.Active {
                logger.WithComponent("group-controller").Debugf("start group %s: group is not active", name)
                c.JSON(http.StatusForbidden, gin.H{"error": "group is not active"})
                return
        }

        // Start all containers in the group in background
        for _, containerName := range group.Container {
                gc.startContainerInBackground(containerName)
        }

        logger.WithComponent("group-controller").Infof("group %s: started %d containers in background", name, len(group.Container))
        c.JSON(http.StatusOK, gin.H{
                "name":       name,
                "message":    "group containers starting",
                "containers": group.Container,
        })
}

// sanitizeGroups removes from each group any container names that are not
// present in the document's Containers list.
func sanitizeGroups(doc repository.DataDocument) []repository.Group {
        // Build set of existing container names
        containerSet := make(map[string]struct{}, len(doc.Containers))
        for _, c := range doc.Containers {
                containerSet[c.Name] = struct{}{}
        }

        sanitized := make([]repository.Group, 0, len(doc.Groups))
        for _, g := range doc.Groups {
                newContainers := make([]string, 0, len(g.Container))
                for _, cname := range g.Container {
                        if _, ok := containerSet[cname]; ok {
                                newContainers = append(newContainers, cname)
                        }
                }
                g.Container = newContainers
                sanitized = append(sanitized, g)
        }
        return sanitized
}

// IsRunning checks if a container is currently running.
func (rc *RuntimeController) IsRunning(c *gin.Context) {
        name := c.Param("name")
        if name == "" {
                c.JSON(http.StatusBadRequest, gin.H{"error": "missing container name"})
                return
        }

        // Check if container exists in cache
        doc, err := rc.containerStore.Snapshot()
        if err != nil {
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to read container list"})
                return
        }

        containerExists := false
        for _, container := range doc.Containers {
                if container.Name == name {
                        containerExists = true
                        break
                }
        }
        if !containerExists {
                c.JSON(http.StatusNotFound, gin.H{"error": "container not found"})
                return
        }

        running, err := rc.runtime.IsRunning(c.Request.Context(), name)
        if err != nil {
                // Check if error is "container not found"
                if strings.Contains(err.Error(), "not found") {
                        c.JSON(http.StatusNotFound, gin.H{"error": err.Error()})
                        return
                }
                logger.WithComponent("runtime_controller").Errorf("failed to check if container %s is running: %v", name, err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "Unable to determine container running state"})
                return
        }

        c.JSON(http.StatusOK, gin.H{
                "name":    name,
                "running": running,
        })
}

// WaitingPage serves a waiting HTML page for a container or group.
// It starts containers in background if they are not running.
// Returns 404 if container/group not found, 403 if not active.
func (rc *RuntimeController) WaitingPage(c *gin.Context) {
        name := c.Param("name")
        if name == "" {
                c.JSON(http.StatusBadRequest, gin.H{"error": "missing container or group name"})
                return
        }

        doc, err := rc.containerStore.Snapshot()
        if err != nil {
                logger.WithComponent("runtime_controller").Errorf("failed to read container list: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to read container list"})
                return
        }

        // Try to find as container first
        container, found := rc.findContainer(doc, name)
        if found {
                rc.handleContainerWaitingPage(c, container)
                return
        }

        // Try to find as group
        group, found := rc.findGroup(doc, name)
        if found {
                rc.handleGroupWaitingPage(c, doc, group)
                return
        }

        // Check if container with that name exists in runtime
        found = true
        _, errRuntime := rc.runtime.IsRunning(c.Request.Context(), name)
        if errRuntime != nil {
                found = false
        }

        if found {
                containerFromRuntime := &repository.Container{
                        Name:         name,
                        FriendlyName: name,
                        URL:          strings.ReplaceAll(rc.config.Data.BaseUrl, "$1", name),
                        Active:       func(b bool) *bool { return &b }(true),
                }
                rc.handleContainerWaitingPage(c, containerFromRuntime)
                return
        }

        // Not found as container or group
        c.JSON(http.StatusNotFound, gin.H{"error": fmt.Sprintf("container or group '%s' not found", name)})
}

// findContainer searches for a container by name in the data document.
func (rc *RuntimeController) findContainer(doc repository.DataDocument, name string) (*repository.Container, bool) {
        for i := range doc.Containers {
                if doc.Containers[i].FriendlyName == name {
                        return &doc.Containers[i], true
                }
        }

        for i := range doc.Containers {
                if doc.Containers[i].Name == name {
                        return &doc.Containers[i], true
                }
        }
        return nil, false
}

// findGroup searches for a group by name in the data document.
func (rc *RuntimeController) findGroup(doc repository.DataDocument, name string) (*repository.Group, bool) {
        for i := range doc.Groups {
                if doc.Groups[i].Name == name {
                        return &doc.Groups[i], true
                }
        }
        return nil, false
}

// handleContainerWaitingPage handles the waiting page for a single container.
func (rc *RuntimeController) handleContainerWaitingPage(c *gin.Context, container *repository.Container) {
        // Check if container is active
        if container.Active == nil || !*container.Active {
                c.JSON(http.StatusForbidden, gin.H{"error": fmt.Sprintf("container '%s' is not active", container.Name)})
                return
        }

        // Check if container is running, if not start it in background
        running, err := rc.runtime.IsRunning(c.Request.Context(), container.Name)
        if err != nil {
                logger.WithComponent("runtime_controller").Warnf("failed to check if container %s is running: %v", container.Name, err)
                // Assume not running and try to start
                running = false
        }

        if !running {
                rc.startContainerInBackground(container.Name)
        }

        // Serve the waiting page
        rc.serveWaitingPage(c, container.Name, container.URL)
}

// handleGroupWaitingPage handles the waiting page for a group of containers.
func (rc *RuntimeController) handleGroupWaitingPage(c *gin.Context, doc repository.DataDocument, group *repository.Group) {
        // Check if group is active
        if group.Active == nil || !*group.Active {
                c.JSON(http.StatusForbidden, gin.H{"error": fmt.Sprintf("group '%s' is not active", group.Name)})
                return
        }

        // Find the first container in the group to get the redirect URL
        if len(group.Container) == 0 {
                c.JSON(http.StatusInternalServerError, gin.H{"error": fmt.Sprintf("group '%s' has no containers", group.Name)})
                return
        }

        var firstContainer *repository.Container
        for _, containerName := range group.Container {
                container, found := rc.findContainer(doc, containerName)
                if found {
                        firstContainer = container
                        break
                }
        }

        if firstContainer == nil {
                c.JSON(http.StatusInternalServerError, gin.H{"error": fmt.Sprintf("no valid containers found in group '%s'", group.Name)})
                return
        }

        // Start all containers in the group that are not running (in background)
        for _, containerName := range group.Container {
                container, found := rc.findContainer(doc, containerName)
                if !found {
                        logger.WithComponent("runtime_controller").Warnf("container %s in group %s not found", containerName, group.Name)
                        continue
                }

                // Check if container is active before starting
                if container.Active == nil || !*container.Active {
                        logger.WithComponent("runtime_controller").Debugf("container %s in group %s is not active, skipping", containerName, group.Name)
                        continue
                }

                running, err := rc.runtime.IsRunning(c.Request.Context(), containerName)
                if err != nil {
                        logger.WithComponent("runtime_controller").Warnf("failed to check if container %s is running: %v", containerName, err)
                        running = false
                }

                if !running {
                        rc.startContainerInBackground(containerName)
                }
        }

        // Serve the waiting page with the group name and first container's URL
        rc.serveWaitingPage(c, group.Name, firstContainer.URL)
}

// startContainerInBackground starts a container in a dedicated goroutine.
func (rc *RuntimeController) startContainerInBackground(containerName string) {
        go func(name string) {
                logger.WithComponent("runtime_controller").Infof("starting container %s in background", name)
                if err := rc.runtime.Start(rc.baseCtx, name); err != nil {
                        logger.WithComponent("runtime_controller").Errorf("failed to start container %s in background: %v", name, err)
                } else {
                        logger.WithComponent("runtime_controller").Infof("container %s started successfully", name)
                }
        }(containerName)
}

// serveWaitingPage renders the waiting HTML template with placeholders replaced.
func (rc *RuntimeController) serveWaitingPage(c *gin.Context, containerName, redirectURL string) {
        html := rc.waitingTemplate
        html = strings.ReplaceAll(html, "{{CONTAINER_NAME}}", containerName)
        html = strings.ReplaceAll(html, "{{REDIRECT_URL}}", redirectURL)

        c.Header("Content-Type", "text/html; charset=utf-8")
        c.String(http.StatusOK, html)
}

// ContainerStatsResponse represents the stats for a single container.
type ContainerStatsResponse struct {
        Name       string  `json:"name"`
        Error      string  `json:"error,omitempty"`
        CPUPercent float64 `json:"cpu_percent"`
        MemoryMB   float64 `json:"memory_mb"`
}

// AllStats returns CPU and memory statistics for all containers defined in the store.
// Stats are fetched in parallel to avoid sequential timeout accumulation.
func (rc *RuntimeController) AllStats(c *gin.Context) {
        doc, err := rc.containerStore.Snapshot()
        if err != nil {
                logger.WithComponent("runtime_controller").Errorf("failed to read container list: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to read container list"})
                return
        }

        // Fetch stats for all containers in parallel
        type statsResult struct {
                resp  ContainerStatsResponse
                index int
        }

        resultChan := make(chan statsResult, len(doc.Containers))
        ctx := c.Request.Context()

        // Log context deadline for debugging
        if deadline, ok := ctx.Deadline(); ok {
                logger.WithComponent("runtime_controller").Debugf("AllStats context deadline: %v (in %v)", deadline, time.Until(deadline))
        } else {
                logger.WithComponent("runtime_controller").Debugf("AllStats context has no deadline")
        }

        for i, container := range doc.Containers {
                go func(idx int, name string) {
                        stats, err := rc.runtime.Stats(ctx, name)
                        if err != nil {
                                logger.WithComponent("runtime_controller").Warnf("failed to get stats for container %s: %v", name, err)
                                resultChan <- statsResult{
                                        index: idx,
                                        resp: ContainerStatsResponse{
                                                Name:  name,
                                                Error: err.Error(),
                                        },
                                }
                                return
                        }
                        resultChan <- statsResult{
                                index: idx,
                                resp: ContainerStatsResponse{
                                        Name:       name,
                                        CPUPercent: stats.CPUPercent,
                                        MemoryMB:   stats.MemoryMB,
                                },
                        }
                }(i, container.Name)
        }

        // Collect all results
        results := make([]ContainerStatsResponse, len(doc.Containers))
        for range doc.Containers {
                res := <-resultChan
                results[res.index] = res.resp
        }

        c.JSON(http.StatusOK, results)
}

// SystemStats returns system-wide resource usage (CPU, RAM, disk).
func (rc *RuntimeController) SystemStats(c *gin.Context) {
        stats, err := rc.systemStatsCollector.GetStats(c.Request.Context())
        if err != nil {
                // Client disconnected or request timed out — not a server error.
                if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {
                        logger.WithComponent("runtime_controller").Debugf("system stats request aborted: %v", err)
                        return
                }
                logger.WithComponent("runtime_controller").Errorf("failed to get system stats: %v", err)
                c.JSON(http.StatusInternalServerError, gin.H{"error": fmt.Sprintf("failed to get system stats: %v", err)})
                return
        }
        c.JSON(http.StatusOK, stats)
}

// ScheduleCrudValidator implements CrudValidator for schedules.
type ScheduleCrudValidator struct {
        validator *validator.Validate
        store     cache.ScheduleStore
}

// Validate checks the schedule payload: struct-level rules (required fields,
// target type, HH:MM times), the timer contract (at least one of
// StartTime/StopTime must be provided) and the existence of the target. The
// last check matters because the data loader prunes schedules whose target no
// longer exists: accepting an orphan here would persist a schedule that
// silently disappears on the next reload.
func (v *ScheduleCrudValidator) Validate(item repository.Schedule) error {
        if err := v.validator.Struct(item); err != nil {
                return err
        }
        for i, timer := range item.Timers {
                if timer.StartTime == "" && timer.StopTime == "" {
                        return fmt.Errorf("timer %d must define startTime, stopTime or both", i)
                }
        }
        return v.validateTargetExists(item)
}

// validateTargetExists rejects schedules whose target container or group is not
// part of the current data document.
func (v *ScheduleCrudValidator) validateTargetExists(item repository.Schedule) error {
        if v.store == nil {
                return nil
        }

        doc, err := v.store.Snapshot()
        if err != nil {
                return fmt.Errorf("cannot verify schedule target: %w", err)
        }

        switch item.TargetType {
        case "container":
                for _, c := range doc.Containers {
                        if c.Name == item.Target {
                                return nil
                        }
                }
                return fmt.Errorf("target container %q not found", item.Target)
        case "group":
                for _, g := range doc.Groups {
                        if g.Name == item.Target {
                                return nil
                        }
                }
                return fmt.Errorf("target group %q not found", item.Target)
        }

        return nil
}

// CORSMiddleware returns a Gin middleware that handles CORS preflight and headers.
// allowedOrigins is a comma-separated list of allowed origins, or "*" for all.
func CORSMiddleware(allowedOrigins string) gin.HandlerFunc {
        // Pre-parse allowed origins for efficiency
        var allowAll bool
        var originSet map[string]struct{}

        if allowedOrigins == "*" {
                allowAll = true
        } else {
                originSet = make(map[string]struct{})
                for _, o := range strings.Split(allowedOrigins, ",") {
                        o = strings.TrimSpace(o)
                        if o == "" {
                                continue
                        }
                        originSet[o] = struct{}{}
                }
        }

        return func(c *gin.Context) {
                origin := c.Request.Header.Get("Origin")

                // Determine which origin to return
                var allowOrigin string
                if allowAll {
                        allowOrigin = "*"
                } else if origin != "" {
                        if _, ok := originSet[origin]; ok {
                                allowOrigin = origin
                        }
                }

                // If origin not allowed, skip CORS headers (browser will block)
                if allowOrigin == "" {
                        c.Next()
                        return
                }

                c.Header("Access-Control-Allow-Origin", allowOrigin)
                c.Header("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")
                c.Header("Access-Control-Max-Age", "86400")

                // For preflight, echo requested headers if present; otherwise use defaults
                reqHeaders := c.Request.Header.Get("Access-Control-Request-Headers")
                if strings.TrimSpace(reqHeaders) != "" {
                        c.Header("Access-Control-Allow-Headers", reqHeaders)
                } else {
                        c.Header("Access-Control-Allow-Headers", "Origin, Content-Type, Accept, Authorization")
                }

                // Only set credentials if not using wildcard origin
                if allowOrigin != "*" {
                        c.Header("Access-Control-Allow-Credentials", "true")
                }

                // Handle preflight requests
                if c.Request.Method == http.MethodOptions {
                        c.AbortWithStatus(http.StatusNoContent)
                        return
                }

                c.Next()
        }
}

// HoneybadgerMiddleware sends error/warning notifications to Honeybadger.
// On panic, it notifies Honeybadger and re-panics to allow gin.Recovery to handle the response.
func HoneybadgerMiddleware(logger *logrus.Logger) gin.HandlerFunc {
        apiKey := os.Getenv("HONEYBADGER_API_KEY")
        if apiKey == "" {
                logger.Info("Honeybadger is not active. To enable error reporting, set the HONEYBADGER_API_KEY environment variable.")
                return func(c *gin.Context) {
                        c.Next()
                }
        }

        honeybadger.Configure(honeybadger.Configuration{
                APIKey: apiKey,
                Env:    os.Getenv("GO_ENV"),
        })

        logger.Info("Honeybadger error reporting is enabled.")

        return func(c *gin.Context) {
                defer func() {
                        if rec := recover(); rec != nil {
                                // Notify Honeybadger with stacktrace, then re-panic.
                                // Honeybadger delivery failures are logged but do not block panic re-raise.
                                // Notify returns (noticeID string, error); we only care about the error here.
                                if _, err := honeybadger.Notify(fmt.Sprintf("Panic: %s %s", c.Request.Method, c.Request.URL.Path),
                                        c.Request, honeybadger.Context{"stack": string(debug.Stack())}, honeybadger.Tags{"panic", "http"}); err != nil {
                                        logger.Debugf("Honeybadger notify failed (non-critical): %v", err)
                                }
                                logger.Error("Recovered from panic, notified Honeybadger: ", rec)
                                panic(rec) // propagate panic to let gin.Recovery handle it
                        }
                }()

                c.Next()

                status := c.Writer.Status()
                if status >= 400 && status != 404 {
                        if status >= 500 {
                                // Send stacktrace for 5xx errors
                                if _, err := honeybadger.Notify(fmt.Sprintf("Error: HTTP %d: %s %s", status, c.Request.Method, c.Request.URL.Path), c.Request, honeybadger.Tags{"5XX", "http"}); err != nil {
                                        logger.Debugf("Honeybadger notify failed (non-critical): %v", err)
                                }
                        } else {
                                // For warnings (4xx), send as notice without stacktrace
                                if _, err := honeybadger.Notify(fmt.Sprintf("Warning: HTTP %d: %s %s", status, c.Request.Method, c.Request.URL.Path), honeybadger.Tags{"4XX", "http"}); err != nil {
                                        logger.Debugf("Honeybadger notify failed (non-critical): %v", err)
                                }
                        }
                        logger.Warnf("Honeybadger reported HTTP %d for %s %s", status, c.Request.Method, c.Request.URL.Path)
                }
        }
}

// RequestTimeout sets a per-request context deadline.
// It does NOT forcibly kill the handler; downstream code must honor ctx.Done().
func RequestTimeout(d time.Duration) gin.HandlerFunc {
        if d <= 0 {
                return func(c *gin.Context) { c.Next() }
        }

        return func(c *gin.Context) {
                ctx, cancel := context.WithTimeout(c.Request.Context(), d)
                defer cancel()

                c.Request = c.Request.WithContext(ctx)
                c.Next()

                // If the context timed out and nothing was written, return 504.
                // (If something was already written, we can't change the response safely.)
                if ctx.Err() == context.DeadlineExceeded && !c.Writer.Written() {
                        c.AbortWithStatusJSON(http.StatusGatewayTimeout, gin.H{
                                "error": "request timeout",
                        })
                        return
                }
        }
}

// isDir reports whether path exists and is a directory.
func isDir(path string) bool {
        info, err := os.Stat(path)
        return err == nil && info.IsDir()
}

// firstExistingDir returns the first candidate that is an existing directory,
// or "" when none is. Kept separate from resolveUIDir so the preference order
// can be tested without relying on the layout around the test binary.
func firstExistingDir(candidates ...string) string {
        for _, candidate := range candidates {
                if isDir(candidate) {
                        return candidate
                }
        }
        return ""
}

// resolveUIDir returns the directory holding the UI assets. It prefers ./ui
// (the layout of the container image and of a checkout) and falls back to a
// "ui" directory next to the executable, so a binary started from another
// working directory still serves the pages and real cache-busting hashes
// instead of silently falling back to the "dev" placeholder.
//
// Resolved on every call rather than cached: the value depends on the working
// directory, which the process (or a test) can change after startup, and a
// stale cache would pin every later read to the wrong directory.
func resolveUIDir() string {
        if dir := firstExistingDir("./ui"); dir != "" {
                return dir
        }
        if exe, err := os.Executable(); err == nil {
                if dir := firstExistingDir(filepath.Join(filepath.Dir(exe), "ui")); dir != "" {
                        return dir
                }
        }
        // Debug level: the fallback mirrors the historical behaviour, and the
        // unreadable file paths are already reported by fileHash.
        logger.WithComponent("ui_router").Debugf("UI directory not found at ./ui or next to the executable, using ./ui")
        return "./ui"
}

// uiPath joins name to the resolved UI directory.
func uiPath(name string) string {
        return filepath.Join(resolveUIDir(), name)
}

// assetVersion returns the content hash used as the ?v= cache-busting value
// for a bundle. Computed per request: the files are small, and a cached value
// would go stale if the UI directory changes under a running process.
func assetVersion(name string) string {
        return fileHash(uiPath(name))
}

// fileHash returns the first 8 hex characters of the SHA-256 hash of the
// file at path, or "dev" if the file cannot be read.
func fileHash(path string) string {
        data, err := os.ReadFile(path)
        if err != nil {
                logger.WithComponent("ui_router").Warnf("cannot read %s for cache-busting hash: %v", path, err)
                return "dev"
        }
        h := sha256.Sum256(data)
        return fmt.Sprintf("%x", h[:4])
}

// cacheBustTag appends the content hash of an asset to the attribute that
// references it, turning href="/ui/assets/icon.svg" into
// href="/ui/assets/icon.svg?v=<hash>". The assets are served with a one-year
// cache lifetime, so the hash is what makes a deployment pick up a new file
// even when the service worker cache is still warm. The files are small, so
// hashing them on every page request keeps a deploy live without a restart.
func cacheBustTag(html, attr, asset string) string {
        target := attr + `="/ui/` + asset + `"`
        busted := attr + `="/ui/` + asset + "?v=" + assetVersion(asset) + `"`
        return strings.Replace(html, target, busted, 1)
}

// serveGhostboardPage serves the ghostboard page with the script and icon tags
// rewritten to carry the content hash of their assets. The configured board
// size is injected as data-max-bytes, so the file picker can reject an
// oversized file before reading it into memory.
func serveGhostboardPage(maxTextBytes int) gin.HandlerFunc {
        return func(c *gin.Context) {
                c.Header("Cache-Control", "no-cache, must-revalidate")
                html, err := os.ReadFile(uiPath("ghostboard.html"))
                if err != nil {
                        logger.WithComponent("ui_router").Errorf("cannot read ghostboard page: %v", err)
                        c.Status(http.StatusInternalServerError)
                        return
                }

                rewritten := string(html)
                rewritten = cacheBustTag(rewritten, "src", "assets/ghostboard.js")
                rewritten = cacheBustTag(rewritten, "href", "assets/ghostboard-icon.svg")
                rewritten = cacheBustTag(rewritten, "href", "assets/ghostboard-icon-32.png")
                rewritten = cacheBustTag(rewritten, "href", "assets/ghostboard-icon-180.png")
                rewritten = strings.Replace(rewritten, `data-max-bytes=""`,
                        fmt.Sprintf(`data-max-bytes="%d"`, maxTextBytes), 1)

                c.Data(http.StatusOK, "text/html; charset=utf-8", []byte(rewritten))
        }
}

// NewUIRouter sets up routes to serve the UI static files under /ui.
// It serves index.html for the root and any sub-paths (SPA routing).
func NewUIRouter(r *gin.Engine, ghostboardMaxTextBytes int) {
        // Serve static assets (JS, CSS, images) with long-term caching.
        // Manifests are excluded from long-term caching so icon/scope changes
        // are picked up immediately by browsers.
        assetsGroup := r.Group("/ui/assets")
        assetsGroup.Use(func(c *gin.Context) {
                if strings.Contains(c.Request.URL.Path, "manifest") {
                        c.Header("Content-Type", "application/manifest+json")
                        c.Header("Cache-Control", "no-cache, must-revalidate")
                } else {
                        c.Header("Cache-Control", "public, max-age=31536000")
                }
                c.Next()
        })
        assetsGroup.Static("/", uiPath("assets"))

        // Serve favicon
        r.GET("/favicon.ico", func(c *gin.Context) {
                c.Header("Content-Type", "image/x-icon")
                c.Header("Cache-Control", "public, max-age=86400")
                c.File(uiPath("assets/vite.ico"))
        })

        // Redirect root to /ui
        r.GET("/", func(c *gin.Context) {
                c.Redirect(http.StatusMovedPermanently, "/ui")
        })

        // Serve service worker from /ui/ so its default scope covers all /ui/* pages.
        // Service-Worker-Allowed header allows explicit scope override if needed.
        // Browsers cache SW for at most 24h per spec; we use 1h for rapid iteration.
        // MUST be registered before /ui route to avoid Gin routing conflicts.
        r.GET("/ui/service-worker.js", func(c *gin.Context) {
                c.Header("Content-Type", "application/javascript; charset=utf-8")
                c.Header("Service-Worker-Allowed", "/")
                c.Header("Cache-Control", "public, max-age=3600")
                c.File(uiPath("service-worker.js"))
        })

        // Serve the shared service-worker registration script. It sits next to
        // service-worker.js rather than under /ui/assets so it gets a short cache
        // lifetime without colliding with the static asset wildcard.
        r.GET("/ui/sw-register.js", func(c *gin.Context) {
                c.Header("Content-Type", "application/javascript; charset=utf-8")
                c.Header("Cache-Control", "public, max-age=3600")
                c.File(uiPath("sw-register.js"))
        })

        // Serve index.html for the /ui root (no caching for HTML).
        // The /ui/assets/app.js <script> tag is rewritten on the fly to include a
        // ?v=<hash> cache-busting query parameter so that deployments are picked
        // up immediately even when the service worker hasn't refreshed yet.
        r.GET("/ui", func(c *gin.Context) {
                c.Header("Cache-Control", "no-cache, must-revalidate")

                html, err := os.ReadFile(uiPath("index.html"))
                if err != nil {
                        c.File(uiPath("index.html"))
                        return
                }

                // Rewrite src="/ui/assets/app.js" → src="/ui/assets/app.js?v=<hash>"
                rewritten := strings.Replace(
                        string(html),
                        `src="/ui/assets/app.js"`,
                        fmt.Sprintf(`src="/ui/assets/app.js?v=%s"`, assetVersion("assets/app.js")),
                        1,
                )

                c.Data(http.StatusOK, "text/html; charset=utf-8", []byte(rewritten))
        })

        // Serve home.html for the dedicated homepage route (no caching for HTML).
        // The /ui/home.js <script> tag is rewritten on the fly to include a
        // ?v=<hash> cache-busting query parameter so that deployments are picked
        // up immediately even when the service worker hasn't refreshed yet.
        r.GET("/ui/home", func(c *gin.Context) {
                c.Header("Cache-Control", "no-cache, must-revalidate")

                html, err := os.ReadFile(uiPath("home.html"))
                if err != nil {
                        c.File(uiPath("home.html"))
                        return
                }

                // Rewrite src="/ui/home.js" → src="/ui/home.js?v=<hash>"
                rewritten := strings.Replace(
                        string(html),
                        `src="/ui/home.js"`,
                        fmt.Sprintf(`src="/ui/home.js?v=%s"`, assetVersion("home.js")),
                        1,
                )

                c.Data(http.StatusOK, "text/html; charset=utf-8", []byte(rewritten))
        })
        r.GET("/ui/home.js", func(c *gin.Context) {
                c.Header("Cache-Control", "public, max-age=3600")
                c.File(uiPath("home.js"))
        })

        // Serve the shared text board. The board name is part of the path
        // (/ui/ghostboard/lavoro) and is resolved client-side, so every sub-path
        // returns the same page.
        ghostboardPage := serveGhostboardPage(ghostboardMaxTextBytes)
        r.GET("/ui/ghostboard", ghostboardPage)
        r.GET("/ui/ghostboard/*board", ghostboardPage)

        // Serve index.html for any sub-path under /ui (SPA client-side routing)
        r.NoRoute(func(c *gin.Context) {
                p := c.Request.URL.Path

                // Only handle /ui/* paths, return 404 for others
                if p == "/ui" || strings.HasPrefix(p, "/ui/") {
                        c.File(uiPath("index.html"))
                        return
                }
                c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
        })
}

        Cache      cache.AppStore
        Runtime    runtime.ContainerRuntime
        Ghostboard *ghostboard.Store

        BaseCtx     context.Context
        Cancel      context.CancelFunc
        persistDone <-chan struct{} // signal for completion of persistence scheduler
}

func New(cfg *config.Config, repo repository.Repository, store cache.AppStore, rt runtime.ContainerRuntime) (*App, error) {
        logger.WithComponent("app").Debugf("initializing app container")

        if cfg == nil {
                logger.WithComponent("app").Errorf("config is nil")
                return nil, errors.New("config is nil")
        }
        if repo == nil {
                logger.WithComponent("app").Errorf("repo is nil")
                return nil, errors.New("repo is nil")
        }
        if store == nil {
                logger.WithComponent("app").Errorf("cache store is nil")
                return nil, errors.New("cache store is nil")
        }
        if rt == nil {
                logger.WithComponent("app").Errorf("runtime is nil")
                return nil, errors.New("runtime is nil")
        }

        logger.WithComponent("app").Debugf("all dependencies validated")

        ghostboardCfg := cfg.Ghostboard.WithDefaults()
        ghostboardStore, err := ghostboard.New(ghostboard.Config{
                FilePath:        ghostboardCfg.FilePath,
                ClearDelay:      ghostboardCfg.ClearDelay,
                MaxTextBytes:    ghostboardCfg.MaxTextBytes,
                PersistInterval: ghostboardCfg.PersistInterval,
        })
        if err != nil {
                logger.WithComponent("app").Errorf("cannot initialize ghostboard store: %v", err)
                return nil, err
        }

        ctx, cancel := context.WithCancel(context.Background())
        return &App{
                Config:     cfg,
                Repo:       repo,
                Cache:      store,
                Runtime:    rt,
                Ghostboard: ghostboardStore,
                BaseCtx:    ctx,
                Cancel:     cancel,
        }, nil
}

func (a *App) Shutdown() {
        logger.WithComponent("app").Debugf("shutting down app container")

        if a == nil || a.Cancel == nil {
                logger.WithComponent("app").Debugf("app or cancel is nil, skipping shutdown")
                return
        }
        a.Cancel()

        // Flush ghostboard changes synchronously: the persistence goroutine may be
        // cancelled before its final write completes.
        if a.Ghostboard != nil {
                if err := a.Ghostboard.Flush(); err != nil {
                        logger.WithComponent("app").Errorf("ghostboard flush on shutdown failed: %v", err)
                }
        }

        // Attende il completamento del persistence scheduler
        if a.persistDone != nil {
                logger.WithComponent("app").Debugf("waiting for persistence scheduler to complete")
                <-a.persistDone
        }

        logger.WithComponent("app").Debugf("app shutdown completed")
}

func (a *App) StartWatchers() {
        logger.WithComponent("app").Debugf("starting watchers")

        if err := a.Repo.StartWatcher(a.BaseCtx, a.Cache); err != nil {
                logger.WithComponent("app").Fatalf("cannot start config file watcher: %v", err)
        }

        logger.WithComponent("app").Debugf("file watcher started")

        // Start scheduled persistence goroutine
        a.persistDone = cache.StartPersistenceScheduler(a.BaseCtx, a.Cache, a.Repo, a.Config.Data.PersistInterval)
        logger.WithComponent("app").Debugf("persistence scheduler started")

        if a.Ghostboard != nil {
                a.Ghostboard.StartPersistence(a.BaseCtx)
                logger.WithComponent("app").Debugf("ghostboard persistence started")
        }

        if a.Config.Data.SchedulingEnabled {
                loc := time.Local
                if a.Config.Misc.SchedulingTZ != "" && a.Config.Misc.SchedulingTZ != "Local" {
                        l, err := time.LoadLocation(a.Config.Misc.SchedulingTZ)
                        if err != nil {
                                logger.WithComponent("app").Fatalf("invalid scheduling timezone: %v", err)
                        }
                        loc = l
                }

                logger.WithComponent("app").Debugf("starting polling scheduler with timezone: %v", loc)
                s := scheduler.NewPollingScheduler(a.Cache, a.Runtime, a.Config.Data.SchedulingPoll, loc)
                s.Start(a.BaseCtx)
        }

        logger.WithComponent("app").Debugf("all watchers started successfully")
}

// Package boardclient is the HTTP client behind the gb-cli CLI. It talks
// to the plain-text board API (/api/boards/:board) served by GoSpin, so the CLI
// needs no access to the server's files and works against a remote instance.
package boardclient

import (
        "bufio"
        "bytes"
        "context"
        "encoding/json"
        "errors"
        "fmt"

const (
        // boardsPath is the plain-text board API exposed by the GoSpin server.
        boardsPath = "api/boards"

        // statePath is the JSON board API, which also carries the revision.
        statePath = "ghostboard"

        // boardEndpoint, stateEndpoint and eventEndpoint name the routes the CLI
        // depends on in the messages about a missing route.
        boardEndpoint = "/api/boards/:board"
        stateEndpoint = "/ghostboard/:board"
        eventEndpoint = "/ghostboard/:board/events"

        // errorBodyLimit bounds how much of an unexpected response body is quoted
        // in an error message.
        errorBodyLimit = 512

        // DefaultRetryDelay is the pause before the first retry; it doubles on every
        // further attempt, up to maxRetryDelay.
        DefaultRetryDelay = 200 * time.Millisecond

        // maxRetryDelay caps the exponential backoff between two attempts.
        maxRetryDelay = 2 * time.Second
)

// ErrBoardName is matched by BoardNameError, so a caller can tell an invalid
// board name from a transport or server failure.
var ErrBoardName = errors.New("invalid board name")

// ResponseError describes a non-success answer from the server.
type ResponseError struct {
        Board   string `json:"-"`
        Action  string `json:"-"`
        Message string `json:"error"`
        Status  int    `json:"-"`
}

// Error renders the failure with the action, the board and the server message.
func (e *ResponseError) Error() string {
        if e.Message == "" {
                return fmt.Sprintf("%s board %q: server answered %d", e.Action, e.Board, e.Status)
        }
        return fmt.Sprintf("%s board %q: server answered %d: %s", e.Action, e.Board, e.Status, e.Message)
}

// EndpointError reports a 404 on a board route: the URL does not point at a
// GoSpin board API. Only GoSpin serves those paths, so a missing route means
// the request went somewhere else (wrong port, host or path prefix, or a GoSpin
// release without the route) far more often than it means a missing board: the
// board API answers an unknown board with an empty 200.
type EndpointError struct {
        URL      string
        Endpoint string
}

// Error describes the missing route and points at the URL that was requested.
func (e *EndpointError) Error() string {
        return fmt.Sprintf("404 Not Found on %s: the server does not expose %s; check --url and the configuration file, and that the server runs a GoSpin release that serves the board API", e.URL, e.Endpoint)
}

// BoardNameError reports a board name that cannot address a board.
type BoardNameError struct {
        Name string
}

// Error explains which names are accepted.
func (e *BoardNameError) Error() string {
        return fmt.Sprintf("invalid board name %q: use 1-64 letters, digits, dots, dashes or underscores", e.Name)
}

// Unwrap lets callers match the failure with errors.Is(ErrBoardName).
func (e *BoardNameError) Unwrap() error {
        return ErrBoardName
}

// Board is the state carried by the JSON API and the event stream: the whole
// text plus the revision and the last update time.
//
// UpdatedAt is a pointer so a board that was never written omits the field
// instead of serialising a timestamp in year 1.
type Board struct {
        UpdatedAt *time.Time `json:"updated_at,omitempty"`
        Name      string     `json:"name"`
        Text      string     `json:"text"`
        Revision  uint64     `json:"revision"`
}

// normalizeBoard drops the zero update time the server sends for a board that
// was never written, so a consumer sees "no update yet" instead of 0001-01-01.
func normalizeBoard(board Board) Board {
        if board.UpdatedAt != nil && board.UpdatedAt.IsZero() {
                board.UpdatedAt = nil
        }
        return board
}

// Options tunes a Client. Only Timeout is required.
type Options struct {
        // Headers are sent with every request, for a reverse proxy that requires
        // authentication.
        Headers http.Header
        // Timeout bounds a single request.
        Timeout time.Duration
        // Retries is the number of extra attempts after a transient failure.
        Retries int
        // RetryDelay is the pause before the first retry; it doubles on every
        // further attempt. Zero means DefaultRetryDelay.
        RetryDelay time.Duration
}

// Client reads and writes boards on a running GoSpin server.
type Client struct {
        http       *http.Client
        streamHTTP *http.Client
        headers    http.Header
        baseURL    string
        retryDelay time.Duration
        retries    int
}

// New creates a client for the given server URL with a plain timeout. The URL
// must be http or https and may carry a path prefix when GoSpin sits behind a
// reverse proxy.
func New(baseURL string, timeout time.Duration) (*Client, error) {
        return NewWithOptions(baseURL, Options{Timeout: timeout})
}

// NewWithOptions creates a client from a full option set.
func NewWithOptions(baseURL string, opts Options) (*Client, error) {
        normalized, err := NormalizeURL(baseURL)
        if err != nil {
                return nil, err
        }
        if opts.Timeout <= 0 {
                return nil, errors.New("request timeout must be positive")
        }
        if opts.Retries < 0 {
                return nil, errors.New("retries must not be negative")
        }
        if opts.RetryDelay < 0 {
                return nil, errors.New("retry delay must not be negative")
        }

        retryDelay := opts.RetryDelay
        if retryDelay == 0 {
                retryDelay = DefaultRetryDelay
        }

        return &Client{
                baseURL:    normalized,
                headers:    cloneHeader(opts.Headers),
                retries:    opts.Retries,
                retryDelay: retryDelay,
                http:       &http.Client{Timeout: opts.Timeout},
                // The event stream stays open for as long as the user watches it, so it
                // must not carry the per-request timeout.
                streamHTTP: &http.Client{Transport: http.DefaultTransport},
        }, nil
}

// NormalizeURL validates a server base URL and strips the trailing slashes so
// board URLs are built without a double slash.
func NormalizeURL(raw string) (string, error) {
        trimmed := strings.TrimSpace(raw)
        if trimmed == "" {
                return "", errors.New("server URL is empty")
        }

        parsed, err := url.Parse(trimmed)
        if err != nil {
                return "", fmt.Errorf("invalid server URL %q: %w", raw, err)
        }
        if parsed.Scheme != "http" && parsed.Scheme != "https" {
                return "", fmt.Errorf("invalid server URL %q: scheme must be http or https", raw)
        }
        if parsed.Host == "" {
                return "", fmt.Errorf("invalid server URL %q: missing host", raw)
        }
        if parsed.RawQuery != "" || parsed.Fragment != "" {
                return "", fmt.Errorf("invalid server URL %q: must not carry a query or a fragment", raw)
        }
        // /ui/* serves the web pages of GoSpin; the board API lives at the root of
        // the server (or under the prefix a reverse proxy mounts it on), so a page
        // URL would be used as an API prefix and only ever answer 404.
        if isUIPath(parsed.Path) {
                return "", fmt.Errorf("invalid server URL %q: %s is a GoSpin page, not an API base; use the server root, e.g. %s://%s", raw, parsed.Path, parsed.Scheme, parsed.Host)
        }

        return strings.TrimRight(parsed.String(), "/"), nil
}

// isUIPath reports whether a URL path points at the web UI instead of the API.
func isUIPath(path string) bool {
        return path == "/ui" || strings.HasPrefix(path, "/ui/")
}

// Get returns the board text. A board that was never written comes back as an
// empty string, exactly like a board that was cleared.
func (c *Client) Get(ctx context.Context, board string) (string, error) {
        payload, err := c.fetch(ctx, c.boardURL(board), board, boardEndpoint, "reading")
        if err != nil {
                return "", err
        }
        return string(payload), nil
}

// GetBoard returns the board state from the JSON API: the text plus the
// revision and the last update time.
func (c *Client) GetBoard(ctx context.Context, board string) (Board, error) {
        payload, err := c.fetch(ctx, c.stateURL(board), board, stateEndpoint, "reading")
        if err != nil {
                return Board{}, err
        }

        var state Board
        if err := json.Unmarshal(payload, &state); err != nil {
                return Board{}, fmt.Errorf("decoding the state of board %q: %w", board, err)
        }
        return normalizeBoard(state), nil
}

// Set replaces the board text with text, byte for byte. An empty text clears
// the board for every connected client.
func (c *Client) Set(ctx context.Context, board, text string) error {
        _, err := c.send(ctx, http.MethodPost, c.boardURL(board), board, boardEndpoint, []byte(text), "writing")
        return err
}

// Append adds text to the end of the board and returns the resulting text.
//
// The text is separated from the board by a newline when the board does not
// already end with one (see AppendText), so the addition never glues onto the
// last line. Append is a read followed by a write, so a concurrent writer in
// between can be overwritten; Set is the last-writer-wins alternative.
func (c *Client) Append(ctx context.Context, board, text string) (string, error) {
        current, err := c.Get(ctx, board)
        if err != nil {
                return "", err
        }

        updated := AppendText(current, text)
        if err := c.Set(ctx, board, updated); err != nil {
                return "", err
        }
        return updated, nil
}

// AppendText returns the text Append would store for the given board content.
// A board is a text, not a list of lines, so a newline is inserted between the
// two when the board does not already end with one. The separator is skipped
// for an empty board, for an empty addition and when the addition already
// starts with a line break, so repeated appends never produce blank lines.
func AppendText(current, text string) string {
        if text == "" || current == "" || strings.HasSuffix(current, "\n") || strings.HasPrefix(text, "\n") {
                return current + text
        }
        return current + "\n" + text
}

// Follow streams a board until ctx is cancelled, calling handle with the
// current snapshot and then with every change. Each callback carries the whole
// board text, so a caller never has to merge partial payloads. It returns nil
// when the stream ends because ctx was cancelled.
func (c *Client) Follow(ctx context.Context, board string, handle func(Board)) error {
        if err := ghostboard.ValidateBoardName(board); err != nil {
                return &BoardNameError{Name: board}
        }

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.eventURL(board), nil)
        if err != nil {
                return fmt.Errorf("building the event request for board %q: %w", board, err)
        }
        req.Header.Set("Accept", "text/event-stream")
        req.Header.Set("Cache-Control", "no-cache")
        c.applyHeaders(req)

        resp, err := c.streamHTTP.Do(req)
        if err != nil {
                return fmt.Errorf("contacting the GoSpin server at %s: %w", c.baseURL, err)
        }
        defer closeBody(resp, board)

        if resp.StatusCode == http.StatusNotFound {
                return &EndpointError{URL: c.eventURL(board), Endpoint: eventEndpoint}
        }
        if resp.StatusCode != http.StatusOK {
                return responseError(resp, board, "following")
        }

        // The stream is read line by line without a fixed cap: an event carries the
        // whole board, whose size the server already bounds with
        // ghostboard.max_text_bytes, so a scanner buffer could only disagree with
        // that limit and break on a large board.
        reader := bufio.NewReader(resp.Body)
        for {
                line, readErr := reader.ReadString('\n')
                if payload, ok := eventPayload(strings.TrimRight(line, "\r\n")); ok {
                        var state Board
                        if err := json.Unmarshal([]byte(payload), &state); err != nil {
                                return fmt.Errorf("decoding an event of board %q: %w", board, err)
                        }
                        handle(normalizeBoard(state))
                }

                if readErr == nil {
                        continue
                }
                if errors.Is(readErr, io.EOF) {
                        return nil
                }
                if ctx.Err() != nil {
                        return nil
                }
                return fmt.Errorf("reading the event stream of board %q: %w", board, readErr)
        }
}

// fetch performs a GET against a board route, retrying a transient failure.
func (c *Client) fetch(ctx context.Context, requestURL, board, endpoint, action string) ([]byte, error) {
        return c.send(ctx, http.MethodGet, requestURL, board, endpoint, nil, action)
}

// send performs one request, retrying transient failures up to the configured
// number of attempts. The body is rebuilt on every attempt so a retry sends
// exactly what the first attempt did.
func (c *Client) send(ctx context.Context, method, requestURL, board, endpoint string, body []byte, action string) ([]byte, error) {
        if err := ghostboard.ValidateBoardName(board); err != nil {
                return nil, &BoardNameError{Name: board}
        }

        for attempt := 0; ; attempt++ {
                payload, retriable, err := c.attempt(ctx, method, requestURL, board, endpoint, body, action)
                if err == nil {
                        return payload, nil
                }
                if !retriable || attempt >= c.retries {
                        return nil, err
                }
                if waitErr := waitBeforeRetry(ctx, c.delay(attempt)); waitErr != nil {
                        return nil, err
                }
        }
}

// attempt is a single request. It reports whether the failure is worth another
// attempt: a transport error or one of the statuses of isRetriableStatus.
func (c *Client) attempt(ctx context.Context, method, requestURL, board, endpoint string, body []byte, action string) ([]byte, bool, error) {
        var reader io.Reader
        if body != nil {
                reader = bytes.NewReader(body)
        }

        req, err := http.NewRequestWithContext(ctx, method, requestURL, reader)
        if err != nil {
                return nil, false, fmt.Errorf("building request for board %q: %w", board, err)
        }
        if body != nil {
                req.Header.Set("Content-Type", "text/plain; charset=utf-8")
        }
        c.applyHeaders(req)

        resp, err := c.http.Do(req)
        if err != nil {
                retriable := ctx.Err() == nil
                return nil, retriable, fmt.Errorf("contacting the GoSpin server at %s: %w", c.baseURL, err)
        }
        defer closeBody(resp, board)

        if resp.StatusCode == http.StatusNotFound {
                return nil, false, &EndpointError{URL: requestURL, Endpoint: endpoint}
        }
        if isRetriableStatus(resp.StatusCode) {
                return nil, true, responseError(resp, board, action)
        }
        if resp.StatusCode < http.StatusOK || resp.StatusCode > http.StatusMultipleChoices-1 {
                return nil, false, responseError(resp, board, action)
        }

        payload, err := io.ReadAll(resp.Body)
        if err != nil {
                return nil, false, fmt.Errorf("reading the text of board %q: %w", board, err)
        }
        return payload, false, nil
}

// isRetriableStatus reports whether a status is worth another attempt: the
// server is overloaded or a proxy in front of it failed to reach it.
func isRetriableStatus(status int) bool {
        switch status {
        case http.StatusTooManyRequests, http.StatusBadGateway, http.StatusServiceUnavailable, http.StatusGatewayTimeout:
                return true
        default:
                return false
        }
}

// delay returns the backoff before the next attempt.
func (c *Client) delay(attempt int) time.Duration {
        delay := c.retryDelay
        for i := 0; i < attempt; i++ {
                delay *= 2
                if delay >= maxRetryDelay {
                        return maxRetryDelay
                }
        }
        return delay
}

// waitBeforeRetry sleeps unless the context ends first.
func waitBeforeRetry(ctx context.Context, delay time.Duration) error {
        timer := time.NewTimer(delay)
        defer timer.Stop()

        select {
        case <-ctx.Done():
                return ctx.Err()
        case <-timer.C:
                return nil
        }
}

// applyHeaders copies the configured headers onto a request.
func (c *Client) applyHeaders(req *http.Request) {
        for name, values := range c.headers {
                for _, value := range values {
                        req.Header.Add(name, value)
                }
        }
}

// boardURL builds the plain-text API URL of a board, escaping the name so a dot
// or a dash never changes the meaning of the path.
func (c *Client) boardURL(board string) string {
        return c.baseURL + "/" + boardsPath + "/" + url.PathEscape(board)
}

// stateURL builds the JSON API URL of a board.
func (c *Client) stateURL(board string) string {
        return c.baseURL + "/" + statePath + "/" + url.PathEscape(board)
}

// eventURL builds the Server-Sent Events URL of a board.
func (c *Client) eventURL(board string) string {
        return c.baseURL + "/" + statePath + "/" + url.PathEscape(board) + "/events"
}

// eventPayload extracts the JSON carried by a Server-Sent Event data line. The
// other fields of the stream (event name, keep-alive comments) are skipped.
func eventPayload(line string) (string, bool) {
        const prefix = "data:"
        if !strings.HasPrefix(line, prefix) {
                return "", false
        }
        return strings.TrimSpace(strings.TrimPrefix(line, prefix)), true
}

// responseError turns a non-success answer into an error, preferring the JSON
// error envelope of the API. An HTML body (the page of a web server or of a
// reverse proxy) is not quoted: it says nothing about the board and only hides
// the status.
func responseError(resp *http.Response, board, action string) error {
        message := ""
        if !isHTML(resp.Header.Get("Content-Type")) {
                body, err := io.ReadAll(io.LimitReader(resp.Body, errorBodyLimit))
                if err == nil {
                        message = errorMessage(body)
                }
        }
        if message == "" {
                message = resp.Status
        }

        return &ResponseError{
                Status:  resp.StatusCode,
                Board:   board,
                Action:  action,
                Message: message,
        }
}

// errorMessage returns the JSON error of the API when there is one, the plain
// text body otherwise, and an empty string when neither carries a message.
func errorMessage(body []byte) string {
        var envelope struct {
                Error string `json:"error"`
        }
        if err := json.Unmarshal(body, &envelope); err == nil {
                return strings.TrimSpace(envelope.Error)
        }
        return strings.TrimSpace(string(body))
}

// isHTML reports whether a content type carries a web page, whose body is
// useless in an error message.
func isHTML(contentType string) bool {
        return strings.HasPrefix(strings.ToLower(contentType), "text/html")
}

// cloneHeader copies a header so later changes to the caller's map never reach
// the client.
func cloneHeader(header http.Header) http.Header {
        clone := make(http.Header, len(header))
        for name, values := range header {
                clone[name] = append([]string(nil), values...)
        }
        return clone
}

// closeBody releases a response body, reporting a failure at debug level since
// the request itself already succeeded.
func closeBody(resp *http.Response, board string) {
        if err := resp.Body.Close(); err != nil {
                logger.WithComponent("boardclient").Debugf("closing response body for board %q: %v", board, err)
        }
}

// Config is the resolved configuration of the CLI, before the command-line

        // not understand, such as a mistyped key. Reporting them is the whole

// LoadOptions tells Load where to look. Every field is optional: an empty
// ConfigPath searches the standard locations, an empty HomeDir or
// ExecutablePath skips that location.
type LoadOptions struct {
        // Getenv reads the environment; os.Getenv is used when nil.
        Getenv func(string) string
        // ConfigPath is an explicitly requested configuration file.
        ConfigPath string
        // HomeDir is the user's home directory.
        HomeDir string
        // ExecutablePath is the CLI executable, whose directory is searched first.
        ExecutablePath string
        // XDGConfigHome is $XDG_CONFIG_HOME; when set it replaces <home>/.config.
        XDGConfigHome string
        // SkipFile ignores every configuration file, including ConfigPath.
        SkipFile bool
}

// Load resolves the CLI configuration. Precedence, lowest first: built-in
// defaults, configuration file, GO_SPIN_BOARD_* environment variables. The
// command-line flags are applied on top of the result by the caller, so the
// values are validated afterwards by Validate: a file that names a wrong URL
// must not break a command that overrides it with a flag.
func Load(opts LoadOptions) (Config, error) {
        getenv := opts.Getenv
        if getenv == nil {
                getenv = os.Getenv
        }

        cfg := Config{
                URL:     DefaultURL,
                Board:   ghostboard.DefaultBoard,
                Timeout: DefaultTimeout,
        }

        path, err := resolveConfigPath(opts)
        if err != nil {
                return Config{}, err
        }
        if path != "" {
                warnings, err := loadFile(path, &cfg)
                if err != nil {
                        return Config{}, err
                }
                cfg.FilePath = path
                cfg.Warnings = warnings
        }

        if value := strings.TrimSpace(getenv(EnvURL)); value != "" {
                cfg.URL = value
                cfg.EnvOverrides = append(cfg.EnvOverrides, EnvURL)
        }
        if value := strings.TrimSpace(getenv(EnvTimeout)); value != "" {
                secs, convErr := strconv.Atoi(value)
                if convErr != nil || secs <= 0 {
                        return Config{}, fmt.Errorf("%s must be a positive number of seconds, got %q", EnvTimeout, value)
                }
                cfg.Timeout = time.Duration(secs) * time.Second
                cfg.EnvOverrides = append(cfg.EnvOverrides, EnvTimeout)
        }

        return cfg, nil
}

// Validate checks the configuration that will actually be used. It is called
// once the command-line flags have been applied, so a value from the file can
// always be corrected on the command line.
func (c Config) Validate() error {
        if _, err := NormalizeURL(c.URL); err != nil {
                return err
        }
        if err := ghostboard.ValidateBoardName(c.Board); err != nil {
                return &BoardNameError{Name: c.Board}
        }
        if c.Timeout <= 0 {
                return fmt.Errorf("request timeout must be positive, got %s", c.Timeout)
        }
        return nil
}

// ConfigPaths returns the configuration files the CLI looks for, most specific
// first: next to the executable (portable installs), then the per-user file. It
// never touches the filesystem.
func ConfigPaths(opts LoadOptions) []string {
        candidates := make([]string, 0, 2)
        if dir := executableDir(opts.ExecutablePath); dir != "" {
                candidates = append(candidates, filepath.Join(dir, ConfigFileName))
        }
        if path := userConfigPath(opts); path != "" {
                candidates = append(candidates, path)
        }
        return candidates
}

// userConfigPath returns the per-user configuration file, following
// XDG_CONFIG_HOME when it is set.
func userConfigPath(opts LoadOptions) string {
        if opts.XDGConfigHome != "" {
                return filepath.Join(opts.XDGConfigHome, userConfigDirName, ConfigFileName)
        }
        if opts.HomeDir != "" {
                return filepath.Join(opts.HomeDir, homeConfigSubdir, userConfigDirName, ConfigFileName)
        }
        return ""
}

// resolveConfigPath returns the configuration file to read, or an empty string
// when none exists. An explicitly requested file must be readable.
func resolveConfigPath(opts LoadOptions) (string, error) {
        if opts.SkipFile {
                return "", nil
        }

        if opts.ConfigPath != "" {
                if _, err := os.Stat(opts.ConfigPath); err != nil {
                        return "", fmt.Errorf("config file %s: %w", opts.ConfigPath, err)
                }
                return opts.ConfigPath, nil
        }

        for _, candidate := range ConfigPaths(opts) {
                info, err := os.Stat(candidate)
                switch {
                case err == nil && info.IsDir():
                        continue
                case err == nil:
                        return candidate, nil
                case errors.Is(err, fs.ErrNotExist):
                        continue
                default:
                        return "", fmt.Errorf("checking config file %s: %w", candidate, err)
                }
        }
        return "", nil
}

func executableDir(executablePath string) string {
        if executablePath == "" {
                return ""
        }
        return filepath.Dir(executablePath)
}

// knownConfigKeys are the keys a configuration file may carry.
var knownConfigKeys = []string{"url", "board", "timeout_secs"}

// loadFile applies the keys of a configuration file on top of cfg. Absent keys
// keep their current value; the values themselves are validated later, on the
// effective configuration, so a flag can override a mistyped file entry. The
// returned warnings name the keys the CLI does not understand.
func loadFile(path string, cfg *Config) ([]string, error) {
        reader := viper.New()
        reader.SetConfigFile(path)
        if err := reader.ReadInConfig(); err != nil {
                return nil, fmt.Errorf("reading config file %s: %w", path, err)
        }

        if reader.InConfig("url") {
                cfg.URL = strings.TrimSpace(reader.GetString("url"))
        }
        if reader.InConfig("board") {
                cfg.Board = strings.TrimSpace(reader.GetString("board"))
        }
        if reader.InConfig("timeout_secs") {
                cfg.Timeout = time.Duration(reader.GetInt("timeout_secs")) * time.Second
        }

        return unknownKeys(reader, path), nil
}

// unknownKeys names the configuration entries the CLI does not read, so a
// mistyped key is reported instead of silently keeping a default.
func unknownKeys(reader *viper.Viper, path string) []string {
        settings := reader.AllSettings()
        names := make([]string, 0, len(settings))
        for name := range settings {
                names = append(names, name)
        }
        sort.Strings(names)

        var warnings []string
        for _, name := range names {
                if slices.Contains(knownConfigKeys, name) {
                        continue
                }
                warnings = append(warnings, fmt.Sprintf("unknown key %q in %s: known keys are url, board and timeout_secs", name, path))
        }
        return warnings
}

)

// StartPersistenceScheduler runs a goroutine that periodically flushes dirty cache to disk.
// On ctx.Done, it performs a final flush before returning.
// Returns a channel that is closed when the scheduler has completed shutdown.
func StartPersistenceScheduler(
        ctx context.Context,
        store PersistableStore,
        repo repository.Saver,
        interval time.Duration,
) <-chan struct{} {
        done := make(chan struct{})
        logger.WithComponent("persist").Debugf("starting persistence scheduler with interval: %v", interval)
        ticker := time.NewTicker(interval)
        go func() {
                defer close(done)
                defer ticker.Stop()
                logger.WithComponent("persist").Debugf("persistence scheduler running")
                for {
                        select {
                        case <-ctx.Done():
                                logger.WithComponent("persist").Debugf("persistence scheduler received context cancellation, performing final flush")
                                // Final flush on shutdown: preserve any context values (trace IDs, deadlines) from ctx
                                // but detach its cancellation so the flush can complete even though the parent is cancelled.
                                shutdownCtx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 30*time.Second)
                                flushCache(shutdownCtx, store, repo)
                                cancel()
                                logger.WithComponent("persist").Info("persistence scheduler stopped after final flush")
                                return
                        case <-ticker.C:
                                logger.WithComponent("persist").Tracef("persistence scheduler tick, checking if dirty")
                                flushCache(ctx, store, repo)
                        }
                }
        }()
        return done
}

// flushCache persists the cache to disk if dirty, using optimistic locking.
// It respects context cancellation to allow graceful shutdown.
func flushCache(ctx context.Context, store PersistableStore, repo repository.Saver) {
        if !store.IsDirty() {
                logger.WithComponent("persist").Tracef("cache is clean, skipping flush")
                return
        }

        // Check for context cancellation before proceeding
        if err := ctx.Err(); err != nil {
                logger.WithComponent("persist").Debugf("flush cancelled: %v", err)
                return
        }

        logger.WithComponent("persist").Debugf("cache is dirty, flushing to disk")
        // Cache is dirty → persist
        snapshot, err := store.Snapshot()
        if err != nil {
                logger.WithComponent("persist").Errorf("persist error: failed to get snapshot: %v", err)
                return
        }

        snapshot.Metadata.LastUpdate = time.Now().UnixMilli()

        if err := repo.Save(ctx, &snapshot); err != nil {
                logger.WithComponent("persist").Errorf("persist error: failed to save: %v", err)
                return
        }

        store.ClearDirty()
        store.SetLastUpdate(snapshot.Metadata.LastUpdate)
        logger.WithComponent("persist").Info("cache persisted to disk")
}

// Store keeps an in-memory copy of the data document.
type Store struct {
        data       repository.DataDocument
        lastUpdate int64
        mu         sync.RWMutex
        dirty      bool
}

// NewStore creates an empty cache store.
func NewStore(doc repository.DataDocument) *Store {
        return &Store{data: doc, lastUpdate: doc.Metadata.LastUpdate}
}

// MarkDirty sets the dirty flag to true.
func (s *Store) MarkDirty() {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.dirty = true
}

// IsDirty returns true if cache has uncommitted changes.
func (s *Store) IsDirty() bool {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return s.dirty
}

// ClearDirty resets the dirty flag.
func (s *Store) ClearDirty() {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.dirty = false
}

// GetLastUpdate returns the cache's last update timestamp.
func (s *Store) GetLastUpdate() int64 {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return s.lastUpdate
}

// SetLastUpdate sets the cache's last update timestamp.
func (s *Store) SetLastUpdate(ts int64) {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.lastUpdate = ts
}

// Snapshot returns a deep copy of the cached data.
func (s *Store) Snapshot() (repository.DataDocument, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return cloneData(s.data)
}

// Replace swaps the cached data.
func (s *Store) Replace(doc repository.DataDocument) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        cloned, err := cloneData(doc)
        if err != nil {
                return err
        }
        s.data = cloned
        s.lastUpdate = doc.Metadata.LastUpdate
        s.dirty = false

        return nil
}

// AddContainer upserts a container by name, updating order and returning the new snapshot.
func (s *Store) AddContainer(container repository.Container) (repository.DataDocument, error) {
        logger.WithComponent("cache").Debugf("adding/updating container: %s", container.Name)
        s.mu.Lock()
        defer s.mu.Unlock()

        clonedContainer, err := cloneContainer(container)
        if err != nil {
                return repository.DataDocument{}, err
        }

        // Normalize FriendlyName to lowercase for consistency
        clonedContainer.FriendlyName = strings.ToLower(clonedContainer.FriendlyName)

        inOrder := false
        for _, name := range s.data.Order {
                if name == clonedContainer.Name {
                        inOrder = true
                        break
                }
        }

        replaced := false
        for i := range s.data.Containers {
                if s.data.Containers[i].Name == clonedContainer.Name {
                        s.data.Containers[i] = clonedContainer
                        replaced = true
                        break
                }
        }

        if !replaced {
                s.data.Containers = append(s.data.Containers, clonedContainer)
        }

        if !inOrder {
                s.data.Order = append(s.data.Order, clonedContainer.Name)
        }

        // Mark cache as dirty after mutation
        s.dirty = true

        return cloneData(s.data)
}

// RemoveContainer deletes a container by name and removes it from the order list.
func (s *Store) RemoveContainer(name string) (repository.DataDocument, error) {
        logger.WithComponent("cache").Debugf("removing container: %s", name)
        s.mu.Lock()
        defer s.mu.Unlock()

        idx := -1
        for i := range s.data.Containers {
                if s.data.Containers[i].Name == name {
                        idx = i
                        break
                }
        }
        if idx == -1 {
                return repository.DataDocument{}, ErrContainerNotFound
        }

        // Remove from Containers slice
        s.data.Containers = append(s.data.Containers[:idx], s.data.Containers[idx+1:]...)

        // Remove from Order slice
        for i := 0; i < len(s.data.Order); i++ {
                if s.data.Order[i] == name {
                        s.data.Order = append(s.data.Order[:i], s.data.Order[i+1:]...)
                        i--
                }
        }

        // Mark cache as dirty after mutation
        s.dirty = true

        // Remove schedules that target this container
        newSchedules := make([]repository.Schedule, 0, len(s.data.Schedules))
        for _, sch := range s.data.Schedules {
                if sch.TargetType == "container" && sch.Target == name {
                        logger.WithComponent("cache").Debugf("removing schedule %s because it targets deleted container %s", sch.ID, name)
                        continue
                }
                newSchedules = append(newSchedules, sch)
        }
        s.data.Schedules = newSchedules

        // Remove container references from all groups
        for gi := range s.data.Groups {
                newContainers := make([]string, 0, len(s.data.Groups[gi].Container))
                for _, cname := range s.data.Groups[gi].Container {
                        if cname == name {
                                logger.WithComponent("cache").Debugf("removing container %s from group %s", name, s.data.Groups[gi].Name)
                                continue
                        }
                        newContainers = append(newContainers, cname)
                }
                s.data.Groups[gi].Container = newContainers
        }

        return cloneData(s.data)
}

import (

        "os"

        "gopkg.in/yaml.v3"

// ActionConfig holds the technical configuration for a single HTTP action.
// These fields are server-side only and never exposed to the browser.
type ActionConfig struct {
        Method  string            `yaml:"method"  json:"method"`
        URL     string            `yaml:"url"     json:"url"`
        Headers map[string]string `yaml:"headers" json:"headers"`
        Body    string            `yaml:"body"    json:"body"`
}

// ActionsFile is the root structure of the actions.yml configuration file.
type ActionsFile struct {
        Actions map[string]ActionConfig `yaml:"actions"`
}

// LoadActionsConfig reads the YAML file at the given path and returns a parsed
// ActionsFile. If the file does not exist, it returns an empty actions map
// with a nil error.
func LoadActionsConfig(path string) (*ActionsFile, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                if os.IsNotExist(err) {
                        return &ActionsFile{Actions: make(map[string]ActionConfig)}, nil
                }
                return nil, fmt.Errorf("failed to read actions config file %s: %w", path, err)
        }

        var cfg ActionsFile
        if err := yaml.Unmarshal(data, &cfg); err != nil {
                return nil, fmt.Errorf("failed to parse actions config file %s: %w", path, err)
        }

        if cfg.Actions == nil {
                cfg.Actions = make(map[string]ActionConfig)
        }

        // Validate required fields for each action
        for name, action := range cfg.Actions {
                if action.Method == "" {
                        return nil, fmt.Errorf("action %q: method is required", name)
                }
                if action.URL == "" {
                        return nil, fmt.Errorf("action %q: url is required", name)
                }
                // SSRF guard: only http/https are permitted. This blocks dangerous
                // schemes such as file://, ftp:// or gopher:// from being proxied.
                u, err := url.Parse(action.URL)
                if err != nil {
                        return nil, fmt.Errorf("action %q: invalid url: %w", name, err)
                }
                if u.Scheme != "http" && u.Scheme != "https" {
                        return nil, fmt.Errorf("action %q: unsupported url scheme %q (only http and https are allowed)", name, u.Scheme)
                }
        }

        return &cfg, nil
}

// defaultCORSAllowedOrigins is the permissive allow-list used when the CORS key
// is absent from both the configuration file and the environment.
const defaultCORSAllowedOrigins = "*"

// corsAllowedOrigins resolves the CORS allow-list. The canonical key is
// misc.cors_allowed_origins (the one documented in config/config.yaml and
// README.md); server.cors_allowed_origins is still
// honoured as a deprecated alias so upgrades never silently widen the
// allow-list to the permissive default. When both keys are set to different
// values the canonical one wins, with a warning naming both: an installation that
// still overrides the deprecated key through the environment would otherwise
// switch allow-lists without any trace in the logs.
func corsAllowedOrigins() string {
        miscOrigins := viper.GetString("misc.cors_allowed_origins")
        serverOrigins := viper.GetString("server.cors_allowed_origins")

        switch {
        case miscOrigins != "" && serverOrigins != "" && miscOrigins != serverOrigins:
                logger.WithComponent("config").Warnf("both misc.cors_allowed_origins (%q) and the deprecated server.cors_allowed_origins (%q) are set, using misc.cors_allowed_origins", miscOrigins, serverOrigins)
                return miscOrigins
        case miscOrigins != "":
                return miscOrigins
        case serverOrigins != "":
                logger.WithComponent("config").Warn("server.cors_allowed_origins is deprecated, use misc.cors_allowed_origins")
                return serverOrigins
        default:
                return defaultCORSAllowedOrigins
        }
}

func dataFileExistenceCheck() error {
        fileStorePath := viper.GetString("data.file_path")
        logger.WithComponent("config").Infof("Using data file: %s", fileStorePath)

        // Ensure the directory for the data file exists
        dataDir := filepath.Dir(fileStorePath)
        if err := os.MkdirAll(dataDir, 0755); err != nil {
                return fmt.Errorf("failed to create data directory %s: %w", dataDir, err)
        }

        //if the file does not exist, create an empty one with empty json object
        if _, err := os.Stat(fileStorePath); os.IsNotExist(err) {
                emptyFile, err := os.Create(fileStorePath)
                if err != nil {
                        return fmt.Errorf("failed to create data file %s: %w", fileStorePath, err)
                }
                defer func() {
                        if cerr := emptyFile.Close(); cerr != nil {
                                logger.WithComponent("config").Errorf("failed to close data file %s: %v", fileStorePath, cerr)
                        }
                }()
                if _, err := emptyFile.WriteString("{}"); err != nil {
                        return fmt.Errorf("failed to write to data file %s: %w", fileStorePath, err)
                }
                logger.WithComponent("config").Infof("Created new EMPTY data file: %s", fileStorePath)
        }
        return nil
}

// validate checks required configuration fields
func (c *Config) validate() error {
        if c.Data.RefreshIntervalSecs <= 0 {
                return fmt.Errorf("data.refresh_interval_secs must be positive")
        }
        if c.Data.StatsRefreshIntervalSecs <= 0 {
                return fmt.Errorf("data.stats_refresh_interval_secs must be positive")
        }
        if c.Data.FilePath == "" {
                return fmt.Errorf("data.file_path configuration is required")
        }
        if c.Data.PersistInterval <= 0 {
                return fmt.Errorf("data.persist_interval_secs must be positive")
        }
        if c.Data.SchedulingPoll <= 0 {
                return fmt.Errorf("data.scheduling_poll_interval_secs must be positive")
        }
        if c.Server.Port < 1 || c.Server.Port > 65535 {
                return fmt.Errorf("server.port must be a valid TCP port (1-65535)")
        }
        if c.Server.ShutDownTimeout <= 0 {
                return fmt.Errorf("server.shutdown_timeout_secs must be positive")
        }
        if c.Server.IdleTimeout <= 0 {
                return fmt.Errorf("server.idle_timeout_secs must be positive")
        }
        if c.Server.WriteTimeout <= 0 {
                return fmt.Errorf("server.write_timeout_secs must be positive")
        }
        if c.Server.ReadTimeout <= 0 {
                return fmt.Errorf("server.read_timeout_secs must be positive")
        }
        if c.Server.RequestTimeout <= 0 {
                return fmt.Errorf("server.request_timeout_millis must be positive")
        }
        if c.Misc.SchedulingTZ != "" && c.Misc.SchedulingTZ != "Local" {
                if _, err := time.LoadLocation(c.Misc.SchedulingTZ); err != nil {
                        return fmt.Errorf("misc.scheduling_timezone is invalid: %w", err)
                }
        }
        if c.Data.ActionsMode != "" && c.Data.ActionsMode != "sync" && c.Data.ActionsMode != "async" {
                return fmt.Errorf("data.actions_mode must be 'sync' or 'async', got %q", c.Data.ActionsMode)
        }
        if c.Data.ActionsTimeoutSeconds <= 0 {
                return fmt.Errorf("data.actions_timeout_seconds must be positive")
        }
        if c.Ghostboard.ClearDelay <= 0 {
                return fmt.Errorf("ghostboard.clear_delay_secs must be positive")
        }
        if c.Ghostboard.MaxTextBytes <= 0 {
                return fmt.Errorf("ghostboard.max_text_bytes must be positive")
        }
        if c.Ghostboard.PersistInterval <= 0 {
                return fmt.Errorf("ghostboard.persist_interval_secs must be positive")
        }
        if c.Ghostboard.HeartbeatInterval <= 0 {
                return fmt.Errorf("ghostboard.heartbeat_interval_secs must be positive")
        }

        return nil
}

// getEnvOrDefault returns env var value or default
func getEnvOrDefault(key, defaultValue string) string {
        if value := os.Getenv(key); value != "" {
                return value
        }
        return defaultValue
}

func getEnvOrViperPort(envKey, viperKey string) (int, error) {
        if value := os.Getenv(envKey); value != "" {
                port, err := strconv.Atoi(value)
                if err != nil {
                        return 0, fmt.Errorf("%s must be a valid integer TCP port: %w", envKey, err)
                }
                return port, nil
        }
        return viper.GetInt(viperKey), nil
}

func maskToken(token string) string {
        if len(token) <= 8 {
                if len(token) == 0 {
                        return "(not set)"
                }
                return "***"
        }
        return token[:4] + "..." + token[len(token)-4:]
}

// dashboardIconsBase is the CDN base URL for homarr-labs dashboard-icons
const dashboardIconsBase = "https://cdn.jsdelivr.net/gh/homarr-labs/dashboard-icons"

// resolveIconURL resolves an icon reference to a full CDN URL.
// If the icon is already a full URL (http/https), it is returned as-is.
// Otherwise, the extension is extracted and the URL is built as:
//
//        https://cdn.jsdelivr.net/gh/homarr-labs/dashboard-icons/{ext}/{filename}
//
// Examples:
//
//        bitmagnet.png → https://cdn.jsdelivr.net/gh/homarr-labs/dashboard-icons/png/bitmagnet.png
//        palla.svg     → https://cdn.jsdelivr.net/gh/homarr-labs/dashboard-icons/svg/palla.svg
func resolveIconURL(icon string) string {
        if icon == "" {
                return ""
        }

        // Already a full URL
        if strings.HasPrefix(icon, "http://") || strings.HasPrefix(icon, "https://") {
                return icon
        }

        // Extract extension (without dot) and build CDN URL
        ext := strings.TrimPrefix(filepath.Ext(icon), ".")
        if ext == "" {
                return icon // no extension, return as-is
        }

        return fmt.Sprintf("%s/%s/%s", dashboardIconsBase, ext, icon)
}

// DefaultTheme is the theme used when homepage.yaml omits the theme or names
// one the UI does not ship. "auto" delegates to the dark or light theme of the
// OS, which is the behaviour of the unconfigured page.
const DefaultTheme = "auto"

// HomepageThemes lists the body classes styled in ui/home.html. Any other name
// has no CSS variables, so it would render an unstyled page.
var HomepageThemes = []string{
        "auto",
        "tokyo-night",
        "catppuccin-latte",
        "nord",
        "dracula",
        "gruvbox",
        "night-stars",
}

// NormalizeHomepageTheme lowercases a configured theme name, turns spaces and
// underscores into dashes and falls back to DefaultTheme when the result is not
// one of HomepageThemes. Without the fallback a typo or a name from an older
// release would reach the browser, where it matches no CSS rule and leaves the
// homepage without colours or layout.
func NormalizeHomepageTheme(theme string) string {
        name := strings.Join(strings.Fields(strings.ToLower(theme)), "-")
        name = strings.ReplaceAll(name, "_", "-")

        for _, known := range HomepageThemes {
                if name == known {
                        return name
                }
        }

        return DefaultTheme
}

// normalizeHomepageConfig applies the documented defaults and makes every
// collection non-nil. The JSON response must carry empty arrays instead of
// null: the Alpine templates iterate the sections with x-for, which fails on a
// null value, and a missing YAML key is a normal configuration, not an error.
func normalizeHomepageConfig(cfg *HomepageConfig) {
        if cfg.Services == nil {
                cfg.Services = []ServiceGroup{}
        }
        if cfg.Bookmarks == nil {
                cfg.Bookmarks = []BookmarkGroup{}
        }
        if cfg.Actions == nil {
                cfg.Actions = []ActionGroup{}
        }

        if cfg.Settings.PollingIntervalSeconds <= 0 {
                cfg.Settings.PollingIntervalSeconds = DefaultPollingIntervalSeconds
        }
        if cfg.Settings.StatsPollingIntervalSeconds <= 0 {
                cfg.Settings.StatsPollingIntervalSeconds = DefaultStatsPollingIntervalSeconds
        }
        if cfg.Settings.FontFamily == "" {
                cfg.Settings.FontFamily = DefaultFontFamily
        }
        if cfg.Settings.FontSize == "" {
                cfg.Settings.FontSize = DefaultFontSize
        }
        if cfg.Settings.TitleFontSize == "" {
                cfg.Settings.TitleFontSize = DefaultTitleFontSize
        }
        cfg.Settings.Theme = NormalizeHomepageTheme(cfg.Settings.Theme)

        // Resolve icon URLs and keep the item lists non-nil for the same reason as
        // the sections above.
        for i := range cfg.Services {
                if cfg.Services[i].Items == nil {
                        cfg.Services[i].Items = []ServiceItem{}
                }
                for j := range cfg.Services[i].Items {
                        cfg.Services[i].Items[j].Icon = resolveIconURL(cfg.Services[i].Items[j].Icon)
                }
        }
        for i := range cfg.Bookmarks {
                if cfg.Bookmarks[i].Items == nil {
                        cfg.Bookmarks[i].Items = []BookmarkItem{}
                }
        }
        for i := range cfg.Actions {
                if cfg.Actions[i].Items == nil {
                        cfg.Actions[i].Items = []ActionItem{}
                }
                for j := range cfg.Actions[i].Items {
                        cfg.Actions[i].Items[j].Icon = resolveIconURL(cfg.Actions[i].Items[j].Icon)
                }
        }
}

// LoadHomepageConfig reads the YAML file at the given path and returns a parsed HomepageConfig.
// If the file does not exist, it returns an empty config with a nil error.
// It also returns the SHA-256 hash of the raw file content for change detection.
func LoadHomepageConfig(path string) (*HomepageConfig, string, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                if os.IsNotExist(err) {
                        cfg := &HomepageConfig{}
                        normalizeHomepageConfig(cfg)

                        return cfg, "", nil
                }
                return nil, "", fmt.Errorf("failed to read homepage config file: %w", err)
        }

        // Compute hash for change detection
        hash := fmt.Sprintf("%x", sha256.Sum256(data))

        var cfg HomepageConfig
        if err := yaml.Unmarshal(data, &cfg); err != nil {
                return nil, "", fmt.Errorf("failed to parse homepage config file: %w", err)
        }

        normalizeHomepageConfig(&cfg)

        return &cfg, hash, nil
}

type RouteResponse struct {
        Status string      `json:"status"`
        Data   []RouteItem `json:"data"`
}

type RouteItem struct {
        Mode          string `json:"Mode"`
        Name          string `json:"Name"`
        Target        string `json:"Target"`
        Description   string `json:"Description"`
        Host          string `json:"Host"`
        PathPrefix    string `json:"PathPrefix"`
        Disabled      bool   `json:"Disabled"`
        UseHost       bool   `json:"UseHost"`
        UsePathPrefix bool   `json:"UsePathPrefix"`
}

type Client struct {
        httpClient *http.Client
}

func NewClient() *Client {
        return &Client{
                httpClient: &http.Client{
                        Timeout: 30 * time.Second,
                },
        }
}

func (c *Client) FetchRoutes(ctx context.Context, baseUrl, token string) ([]RouteItem, error) {
        url := fmt.Sprintf("%s/cosmos/api/routes", baseUrl)
        logger.WithComponent("cosmos-client").Debugf("fetching routes from: %s", url)

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
        if err != nil {
                return nil, fmt.Errorf("failed to create request: %w", err)
        }

        req.Header.Set("accept", "application/json")
        req.Header.Set("Authorization", fmt.Sprintf("Bearer %s", token))

        resp, err := c.httpClient.Do(req)
        if err != nil {
                return nil, fmt.Errorf("failed to fetch routes: %w", err)
        }
        defer func() {
                if cerr := resp.Body.Close(); cerr != nil {
                        logger.WithComponent("cosmos-client").Debugf("failed to close response body: %v", cerr)
                }
        }()

        if resp.StatusCode != http.StatusOK {
                body, readErr := io.ReadAll(resp.Body)
                if readErr != nil {
                        return nil, fmt.Errorf("API returned status %d (and body read failed: %w)", resp.StatusCode, readErr)
                }
                return nil, fmt.Errorf("API returned status %d: %s", resp.StatusCode, string(body))
        }

        var routeResp RouteResponse
        if err := json.NewDecoder(resp.Body).Decode(&routeResp); err != nil {
                return nil, fmt.Errorf("failed to decode response: %w", err)
        }

        if routeResp.Status != "OK" {
                return nil, fmt.Errorf("API returned status: %s", routeResp.Status)
        }

        logger.WithComponent("cosmos-client").Debugf("fetched %d routes from cosmos", len(routeResp.Data))
        return routeResp.Data, nil
}

func FilterValidRoutes(routes []RouteItem) []RouteItem {
        var valid []RouteItem
        for _, route := range routes {
                if !route.Disabled && route.Mode == "SERVAPP" && route.UseHost {
                        valid = append(valid, route)
                }
        }
        logger.WithComponent("cosmos-client").Debugf("filtered %d valid routes from %d total", len(valid), len(routes))
        return valid
}

import (

type ImportResult struct {
        ImportedNames   []string `json:"imported_names"`
        SkippedNames    []string `json:"skipped_names"`
        Errors          []string `json:"errors,omitempty"`
        TotalFetched    int      `json:"total_fetched"`
        Filtered        int      `json:"filtered"`
        Imported        int      `json:"imported"`
        SkippedExisting int      `json:"skipped_existing"`
}

type ImportService struct {
        client *Client
        store  cache.ContainerStore
}

func NewImportService(client *Client, store cache.ContainerStore) *ImportService {
        return &ImportService{
                client: client,
                store:  store,
        }
}

func (s *ImportService) ImportContainers(ctx context.Context, baseUrl, token string) (*ImportResult, error) {
        logger.WithComponent("cosmos-import").Info("starting container import from cosmos")

        routes, err := s.client.FetchRoutes(ctx, baseUrl, token)
        if err != nil {
                return nil, err
        }

        result := &ImportResult{
                TotalFetched:  len(routes),
                ImportedNames: []string{},
                SkippedNames:  []string{},
                Errors:        []string{},
        }

        validRoutes := FilterValidRoutes(routes)
        result.Filtered = len(validRoutes)

        existingContainers, err := s.getExistingContainerNames()
        if err != nil {
                return nil, err
        }

        for _, route := range validRoutes {
                containerName := extractContainerName(route.Target)
                if containerName == "" {
                        logger.WithComponent("cosmos-import").Warnf("skipping route with invalid target: %s", route.Target)
                        continue
                }

                if existingContainers[containerName] {
                        logger.WithComponent("cosmos-import").Debugf("skipping existing container: %s", containerName)
                        result.SkippedExisting++
                        result.SkippedNames = append(result.SkippedNames, containerName)
                        continue
                }

                container := s.routeToContainer(route)
                if _, err := s.store.AddContainer(container); err != nil {
                        logger.WithComponent("cosmos-import").Warnf("failed to add container %s: %v", containerName, err)
                        result.Errors = append(result.Errors, err.Error())
                        continue
                }

                logger.WithComponent("cosmos-import").Infof("imported container: %s", containerName)
                result.Imported++
                result.ImportedNames = append(result.ImportedNames, containerName)
        }

        logger.WithComponent("cosmos-import").Infof("import completed: total=%d, filtered=%d, imported=%d, skipped=%d, errors=%d",
                result.TotalFetched, result.Filtered, result.Imported, result.SkippedExisting, len(result.Errors))

        return result, nil
}

func (s *ImportService) getExistingContainerNames() (map[string]bool, error) {
        doc, err := s.store.Snapshot()
        if err != nil {
                return nil, err
        }

        names := make(map[string]bool)
        for _, c := range doc.Containers {
                names[c.Name] = true
        }
        return names, nil
}

func (s *ImportService) routeToContainer(route RouteItem) repository.Container {
        url := route.Host
        if !strings.HasPrefix(url, "http://") && !strings.HasPrefix(url, "https://") {
                url = "https://" + url
        }

        containerName := extractContainerName(route.Target)

        active := true
        favorite := false
        pinned := false
        running := false

        return repository.Container{
                Name:         containerName,
                FriendlyName: route.Name,
                URL:          url,
                Active:       &active,
                Favorite:     &favorite,
                Pinned:       &pinned,
                Running:      &running,
        }
}

func extractContainerName(target string) string {
        if target == "" {
                return ""
        }

        idx := strings.Index(target, "//")
        if idx == -1 {
                return ""
        }

        afterSlashSlash := target[idx+2:]
        endIdx := strings.Index(afterSlashSlash, ":")
        if endIdx == -1 {
                return afterSlashSlash
        }

        return afterSlashSlash[:endIdx]
}

// Package ghostboard implements the multi-board shared text store behind the
// /ui/ghostboard page and its REST/SSE API.
//
// A board is addressed by name and holds the full text plus a revision counter.
// Subscribers (one per open event stream) receive every change so connected
// browsers stay in sync. Boards are kept in memory, optionally mirrored to a
// dedicated JSON file. The text is dropped once the configured clear delay has
// elapsed since the last write or the last subscriber leaving, whichever comes
// last, so a board written only by a script is cleared too.
package ghostboard

import (
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "os"
        "path/filepath"
        "regexp"
        "sort"
        "sync"
        "time"

        "github.com/asgambat/gospin/internal/logger"
)

const (
        // DefaultBoard is the board served when the URL does not name one.
        DefaultBoard = "default"

        // subscriberBuffer is the number of changes buffered per open event stream.
        // A slow subscriber that fills the buffer skips the intermediate events; the
        // next delivered event always carries the full text, so it converges.
        subscriberBuffer = 16

        // filePerm and dirPerm are the permissions used for the persistence file and
        // its directory.
        filePerm = 0o600
        dirPerm  = 0o755
)

// Errors returned by the store.
var (
        ErrInvalidBoardName = errors.New("invalid board name")
        ErrTextTooLarge     = errors.New("text exceeds the maximum allowed size")
)

// boardNamePattern keeps board names URL-safe and filesystem-friendly.
var boardNamePattern = regexp.MustCompile(`^[A-Za-z0-9][A-Za-z0-9._-]{0,63}$`)

// Board is a single shared text board.
type Board struct {
        UpdatedAt time.Time `json:"updated_at"`
        Name      string    `json:"name"`
        Text      string    `json:"text"`
        Revision  uint64    `json:"revision"`
}

// Event is one change broadcast to subscribers. Board fields are inlined and
// ClientID identifies the writer so its own browser can ignore the echo.
type Event struct {
        ClientID string `json:"client_id,omitempty"`
        Board
}

// Config holds the store settings. All fields are required except FilePath,
// which enables persistence when set.
type Config struct {
        FilePath        string
        ClearDelay      time.Duration
        MaxTextBytes    int
        PersistInterval time.Duration
}

// Store keeps every board in memory and fans changes out to subscribers.
type Store struct {
        boards    map[string]*boardState
        cfg       Config
        mu        sync.Mutex
        persistMu sync.Mutex
        dirty     bool
}

type boardState struct {
        subs       map[*Subscription]struct{}
        clearTimer *time.Timer
        board      Board
        // clearSeq is the generation of the pending idle clear. It sits after the
        // pointer fields so the GC pointer prefix stays as small as possible.
        clearSeq uint64
}

// Subscription is one open event stream for a board. Close must be called when
// the stream ends so the board can be cleared once it stays idle.
type Subscription struct {
        store   *Store
        events  chan Event
        board   string
        closeMu sync.Once
}

// Events returns the channel receiving board changes. It is closed when the
// subscription is closed.
func (s *Subscription) Events() <-chan Event {
        return s.events
}

// Close removes the subscription. It is safe to call more than once.
func (s *Subscription) Close() {
        s.closeMu.Do(func() {
                s.store.unsubscribe(s)
        })
}

// New creates a store, loading the persistence file when one is configured.
func New(cfg Config) (*Store, error) {
        if cfg.ClearDelay <= 0 {
                return nil, errors.New("ghostboard clear delay must be positive")
        }
        if cfg.MaxTextBytes <= 0 {
                return nil, errors.New("ghostboard max text size must be positive")
        }
        if cfg.PersistInterval <= 0 {
                return nil, errors.New("ghostboard persist interval must be positive")
        }

        store := &Store{
                cfg:    cfg,
                boards: make(map[string]*boardState),
        }

        if cfg.FilePath != "" {
                if dir := filepath.Dir(cfg.FilePath); dir != "" && dir != "." {
                        if err := os.MkdirAll(dir, dirPerm); err != nil {
                                return nil, fmt.Errorf("create ghostboard directory %s: %w", dir, err)
                        }
                }
                if err := store.load(); err != nil {
                        return nil, err
                }
        }

        return store, nil
}

// ValidateBoardName reports whether name can address a board.
func ValidateBoardName(name string) error {
        if !boardNamePattern.MatchString(name) {
                return ErrInvalidBoardName
        }
        return nil
}

// Get returns the current state of a board. Unknown boards are returned as
// empty boards with revision zero.
func (s *Store) Get(name string) (Board, error) {
        if err := ValidateBoardName(name); err != nil {
                return Board{}, err
        }

        s.mu.Lock()
        defer s.mu.Unlock()

        if state, ok := s.boards[name]; ok {
                return state.board, nil
        }
        return Board{Name: name}, nil
}

// Update stores a new text for the board and broadcasts it to every subscriber.
// Updating a board with its current text is a no-op so typing pauses do not
// bump the revision. Accepted writes count as activity: they re-arm the idle
// clear, so a board written by a script alone is cleared after clear_delay too.
func (s *Store) Update(name, text, clientID string) (Board, error) {
        if err := ValidateBoardName(name); err != nil {
                return Board{}, err
        }
        if len(text) > s.cfg.MaxTextBytes {
                return Board{}, ErrTextTooLarge
        }

        s.mu.Lock()
        defer s.mu.Unlock()

        state := s.boardLocked(name)
        if state.board.Text != text {
                state.board.Text = text
                state.board.Revision++
                state.board.UpdatedAt = time.Now().UTC()
                s.dirty = true

                event := Event{Board: state.board, ClientID: clientID}
                for sub := range state.subs {
                        select {
                        case sub.events <- event:
                        default:
                                logger.WithComponent("ghostboard").Debugf("dropping event for slow subscriber on board %q", name)
                        }
                }
        }

        s.scheduleIdleClearLocked(name, state)

        return state.board, nil
}

// Subscribe registers an event stream for the board and returns the current
// state to send as the initial snapshot. A pending clear is cancelled.
func (s *Store) Subscribe(name string) (*Subscription, Board, error) {
        if err := ValidateBoardName(name); err != nil {
                return nil, Board{}, err
        }

        s.mu.Lock()
        defer s.mu.Unlock()

        state := s.boardLocked(name)
        if state.clearTimer != nil {
                state.clearTimer.Stop()
                state.clearTimer = nil
        }

        sub := &Subscription{
                store:  s,
                board:  name,
                events: make(chan Event, subscriberBuffer),
        }
        state.subs[sub] = struct{}{}

        return sub, state.board, nil
}

// StartPersistence periodically writes dirty boards to disk until ctx is
// cancelled, flushing one last time on shutdown. It is a no-op without a file.
func (s *Store) StartPersistence(ctx context.Context) {
        if s.cfg.FilePath == "" {
                return
        }

        go func() {
                ticker := time.NewTicker(s.cfg.PersistInterval)
                defer ticker.Stop()

                for {
                        select {
                        case <-ctx.Done():
                                if err := s.Flush(); err != nil {
                                        logger.WithComponent("ghostboard").Errorf("final ghostboard flush failed: %v", err)
                                }
                                return
                        case <-ticker.C:
                                if err := s.Flush(); err != nil {
                                        logger.WithComponent("ghostboard").Errorf("ghostboard flush failed: %v", err)
                                }
                        }
                }
        }()
}

// Flush writes the boards to the persistence file when it is enabled and the
// store has pending changes. Empty boards are not persisted.
func (s *Store) Flush() error {
        if s.cfg.FilePath == "" {
                return nil
        }

        s.persistMu.Lock()
        defer s.persistMu.Unlock()

        s.mu.Lock()
        if !s.dirty {
                s.mu.Unlock()
                return nil
        }
        doc := fileDocument{Boards: s.persistedBoardsLocked()}
        s.dirty = false
        s.mu.Unlock()

        data, err := json.MarshalIndent(doc, "", "  ")
        if err != nil {
                s.markDirty()
                return fmt.Errorf("marshal ghostboard file: %w", err)
        }

        tmpPath := s.cfg.FilePath + ".tmp"
        if err := os.WriteFile(tmpPath, append(data, '\n'), filePerm); err != nil {
                s.markDirty()
                return fmt.Errorf("write ghostboard file: %w", err)
        }
        if err := os.Rename(tmpPath, s.cfg.FilePath); err != nil {
                s.markDirty()
                return fmt.Errorf("replace ghostboard file: %w", err)
        }

        logger.WithComponent("ghostboard").Debugf("persisted %d board(s) to %s", len(doc.Boards), s.cfg.FilePath)
        return nil
}

// MaxTextBytes returns the configured limit for a board text.
func (s *Store) MaxTextBytes() int {
        return s.cfg.MaxTextBytes
}

type fileDocument struct {
        Boards []Board `json:"boards"`
}

func (s *Store) load() error {
        data, err := os.ReadFile(s.cfg.FilePath)
        if err != nil {
                if errors.Is(err, os.ErrNotExist) {
                        return nil
                }
                return fmt.Errorf("read ghostboard file: %w", err)
        }

        var doc fileDocument
        if err := json.Unmarshal(data, &doc); err != nil {
                return fmt.Errorf("decode ghostboard file %s: %w", s.cfg.FilePath, err)
        }

        for _, board := range doc.Boards {
                if err := ValidateBoardName(board.Name); err != nil {
                        return fmt.Errorf("ghostboard file contains an invalid board name %q: %w", board.Name, err)
                }
                if len(board.Text) > s.cfg.MaxTextBytes {
                        return fmt.Errorf("board %q in ghostboard file exceeds max_text_bytes", board.Name)
                }
                s.boards[board.Name] = &boardState{
                        board: board,
                        subs:  make(map[*Subscription]struct{}),
                }
        }

        logger.WithComponent("ghostboard").Infof("loaded %d board(s) from %s", len(s.boards), s.cfg.FilePath)
        return nil
}

func (s *Store) boardLocked(name string) *boardState {
        if state, ok := s.boards[name]; ok {
                return state
        }
        state := &boardState{
                board: Board{Name: name},
                subs:  make(map[*Subscription]struct{}),
        }
        s.boards[name] = state
        return state
}

func (s *Store) unsubscribe(sub *Subscription) {
        s.mu.Lock()
        defer s.mu.Unlock()

        state, ok := s.boards[sub.board]
        if ok {
                delete(state.subs, sub)
                if len(state.subs) == 0 {
                        s.scheduleIdleClearLocked(sub.board, state)
                }
        }

        close(sub.events)
}

// scheduleIdleClearLocked (re)arms the idle clear of a board. The text survives
// clear_delay after the last write or the last subscriber leaving, whichever
// happens last; a board with an open stream or no text is never scheduled.
func (s *Store) scheduleIdleClearLocked(name string, state *boardState) {
        if state.clearTimer != nil {
                state.clearTimer.Stop()
                state.clearTimer = nil
        }

        // Every schedule gets its own generation, captured by the timer before it
        // is started: a tick that fires just as a write re-arms the clear belongs
        // to a stale generation and must not empty the freshly written text.
        state.clearSeq++
        seq := state.clearSeq

        if len(state.subs) > 0 || state.board.Text == "" {
                return
        }

        state.clearTimer = time.AfterFunc(s.cfg.ClearDelay, func() {
                s.clearIfIdle(name, seq)
        })
}

func (s *Store) clearIfIdle(name string, seq uint64) {
        s.mu.Lock()
        defer s.mu.Unlock()

        state, ok := s.boards[name]
        if !ok || state.clearSeq != seq {
                // The board was written or rescheduled after this tick was armed: the
                // newer schedule owns the board now, so this stale tick must not clear it.
                return
        }
        state.clearTimer = nil

        if len(state.subs) > 0 || state.board.Text == "" {
                return
        }

        state.board.Text = ""
        state.board.Revision++
        state.board.UpdatedAt = time.Now().UTC()
        s.dirty = true

        logger.WithComponent("ghostboard").Debugf("cleared idle board %q after %s", name, s.cfg.ClearDelay)
}

func (s *Store) persistedBoardsLocked() []Board {
        boards := make([]Board, 0, len(s.boards))
        for _, state := range s.boards {
                if state.board.Text == "" {
                        continue
                }
                boards = append(boards, state.board)
        }
        sort.Slice(boards, func(i, j int) bool { return boards[i].Name < boards[j].Name })
        return boards
}

func (s *Store) markDirty() {
        s.mu.Lock()
        s.dirty = true
        s.mu.Unlock()
}

// CacheStore defines the interface for cache operations needed by the watcher callback.
type CacheStore interface {
        GetLastUpdate() int64
        IsDirty() bool
        Snapshot() (DataDocument, error)
        Replace(doc DataDocument) error
}

// JSONRepository handles disk persistence and watching of the data file.
type JSONRepository struct {
        validator *validator.Validate
        path      string
        dir       string
        base      string
        mu        sync.Mutex
}

// NewJSONRepository creates a repository for the given JSON file path.
// It returns the repository interface to avoid leaking implementation details.
func NewJSONRepository(path string) (Repository, error) {
        if path == "" {
                return nil, errors.New("data file path is required")
        }

        dir := filepath.Dir(path)
        base := filepath.Base(path)
        if dir == "" || dir == "." {
                dir = "."
        }

        v := validator.New()
        return &JSONRepository{path: path, dir: dir, base: base, validator: v}, nil
}

// Load reads the JSON file, parses and validates it.
// It respects context cancellation before performing I/O operations.
func (r *JSONRepository) Load(ctx context.Context) (*DataDocument, error) {
        logger.WithComponent("json-repo").Tracef("loading data from: %s", r.path)

        // Check for context cancellation before acquiring lock
        if err := ctx.Err(); err != nil {
                logger.WithComponent("json-repo").Warnf("load cancelled: %v", err)
                return nil, fmt.Errorf("load cancelled: %w", err)
        }

        r.mu.Lock()
        defer r.mu.Unlock()

        // Check again after acquiring lock
        if err := ctx.Err(); err != nil {
                logger.WithComponent("json-repo").Warnf("load cancelled: %v", err)
                return nil, fmt.Errorf("load cancelled: %w", err)
        }

        doc, err := r.loadUnlocked()
        if err != nil {
                logger.WithComponent("json-repo").Warnf("load failed: %v", err)
                return nil, err
        }
        logger.WithComponent("json-repo").Tracef("loaded data successfully, lastUpdate: %d, containers: %d, groups: %d, schedules: %d", doc.Metadata.LastUpdate, len(doc.Containers), len(doc.Groups), len(doc.Schedules))
        return doc, nil
}

// loadUnlocked reads the JSON file without acquiring the lock (caller must hold it).
func (r *JSONRepository) loadUnlocked() (*DataDocument, error) {
        file, err := os.Open(r.path)
        if err != nil {
                return nil, fmt.Errorf("open data file: %w", err)
        }
        defer func() {
                if cerr := file.Close(); cerr != nil {
                        logger.WithComponent("json-repo").Warnf("failed to close data file %s: %v", r.path, cerr)
                }
        }()

        var doc DataDocument
        if err := json.NewDecoder(file).Decode(&doc); err != nil {
                return nil, fmt.Errorf("decode data file: %w", err)
        }

        doc.ApplyDefaults()

        finalDoc := removeSchedulesWithMissingContainers(&doc)

        if r.validator != nil {
                if err := r.validator.Struct(finalDoc); err != nil {
                        return nil, fmt.Errorf("validate data file: %w", err)
                }
        }

        return finalDoc, nil
}

func removeSchedulesWithMissingContainers(doc *DataDocument) *DataDocument {
        if doc == nil {
                return doc
        }

        // Build lookup sets for existing containers and groups
        containerSet := make(map[string]struct{}, len(doc.Containers))
        for _, c := range doc.Containers {
                containerSet[c.Name] = struct{}{}
        }
        groupSet := make(map[string]struct{}, len(doc.Groups))
        for _, g := range doc.Groups {
                groupSet[g.Name] = struct{}{}
        }

        filtered := make([]Schedule, 0, len(doc.Schedules))
        for _, s := range doc.Schedules {
                if s.TargetType == "container" {
                        if _, ok := containerSet[s.Target]; !ok {
                                logger.WithComponent("json-repo").Warnf("removing schedule %s: target container %s not found", s.ID, s.Target)
                                continue
                        }
                }
                if s.TargetType == "group" {
                        if _, ok := groupSet[s.Target]; !ok {
                                logger.WithComponent("json-repo").Warnf("removing schedule %s: target group %s not found", s.ID, s.Target)
                                continue
                        }
                }
                filtered = append(filtered, s)
        }

        doc.Schedules = filtered
        return doc
}

// Save validates and writes the document atomically to disk.
// It respects context cancellation before performing I/O operations.
func (r *JSONRepository) Save(ctx context.Context, doc *DataDocument) error {
        if doc == nil {
                logger.WithComponent("json-repo").Debugf("save failed: document is nil")
                return errors.New("document is nil")
        }

        // Check for context cancellation before validation
        if err := ctx.Err(); err != nil {
                logger.WithComponent("json-repo").Debugf("save cancelled: %v", err)
                return fmt.Errorf("save cancelled: %w", err)
        }

        logger.WithComponent("json-repo").Debugf("saving data to: %s (lastUpdate: %d)", r.path, doc.Metadata.LastUpdate)
        if r.validator != nil {
                if err := r.validator.Struct(doc); err != nil {
                        logger.WithComponent("json-repo").Debugf("save failed: %v", err)
                        return fmt.Errorf("validate before save: %w", err)
                }
        }

        // Check for context cancellation before acquiring lock
        if err := ctx.Err(); err != nil {
                logger.WithComponent("json-repo").Debugf("save cancelled: %v", err)
                return fmt.Errorf("save cancelled: %w", err)
        }

        r.mu.Lock()
        defer r.mu.Unlock()

        // Check again after acquiring lock
        if err := ctx.Err(); err != nil {
                logger.WithComponent("json-repo").Debugf("save cancelled: %v", err)
                return fmt.Errorf("save cancelled: %w", err)
        }

        if err := r.saveUnlocked(doc); err != nil {
                logger.WithComponent("json-repo").Debugf("save failed: %v", err)
                return err
        }
        logger.WithComponent("json-repo").Debugf("data saved successfully")
        return nil
}

// saveUnlocked writes the document without acquiring the lock (caller must hold it).
func (r *JSONRepository) saveUnlocked(doc *DataDocument) error {
        payload, err := json.MarshalIndent(doc, "", "  ")
        if err != nil {
                return fmt.Errorf("marshal data: %w", err)
        }

        tmpFile, err := os.CreateTemp(r.dir, r.base+".tmp-")
        if err != nil {
                return fmt.Errorf("create temp file: %w", err)
        }
        defer func() {
                if cerr := tmpFile.Close(); cerr != nil && !errors.Is(cerr, os.ErrClosed) {
                        logger.WithComponent("json-repo").Debugf("failed to close temp file %s: %v", tmpFile.Name(), cerr)
                }
                if rerr := os.Remove(tmpFile.Name()); rerr != nil && !errors.Is(rerr, os.ErrNotExist) {
                        logger.WithComponent("json-repo").Tracef("failed to remove temp file %s: %v", tmpFile.Name(), rerr)
                }
        }()

        if _, err := tmpFile.Write(payload); err != nil {
                return fmt.Errorf("write temp file: %w", err)
        }

        if err := tmpFile.Sync(); err != nil {
                return fmt.Errorf("sync temp file: %w", err)
        }

        if err := tmpFile.Close(); err != nil {
                return fmt.Errorf("close temp file: %w", err)
        }

        if err := os.Rename(tmpFile.Name(), r.path); err != nil {
                return fmt.Errorf("replace data file: %w", err)
        }

        return nil
}

// StartWatcher listens for changes to the data file and calls onChange after debounce.
// It watches the parent directory (not the file) so atomic replace sequences (temp+rename)
// are still observed on Linux and Windows. Events are filtered by basename and
// debounced to avoid double reloads on write+chmod/rename cycles. The caller owns the
// provided context: cancel it to stop the goroutine and close the watcher cleanly.
func (r *JSONRepository) StartWatcher(ctx context.Context, cacheStore CacheStore) error {
        logger.WithComponent("json-repo").Debugf("starting file watcher for directory: %s", r.dir)
        onChange := r.MakeWatcherCallback(ctx, cacheStore)
        if onChange == nil {
                return errors.New("onChange callback is required")
        }

        watcher, err := fsnotify.NewWatcher()
        if err != nil {
                logger.WithComponent("json-repo").Debugf("failed to create watcher: %v", err)
                return fmt.Errorf("create watcher: %w", err)
        }

        if err := watcher.Add(r.dir); err != nil {
                if cerr := watcher.Close(); cerr != nil {
                        logger.WithComponent("json-repo").Debugf("failed to close watcher after Add error: %v", cerr)
                }
                logger.WithComponent("json-repo").Debugf("failed to watch directory: %v", err)
                return fmt.Errorf("watch dir: %w", err)
        }

        logger.WithComponent("json-repo").Debugf("file watcher started for: %s", r.dir)

        // Run watcher loop in the background; it exits when ctx is canceled or channels close.
        go func() {
                defer func() {
                        if cerr := watcher.Close(); cerr != nil {
                                logger.WithComponent("json-repo").Debugf("failed to close watcher: %v", cerr)
                        }
                }()

                // debounce coalesces bursty fsnotify events (write+chmod/rename) into a single reload.
                // If the timer is stopped before it fires, the scheduled onChange will not run.
                var debounce *time.Timer
                schedule := func() {
                        if debounce != nil {
                                if !debounce.Stop() {
                                        select {
                                        case <-debounce.C:
                                        default:
                                        }
                                }
                        }
                        debounce = time.AfterFunc(200*time.Millisecond, onChange)
                }

                for {
                        select {
                        case <-ctx.Done():
                                logger.WithComponent("json-repo").Infof("file watcher shutting down")
                                return
                        case event, ok := <-watcher.Events:
                                if !ok {
                                        return
                                }
                                if filepath.Base(event.Name) != r.base {
                                        continue
                                }
                                logger.WithComponent("json-repo").Tracef("file event detected: %s (op: %v)", event.Name, event.Op)
                                // Writes/Create/Chmod cover normal edits and atomic replace; trigger reload.
                                if event.Op&(fsnotify.Write|fsnotify.Create|fsnotify.Chmod) != 0 {
                                        schedule()
                                        continue
                                }
                                // Remove/Rename indicates the file was moved or replaced; wait for next Create.
                                if event.Op&(fsnotify.Remove|fsnotify.Rename) != 0 {
                                        schedule()
                                }
                        case err, ok := <-watcher.Errors:
                                if !ok {
                                        return
                                }
                                logger.WithComponent("json-repo").Errorf("watcher error: %v", err)
                        }
                }
        }()

        return nil
}

// MakeWatcherCallback returns a callback for file watcher that reloads cache from disk if needed.
// The provided ctx is used for the Load operation. Because the callback fires from a debounced
// timer after file events, it inherits the watcher context so cancellation propagates cleanly.
func (r *JSONRepository) MakeWatcherCallback(ctx context.Context, cacheStore CacheStore) func() {
        return func() {
                diskDoc, loadErr := r.Load(ctx)
                if loadErr != nil {
                        logger.WithComponent("json-repo").Errorf("watch reload failed: %v", loadErr)
                        return
                }
                cacheLastUpdate := cacheStore.GetLastUpdate()
                diskLastUpdate := diskDoc.Metadata.LastUpdate

                // If disk is not newer, skip reload
                if diskLastUpdate < cacheLastUpdate {
                        logger.WithComponent("json-repo").Infof("disk version is not newer than cache: diskLastUpdate = %d, cacheLastUpdate = %d", diskLastUpdate, cacheLastUpdate)
                        return
                }

                if cacheStore.IsDirty() {
                        logger.WithComponent("json-repo").Warn("disk data is newer but cache is dirty; skipping reload")
                        // the cache content will be written to file soon anyway
                        return
                }

                isDiskSameAsCache := false
                if diskLastUpdate == cacheLastUpdate {
                        // check if disk content is really the same as cache content
                        snapshot, err := cacheStore.Snapshot()
                        if err != nil {
                                logger.WithComponent("json-repo").Errorf("cache reload error: failed to get snapshot: %v", err)
                                return
                        }
                        isDiskSameAsCache = AreDataDocumentsEqual(&snapshot, diskDoc)
                }
                if !isDiskSameAsCache {
                        if err := cacheStore.Replace(*diskDoc); err != nil {
                                logger.WithComponent("json-repo").Errorf("cache reload error: %v", err)
                                return
                        }
                        logger.WithComponent("json-repo").Info("cache reloaded from newer disk version")
                }
        }
}

// AreDataDocumentsEqual compares two DataDocuments ignoring Metadata.
// Uses JSON serialization for flexible comparison (order-independent for object keys).
func AreDataDocumentsEqual(a, b *DataDocument) bool {
        if a == nil || b == nil {
                return a == b
        }

        // Marshal both to JSON
        aBytes, err := json.Marshal(a)
        if err != nil {
                return false
        }
        bBytes, err := json.Marshal(b)
        if err != nil {
                return false
        }

        // Unmarshal to generic maps
        var aMap, bMap map[string]interface{}
        if err := json.Unmarshal(aBytes, &aMap); err != nil {
                return false
        }
        if err := json.Unmarshal(bBytes, &bMap); err != nil {
                return false
        }

        // Remove metadata from comparison
        delete(aMap, "metadata")
        delete(bMap, "metadata")

        return reflect.DeepEqual(aMap, bMap)
}

// ListContainers returns a list of container names from the Docker daemon.
// Names are returned exactly as stored (case-sensitive), sorted alphabetically (case-insensitive).
// This includes all containers (running and stopped).
func (d *DockerRuntime) ListContainers(ctx context.Context) ([]string, error) {
        logger.WithComponent("docker").Debugf("listing containers")
        result, err := d.cli.ContainerList(ctx, client.ContainerListOptions{All: true})
        if err != nil {
                logger.WithComponent("docker").Errorf("failed to list containers: %v", err)
                return nil, fmt.Errorf("error listing containers: %w", err)
        }
        names := make([]string, 0, len(result.Items))
        for _, c := range result.Items {
                if len(c.Names) > 0 {
                        // Container names are prefixed with '/', strip it
                        name := strings.TrimPrefix(c.Names[0], "/")
                        names = append(names, name)
                }
        }
        // Sort names alphabetically, case-insensitive
        sort.Slice(names, func(i, j int) bool {
                return strings.ToLower(names[i]) < strings.ToLower(names[j])
        })
        logger.WithComponent("docker").Debugf("listed %d containers: %v", len(names), names)
        return names, nil
}

// Stats returns CPU and memory usage statistics for a container.
func (d *DockerRuntime) Stats(ctx context.Context, containerName string) (ContainerStats, error) {
        logger.WithComponent("docker").Debugf("getting stats for container: %s", containerName)

        result, err := d.cli.ContainerStats(ctx, containerName, client.ContainerStatsOptions{
                Stream:                false,
                IncludePreviousSample: true,
        })
        if err != nil {
                if errdefs.IsNotFound(err) {
                        logger.WithComponent("docker").Debugf("container not found: %s", containerName)
                        return ContainerStats{}, fmt.Errorf("container %s not found", containerName)
                }
                logger.WithComponent("docker").Errorf("failed to get stats for container %s: %v", containerName, err)
                return ContainerStats{}, fmt.Errorf("error getting stats for container %s: %w", containerName, err)
        }
        defer func() {
                if cerr := result.Body.Close(); cerr != nil {
                        logger.WithComponent("docker").Errorf("failed to close stats response body for container %s: %v", containerName, cerr)
                }
        }()

        var statsResponse container.StatsResponse
        if err := json.NewDecoder(result.Body).Decode(&statsResponse); err != nil {
                logger.WithComponent("docker").Errorf("failed to decode stats response for container %s: %v", containerName, err)
                return ContainerStats{}, fmt.Errorf("error decoding stats for container %s: %w", containerName, err)
        }

        stats := ContainerStats{
                CPUPercent: calculateCPUPercent(&statsResponse),
                MemoryMB:   float64(statsResponse.MemoryStats.Usage) / (1024 * 1024),
        }

        logger.WithComponent("docker").Debugf("container %s stats: CPU=%.2f%%, Memory=%.2f MB", containerName, stats.CPUPercent, stats.MemoryMB)
        return stats, nil
}

// calculateCPUPercent calculates the CPU usage percentage from Docker stats.
func calculateCPUPercent(stats *container.StatsResponse) float64 {
        cpuDelta := float64(stats.CPUStats.CPUUsage.TotalUsage - stats.PreCPUStats.CPUUsage.TotalUsage)
        systemDelta := float64(stats.CPUStats.SystemUsage - stats.PreCPUStats.SystemUsage)

        if systemDelta > 0 && cpuDelta > 0 {
                cpuPercent := (cpuDelta / systemDelta) * float64(stats.CPUStats.OnlineCPUs) * 100.0
                // Cap CPU percentage at 100% to avoid values > 100% on multi-core systems
                return math.Min(cpuPercent, 100.0)
        }
        return 0.0
}

// ContainerRegistry exposes the containers registered in the data document.

}

// MemoryRuntime is a temporary ContainerRuntime implementation that keeps state in memory.
// It is useful while the Docker-socket implementation is not available to execute tests or other development tasks.
type MemoryRuntime struct {
        running  map[string]bool
        registry ContainerRegistry
        mu       sync.RWMutex
}

func NewMemoryRuntime() *MemoryRuntime {
        return &MemoryRuntime{running: map[string]bool{}}
}

func NewMemoryRuntimeFromDocument(doc repository.DataDocument) *MemoryRuntime {
        mr := NewMemoryRuntime()
        for _, c := range doc.Containers {
                if c.Name == "" {
                        continue
                }
                if c.Running != nil {
                        mr.running[c.Name] = *c.Running
                }
        }
        return mr
}

// NewMemoryRuntimeWithRegistry creates a memory runtime that seeds its running
// state from doc (which may be nil) and answers ListContainers from the live
// registry. Without the registry, containers created after start-up would be
// reported as missing and the orphan cleanup would delete them.
func NewMemoryRuntimeWithRegistry(doc *repository.DataDocument, registry ContainerRegistry) *MemoryRuntime {
        mr := NewMemoryRuntime()
        if doc != nil {
                mr = NewMemoryRuntimeFromDocument(*doc)
        }
        mr.registry = registry
        return mr
}

        return isRunning, nil
}

        m.mu.Lock()
        defer m.mu.Unlock()

// ListContainers returns the names of containers known to the memory runtime:
// the ones it tracks plus every container registered in the data document. Names
// are returned exactly as they are stored (case-sensitive), sorted and without
// duplicates.
func (m *MemoryRuntime) ListContainers(_ context.Context) ([]string, error) {
        m.mu.RLock()
        names := make([]string, 0, len(m.running))
        for n := range m.running {
                names = append(names, n)
        }
        m.mu.RUnlock()

        if m.registry != nil {
                doc, err := m.registry.Snapshot()
                if err != nil {
                        return nil, fmt.Errorf("snapshot registry: %w", err)
                }
                for _, c := range doc.Containers {
                        if c.Name != "" {
                                names = append(names, c.Name)
                        }
                }
        }

        slices.Sort(names)
        names = slices.Compact(names)
        logger.WithComponent("memory-runtime").Debugf("listing containers: %v", names)
        return names, nil
}

// SystemStatsCollector defines the interface for collecting system stats
type SystemStatsCollector interface {

}

// cpuStats holds CPU statistics for delta calculation
type cpuStats struct {
        timestamp time.Time
        total     uint64
        idle      uint64
}

// Byte size constants for human-readable conversions
const (
        KB = 1024
        MB = KB * 1024
        GB = MB * 1024
)

// maxCPUDeltaAge is the maximum age of a previous CPU reading for delta calculation
const maxCPUDeltaAge = 5 * time.Second

// LinuxSystemStats implements SystemStatsCollector for Linux systems
type LinuxSystemStats struct {
        prevStats  *cpuStats
        mountPoint string
        mutex      sync.RWMutex
}

// NewSystemStatsCollector returns a new system stats collector based on the OS
func NewSystemStatsCollector(mountPoint string) SystemStatsCollector {
        return &LinuxSystemStats{
                mountPoint: mountPoint,
        }
}

// GetStats collects CPU, memory, and disk statistics
func (l *LinuxSystemStats) GetStats(ctx context.Context) (SystemStats, error) {
        if err := ctx.Err(); err != nil {
                return SystemStats{}, err
        }

        stats := SystemStats{
                Updated: time.Now(),
        }

        // Get CPU usage
        cpuPercent, err := l.getCPUUsage(ctx)
        if err != nil {
                return stats, fmt.Errorf("failed to get CPU stats: %w", err)
        }
        stats.CPU = cpuPercent

        // Get Memory usage
        memStats, err := l.getMemoryStats(ctx)
        if err != nil {
                return stats, fmt.Errorf("failed to get memory stats: %w", err)
        }
        stats.Memory = memStats

        // Get Disk usage
        diskStats, err := l.getDiskStats(ctx)
        if err != nil {
                return stats, fmt.Errorf("failed to get disk stats: %w", err)
        }
        stats.Disk = diskStats

        return stats, nil
}

// getCPUUsage calculates CPU usage percentage using delta measurement with /proc/stat
func (l *LinuxSystemStats) getCPUUsage(ctx context.Context) (int, error) {
        if err := ctx.Err(); err != nil {
                return 0, err
        }

        // Read /proc/stat for CPU stats
        cpuData, err := os.ReadFile("/proc/stat")
        if err != nil {
                return 0, err
        }

        // Parse first line: cpu  user nice system idle iowait irq softirq steal guest guest_nice
        lines := bytes.Split(cpuData, []byte("\n"))
        if len(lines) < 1 {
                return 0, fmt.Errorf("invalid /proc/stat content")
        }

        fields := bytes.Fields(lines[0])
        if len(fields) < 5 || string(fields[0]) != "cpu" {
                return 0, fmt.Errorf("unexpected /proc/stat format")
        }

        // Parse CPU fields
        total, idle := parseCPUFields(fields)

        // Initialize prevStats on the very first measurement so the next call can compute a delta.
        l.mutex.Lock()
        if l.prevStats == nil {
                l.prevStats = &cpuStats{total: total, idle: idle, timestamp: time.Now()}
                l.mutex.Unlock()

                // Sleep briefly to ensure the next /proc/stat sample reflects a non-trivial time slice.
                select {
                case <-ctx.Done():
                        return 0, ctx.Err()
                case <-time.After(100 * time.Millisecond):
                }

                return l.getCPUUsage(ctx)
        }

        // Snapshot prev under lock, then release the lock so the math below does not hold it across
        // further recursive or sleep calls.
        prev := *l.prevStats
        l.mutex.Unlock()

        // Calculate delta from the previous reading.
        if time.Since(prev.timestamp) < maxCPUDeltaAge {
                totalDelta := total - prev.total
                idleDelta := idle - prev.idle

                if totalDelta > 0 {
                        // CPU% = (total_delta - idle_delta) / total_delta * 100
                        idlePercent := float64(idleDelta) / float64(totalDelta) * 100.0
                        cpuPercent := 100.0 - idlePercent

                        // Cap at 100% and round up
                        result := int(math.Ceil(math.Min(cpuPercent, 100.0)))

                        // Update prevStats for next calculation
                        l.mutex.Lock()
                        l.prevStats = &cpuStats{total: total, idle: idle, timestamp: time.Now()}
                        l.mutex.Unlock()

                        return result, nil
                }
        }

        // Save current stats for next calculation
        l.mutex.Lock()
        l.prevStats = &cpuStats{total: total, idle: idle, timestamp: time.Now()}
        l.mutex.Unlock()

        // Return 0 on first measurement (need two readings for delta)
        return 0, nil
}

// parseCPUFields parses the CPU fields from /proc/stat and returns total and idle time.
// Idle includes the idle and iowait fields (fields 4 and 5 in /proc/stat after "cpu").
func parseCPUFields(fields [][]byte) (uint64, uint64) {
        var total uint64
        var idle uint64

        // Skip the "cpu" prefix and parse all numeric fields:
        // index 1=user, 2=nice, 3=system, 4=idle, 5=iowait, 6=irq, 7=softirq, 8=steal, 9=guest, 10=guest_nice
        for i := 1; i < len(fields); i++ {
                val, err := strconv.ParseUint(string(fields[i]), 10, 64)
                if err != nil {
                        continue
                }
                total += val

                // Idle = idle (field 4) + iowait (field 5)
                if i == 4 || i == 5 {
                        idle += val
                }
        }

        return total, idle
}

// getMemoryStats reads memory information from /proc/meminfo
func (l *LinuxSystemStats) getMemoryStats(ctx context.Context) (MemoryStats, error) {
        if err := ctx.Err(); err != nil {
                return MemoryStats{}, err
        }

        memData, err := os.ReadFile("/proc/meminfo")
        if err != nil {
                return MemoryStats{}, err
        }

        var memTotal, memFree, memAvailable uint64

        lines := bytes.Split(memData, []byte("\n"))
        for _, line := range lines {
                fields := bytes.Fields(line)
                if len(fields) < 2 {
                        continue
                }

                key := string(fields[0])
                val, err := strconv.ParseUint(string(fields[1]), 10, 64)
                if err != nil {
                        continue
                }
                // values in /proc/meminfo are in KB

                switch key {
                case "MemTotal:":
                        memTotal = val
                case "MemFree:":
                        memFree = val
                case "MemAvailable:":
                        memAvailable = val
                }
        }

        if memTotal == 0 {
                return MemoryStats{}, fmt.Errorf("could not read memory info")
        }

        // Use MemAvailable if available, otherwise calculate from MemFree
        available := memAvailable
        if available == 0 {
                available = memFree
        }

        totalGB := float64(memTotal) / 1024 / 1024
        freeGB := float64(available) / 1024 / 1024
        usedGB := totalGB - freeGB

        return MemoryStats{
                Used:  roundToTwoDecimals(usedGB),
                Total: roundToTwoDecimals(totalGB),
                Free:  roundToTwoDecimals(freeGB),
        }, nil
}

// getDiskStats reads disk usage using syscall.Statfs
func (l *LinuxSystemStats) getDiskStats(ctx context.Context) (DiskStats, error) {
        if err := ctx.Err(); err != nil {
                return DiskStats{}, err
        }

        // Actually use syscall to get disk usage
        var stat syscall.Statfs_t
        if err := syscall.Statfs(l.mountPoint, &stat); err != nil {
                return DiskStats{}, err
        }

        blockSize := uint64(stat.Bsize)
        totalBlocks := stat.Blocks
        freeBlocks := stat.Bfree

        totalGB := float64(blockSize*totalBlocks) / 1024 / 1024 / 1024
        freeGB := float64(blockSize*freeBlocks) / 1024 / 1024 / 1024
        usedGB := totalGB - freeGB

        return DiskStats{
                Used:  roundToTwoDecimals(usedGB),
                Total: roundToTwoDecimals(totalGB),
                Free:  roundToTwoDecimals(freeGB),
        }, nil
}

// roundToTwoDecimals rounds a float to two decimal places
func roundToTwoDecimals(v float64) float64 {
        return float64(int(v*100+0.5)) / 100
}

// GetSystemStatsJSON returns system stats as JSON string (for API responses)
func GetSystemStatsJSON(ctx context.Context, mountPoint string) (string, error) {
        collector := NewSystemStatsCollector(mountPoint)
        stats, err := collector.GetStats(ctx)
        if err != nil {
                return "", err
        }

        jsonData, err := json.Marshal(stats)
        if err != nil {
                return "", fmt.Errorf("failed to marshal stats: %w", err)
        }

        return string(jsonData), nil
}

type DayFlags struct {
        StartedDayKey string
        StoppedDayKey string
}

// PollingScheduler evaluates schedules on a fixed interval and performs at most
// one start and one stop action per container per day (in the configured timezone).
//
// Semantics:
//   - A timer may define both StartTime and StopTime, only StartTime (start-only: the
//     target is brought up at that time and left running) or only StopTime
//     (stop-only: the target is stopped at that time even if this scheduler did not
//     start it).
//   - A start+stop window containing the current time keeps the target up, including
//     against a stop-only timer that is due today.
//   - A start-only timer keeps the target up until midnight unless a stop-only timer
//     for the same target is due today: the explicit stop wins.
//   - If StartedDayKey == today, start is never attempted again today, regardless of running state.
//   - If StoppedDayKey == today, stop is never attempted again today.
//   - Stop evaluation is only performed after a start evaluation has happened that
//     day, unless a stop-only timer is due.
//
// NOTE: Flags are in-memory only.
type PollingScheduler struct {
        store   cache.ReadOnlyStore
        runtime runtime.ContainerRuntime
        loc     *time.Location
        flags   map[string]DayFlags
        poll    time.Duration
        mu      sync.Mutex
}

func NewPollingScheduler(store cache.ReadOnlyStore, rt runtime.ContainerRuntime, poll time.Duration, loc *time.Location) *PollingScheduler {
        if loc == nil {
                loc = time.Local
        }

        return &PollingScheduler{
                store:   store,
                runtime: rt,
                poll:    poll,
                loc:     loc,
                flags:   map[string]DayFlags{},
        }
}

func (s *PollingScheduler) Start(ctx context.Context) {
        logger.WithComponent("sched").Debugf("starting polling scheduler with interval: %v, timezone: %s", s.poll, s.loc.String())
        ticker := time.NewTicker(s.poll)
        go func() {
                defer ticker.Stop()
                for {
                        select {
                        case <-ctx.Done():
                                logger.WithComponent("sched").Info("scheduler stopped")
                                return
                        case <-ticker.C:
                                s.tick(ctx)
                        }
                }
        }()
}

func (s *PollingScheduler) tick(ctx context.Context) {
        logger.WithComponent("sched").Debugf("polling scheduler tick started")
        doc, err := s.store.Snapshot()
        if err != nil {
                logger.WithComponent("sched").Errorf("snapshot error: %v", err)
                return
        }

        now := time.Now().In(s.loc)
        todayKey := dayKey(now)
        logger.WithComponent("sched").Debugf("evaluating schedules for today: %s, current time: %s", todayKey, now.Format("15:04:05"))

        // Build lookup maps for efficient access during schedule evaluation.
        containersByName := map[string]repository.Container{}
        for _, c := range doc.Containers {
                if c.Name == "" {
                        continue
                }
                containersByName[c.Name] = c
        }

        groupsByName := map[string]repository.Group{}
        for _, g := range doc.Groups {
                if g.Name == "" {
                        continue
                }
                groupsByName[g.Name] = g
        }

        // Containers a start+stop window says must be running right now. These windows
        // declare their own end, so they win over a stop-only request.
        windowRunning := map[string]bool{}
        // Containers kept up by a start-only timer, which leaves them running until
        // midnight. A stop-only timer that is due today overrides this.
        openEndedRunning := map[string]bool{}
        // Containers a due stop-only timer asks to stop today. It overrides
        // openEndedRunning but not an active windowRunning.
        stopRequested := map[string]bool{}

        // Evaluate all schedules to determine which containers should be running based on active timers.
        for _, sched := range doc.Schedules {
                // Expand the schedule target into a list of container names (handles both "container" and "group" target types).
                containerNames := expandScheduleTargets(sched, containersByName, groupsByName)
                if len(containerNames) == 0 {
                        logger.WithComponent("sched").Debugf("schedule %s expanded to 0 containers", sched.ID)
                        continue
                }

                logger.WithComponent("sched").Tracef("schedule %s (target: %s) expanded to %d containers", sched.ID, sched.Target, len(containerNames))
                for _, timer := range sched.Timers {
                        if timer.Active != nil && !*timer.Active {
                                logger.WithComponent("sched").Debugf("timer inactive for schedule %s", sched.ID)
                                continue
                        }
                        hasStart := timer.StartTime != ""
                        hasStop := timer.StopTime != ""

                        // Start side: a start+stop timer runs inside its window, a start-only
                        // timer runs from its start time until the rest of the day.
                        if hasStart {
                                active := isStartOnlyActiveNow(timer, now)
                                if hasStop {
                                        active = isTimerActiveNow(timer, now)
                                }
                                if active {
                                        logger.WithComponent("sched").Debugf("timer %s is active for schedule %s, marking %d containers as running", timerLabel(timer), sched.ID, len(containerNames))
                                        // For each container targeted by this schedule, mark it as desired running if the container itself is active.
                                        for _, containerName := range containerNames {
                                                c, ok := containersByName[containerName]
                                                if !ok {
                                                        continue
                                                }
                                                // Respect the container's own active flag.
                                                if c.Active != nil && !*c.Active {
                                                        continue
                                                }
                                                if hasStop {
                                                        windowRunning[containerName] = true
                                                        continue
                                                }
                                                openEndedRunning[containerName] = true
                                        }
                                }
                        }

                        // Stop side: a stop-only timer asks for a stop once its stop time is due,
                        // regardless of whether this scheduler started the container today.
                        if hasStop && !hasStart && isStopDueNow(timer, now) {
                                logger.WithComponent("sched").Debugf("timer %s is due for schedule %s, requesting stop of %d containers", timerLabel(timer), sched.ID, len(containerNames))
                                for _, containerName := range containerNames {
                                        c, ok := containersByName[containerName]
                                        if !ok {
                                                continue
                                        }
                                        if c.Active != nil && !*c.Active {
                                                continue
                                        }
                                        stopRequested[containerName] = true
                                }
                        }
                }
        }

        // For each container, decide whether to start or stop based on desired state and day-key flags.
        for containerName := range containersByName {
                // Check for context cancellation to allow early exit during long iterations
                select {
                case <-ctx.Done():
                        logger.WithComponent("sched").Debugf("tick cancelled, exiting container loop")
                        return
                default:
                }

                flags := s.getFlags(containerName)
                // A window timer that contains the current time wins over a due stop-only
                // timer; an open-ended start-only timer does not, so an explicit stop can end
                // the "left running until midnight" state it created.
                shouldRun := windowRunning[containerName] ||
                        (openEndedRunning[containerName] && !stopRequested[containerName])
                logger.WithComponent("sched").Debugf("container %s: shouldRun=%v, stopRequested=%v, startedToday=%v, stoppedToday=%v",
                        containerName, shouldRun, stopRequested[containerName], flags.StartedDayKey == todayKey, flags.StoppedDayKey == todayKey)
                // If we already attempted to start this container today, skip to avoid repeated attempts.
                // This enforces "at most one start per day" even if the container stops later.
                if shouldRun {
                        if flags.StartedDayKey == todayKey {
                                logger.WithComponent("sched").Debugf("container %s already started today, skipping", containerName)
                                continue
                        }
                        // Check current runtime state.
                        running, err := s.runtime.IsRunning(ctx, containerName)
                        if err != nil {
                                logger.WithComponent("sched").Errorf("IsRunning(%s) error: %v", containerName, err)
                                continue
                        }
                        if !running {
                                if err := s.runtime.Start(ctx, containerName); err != nil {
                                        logger.WithComponent("sched").Errorf("Start(%s) error: %v", containerName, err)
                                        continue
                                }
                                logger.WithComponent("sched").Infof("started %s", containerName)
                        }
                        // Mark that a start attempt was made today (even if it was already running).
                        flags.StartedDayKey = todayKey
                        s.setFlags(containerName, flags)
                        continue
                }

                // Container should not be running now.
                // Stop evaluation happens when a stop-only timer is due today, or when a
                // start evaluation already occurred today (to avoid premature stops).
                if !stopRequested[containerName] && flags.StartedDayKey != todayKey {
                        logger.WithComponent("sched").Tracef("container %s not started today, skipping stop evaluation", containerName)
                        continue
                }
                // If we already attempted to stop this container today, skip.
                if flags.StoppedDayKey == todayKey {
                        logger.WithComponent("sched").Debugf("container %s already stopped today, skipping", containerName)
                        continue
                }

                running, err := s.runtime.IsRunning(ctx, containerName)
                if err != nil {
                        logger.WithComponent("sched").Errorf("IsRunning(%s) error: %v", containerName, err)
                        continue
                }
                if running {
                        if err := s.runtime.Stop(ctx, containerName); err != nil {
                                logger.WithComponent("sched").Errorf("Stop(%s) error: %v", containerName, err)
                                continue
                        }
                        logger.WithComponent("sched").Infof("stopped %s", containerName)
                }
                // Mark that a stop attempt was made today (even if it was already stopped).
                flags.StoppedDayKey = todayKey
                s.setFlags(containerName, flags)
        }
        logger.WithComponent("sched").Debugf("polling scheduler tick completed")
}

func (s *PollingScheduler) getFlags(containerName string) DayFlags {
        s.mu.Lock()
        defer s.mu.Unlock()
        return s.flags[containerName]
}

func (s *PollingScheduler) setFlags(containerName string, flags DayFlags) {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.flags[containerName] = flags
}

func dayKey(t time.Time) string {
        return t.Format("2006-01-02")
}

func expandScheduleTargets(
        sched repository.Schedule,
        containersByName map[string]repository.Container,
        groupsByName map[string]repository.Group,
) []string {
        if sched.Target == "" {
                return nil
        }

        switch sched.TargetType {
        case "container":
                if _, ok := containersByName[sched.Target]; !ok {
                        return nil
                }
                return []string{sched.Target}
        case "group":
                g, ok := groupsByName[sched.Target]
                if !ok {
                        return nil
                }
                if g.Active != nil && !*g.Active {
                        return nil
                }
                out := make([]string, 0, len(g.Container))
                for _, name := range g.Container {
                        if name == "" {
                                continue
                        }
                        out = append(out, name)
                }
                return out
        default:
                return nil
        }
}

// timeOfDayLayout is the HH:MM layout used by timer start/stop times in the
// schedule payloads and in config/data/config.json.
const timeOfDayLayout = "15:04"

// timerLabel renders a timer for log messages and covers start-only and
// stop-only timers, where one of the two times is empty.
func timerLabel(timer repository.Timer) string {
        switch {
        case timer.StartTime != "" && timer.StopTime != "":
                return timer.StartTime + "-" + timer.StopTime
        case timer.StartTime != "":
                return "start " + timer.StartTime
        default:
                return "stop " + timer.StopTime
        }
}

// isStartOnlyActiveNow reports whether a start-only timer is active: the weekday
// matches and the start time has already passed today. The container stays
// desired-running until midnight; what happens after that is decided by other
// schedules or by manual actions.
func isStartOnlyActiveNow(timer repository.Timer, now time.Time) bool {
        if timer.StartTime == "" || !containsInt(timer.Days, int(now.Weekday())) {
                return false
        }
        startClock, err := time.Parse(timeOfDayLayout, timer.StartTime)
        if err != nil {
                return false
        }
        start := time.Date(now.Year(), now.Month(), now.Day(), startClock.Hour(), startClock.Minute(), 0, 0, now.Location())
        return !now.Before(start)
}

// isStopDueNow reports whether a stop-only timer must stop its targets today:
// the weekday matches and the stop time has already passed.
func isStopDueNow(timer repository.Timer, now time.Time) bool {
        if timer.StopTime == "" || !containsInt(timer.Days, int(now.Weekday())) {
                return false
        }
        stopClock, err := time.Parse(timeOfDayLayout, timer.StopTime)
        if err != nil {
                return false
        }
        stop := time.Date(now.Year(), now.Month(), now.Day(), stopClock.Hour(), stopClock.Minute(), 0, 0, now.Location())
        return !now.Before(stop)
}

func isTimerActiveNow(timer repository.Timer, now time.Time) bool {
        startClock, err := time.Parse(timeOfDayLayout, timer.StartTime)
        if err != nil {
                return false
        }
        stopClock, err := time.Parse(timeOfDayLayout, timer.StopTime)
        if err != nil {
                return false
        }

        // Check windows anchored to today and yesterday (handles cross-midnight).
        for _, dayOffset := range []int{0, -1} {
                base := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location()).AddDate(0, 0, dayOffset)

                weekday := int(base.Weekday())
                if !containsInt(timer.Days, weekday) {
                        continue
                }

                start := time.Date(base.Year(), base.Month(), base.Day(), startClock.Hour(), startClock.Minute(), 0, 0, now.Location())
                stop := time.Date(base.Year(), base.Month(), base.Day(), stopClock.Hour(), stopClock.Minute(), 0, 0, now.Location())
                if !stop.After(start) {
                        stop = stop.Add(24 * time.Hour)
                }

                if (now.Equal(start) || now.After(start)) && now.Before(stop) {
                        return true
                }
        }

        return false
}

func containsInt(list []int, v int) bool {
        for _, x := range list {
                if x == v {
                        return true
                }
        }
        return false
}

// Package version exposes the software version string and related build
// metadata used to label the running build. The values are intentionally
// internal — users cannot override them via homepage.yaml, environment
// variables, or runtime configuration.
//
// The variables below are mutable (not const) so they can be overridden at
// build time via -ldflags. `task build` / `task build:cli` (and the multi-arch
// Docker workflow) inject four values from the surrounding toolchain:
//
//        go build -ldflags "\
//          -s -w \
//          -X github.com/asgambat/gospin/internal/version.Version=$(git describe ...) \
//          -X github.com/asgambat/gospin/internal/version.BuildTime=$(date ...) \
//          -X github.com/asgambat/gospin/internal/version.GitCommit=$(git rev-parse ...) \
//          -X github.com/asgambat/gospin/internal/version.GoVersion=$(go version ...)"
//
// At runtime the values flow into the homepage JSON response (top-level
// fields, never under "settings" — see internal/api/controller/homepage_controller.go)
// and are rendered in the homepage footer.
package version

// Version is the software version reported by the server. It is rendered in
// the homepage footer (bottom-right) as part of the JSON response's top-level
// "version" field and surfaced to the UI.
//
// At build time this is normally populated by `git describe --tags --always --dirty`
// via the build flags in Taskfile.yml (e.g. "v1.2.3", "v1.2.3-4-gabc1234", or
// "v1.2.3-dirty").
var Version = "0.1.0"

// BuildTime is the UTC timestamp at which the binary was produced. Populated
// at build time from `date -u '+%Y-%m-%d_%H:%M:%S'`. Useful for telling two
// builds of the same commit apart and for correlating a deployed artifact
// with CI logs.
var BuildTime = "unknown"

// GitCommit is the short SHA of the commit the binary was built from.
// Populated at build time from `git rev-parse --short HEAD`.
var GitCommit = "unknown"

// GoVersion is the Go toolchain version the binary was built with (e.g.
// "go1.25.6"). Populated at build time from `go version`.
var GoVersion = "unknown"

// BuildInfo returns a human-readable, single-line summary of the build
// metadata. Kept stable (no newline, no trailing whitespace) so it can be
// embedded verbatim in the homepage footer tooltip and the startup log.
//
// Dev runs (no -ldflags passed) leave GitCommit/BuildTime/GoVersion at their
// "unknown" defaults; those segments are skipped here so the output does
// not render four literal "unknown" tokens.
func BuildInfo() string {
        s := Version
        if GitCommit != "" && GitCommit != "unknown" {
                s += " (" + GitCommit + ")"
        }
        if BuildTime != "" && BuildTime != "unknown" {
                s += " built " + BuildTime + " UTC"
        }
        if GoVersion != "" && GoVersion != "unknown" {
                s += " with " + GoVersion
        }
        return s
}
